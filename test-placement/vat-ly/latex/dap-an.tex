\section{Phần 1 — Cơ học}

\begin{dapan}[1 — Đáp án B]
Vectơ phải có cả độ lớn và hướng. Xét từng phương án:

\begin{itemize}[leftmargin=1.4em, itemsep=2pt]
\item \cjk{磁感应强度} (cảm ứng từ): vectơ, có độ lớn và hướng
\item \cjk{动量} (động lượng) $\vec{p} = m\vec{v}$: vectơ, cùng hướng với vận tốc
\item \cjk{电场强度} (cường độ điện trường): vectơ
\item \textbf{\cjk{势能} (thế năng)}: vô hướng, chỉ có độ lớn, không có hướng
\end{itemize}
\end{dapan}

\begin{dapan}[2 — Đáp án B]
Trọng lực trên Hỏa tinh bằng $\dfrac{1}{3}$ trọng lực trên Trái Đất.
\[
P = \frac{1}{3}mg = \frac{1}{3}\times 4{,}8 \times 10 = 16 \text{ N}
\]

\nhan{Bẫy} Đọc kỹ đề: hỏi trọng lực \textbf{trên Hỏa tinh}, không phải trên Trái Đất. Đáp án C (48 N) là trọng lực trên Trái Đất.
\end{dapan}

\begin{dapan}[3 — Đáp án A]
Vật đang \textbf{đứng yên}. Cần kiểm tra lực kéo có thắng được ma sát nghỉ cực đại không:
\[
f_{\max} = \mu mg = 0{,}1 \times 4 \times 10 = 4 \text{ N}
\]
Lực kéo $F = 2$ N $< 4$ N, nên \textbf{vật không chuyển động}. Khi đó ma sát nghỉ cân bằng đúng với lực kéo:
\[
f = F = 2 \text{ N}
\]

\nhan{Bẫy} Đáp án C (4 N) là ma sát \textbf{cực đại}, chỉ đạt được khi vật bắt đầu trượt. Vật chưa chuyển động thì không dùng giá trị này.
\end{dapan}

\begin{dapan}[4 — Đáp án C]
Từ quãng đường suy ra gia tốc, rồi suy ra lực.
\[
s = \frac{1}{2}at^2 \implies 8 = \frac{1}{2}a(2)^2 \implies a = 4 \text{ m/s}^2
\]
\[
F = ma = 2 \times 4 = 8 \text{ N}
\]
\end{dapan}

\begin{dapan}[5 — Đáp án D]
Quãng đường rơi trong 4 giây:
\[
h = \frac{1}{2}gt^2 = \frac{1}{2}\times 10 \times 4^2 = 80 \text{ m}
\]
Độ cao còn lại so với mặt đất: $100 - 80 = 20$ m.

\nhan{Bẫy} Đề hỏi \textbf{khoảng cách đến mặt đất}, không phải quãng đường đã rơi. Đáp án C (80 m) là quãng đường rơi.
\end{dapan}

\begin{dapan}[6 — Đáp án C]
Trên mặt phẳng nghiêng không ma sát, gia tốc là $a = g\sin\theta$:
\[
a = 10 \times \sin 30^\circ = 5 \text{ m/s}^2
\]
\[
s = \frac{1}{2}at^2 = \frac{1}{2}\times 5 \times 2^2 = 10 \text{ m}
\]
\end{dapan}

\begin{dapan}[7 — Đáp án D]
\begin{itemize}[leftmargin=1.4em, itemsep=2pt]
\item \cjk{路程} (quãng đường): vô hướng, luôn dương
\item \cjk{位移} (độ dời): vectơ, có thể âm hoặc bằng 0
\end{itemize}

Khi vật chuyển động \textbf{thẳng theo một chiều} không đổi hướng, độ lớn độ dời bằng quãng đường. Vậy D đúng.

\nhan{Bẫy} Đáp án B sai vì dùng từ \cjk{一定} (nhất định): hai đại lượng này \textbf{có thể} bằng nhau, chỉ không phải luôn luôn.
\end{dapan}

\begin{dapan}[8 — Đáp án B]
\[
p = mv = 0{,}2 \times 20 = 4 \text{ kg·m/s}
\]
\end{dapan}

\begin{dapan}[9 — Đáp án C]
Định lý động năng: công của ngoại lực bằng độ biến thiên động năng.
\[
W = \frac{1}{2}m(v_2^2 - v_1^2) = \frac{1}{2}\times 3 \times (6^2 - 2^2) = \frac{1}{2}\times 3 \times 32 = 48 \text{ J}
\]

\nhan{Bẫy} Đáp án A (54 J) là động năng \textbf{cuối} ($\frac{1}{2}\times3\times36$), không phải công. Đề hỏi công, tức độ \textbf{biến thiên}.
\end{dapan}

\section{Phần 2 — Điện từ}

\begin{dapan}[10 — Đáp án A]
Lực từ tác dụng lên dây dẫn thẳng đặt vuông góc với từ trường: $F = BIL$.
\[
B = \frac{F}{IL} = \frac{1{,}5}{10 \times 0{,}5} = 0{,}3 \text{ T}
\]
\end{dapan}

\begin{dapan}[11 — Đáp án D]
Hai điện trở mắc song song:
\[
\frac{1}{R} = \frac{1}{R_1} + \frac{1}{R_2} = \frac{1}{20} + \frac{1}{30} = \frac{3+2}{60} = \frac{5}{60} \implies R = 12 \ \Omega
\]

\nhan{Cách nhanh} Công thức song song hai điện trở: $R = \dfrac{R_1 R_2}{R_1 + R_2} = \dfrac{20 \times 30}{50} = 12\ \Omega$.

\nhan{Bẫy} Đáp án A (50 $\Omega$) là kết quả phép \textbf{cộng}, tức mắc nối tiếp.
\end{dapan}

\begin{dapan}[12 — Đáp án B]
\[
F = qE = 3{,}0\times10^{-9} \times 6{,}0\times10^4 = 18\times10^{-5} = 1{,}8\times10^{-4} \text{ N}
\]

\nhan{Chú ý} Đáp án C sai bậc 10 ($10^4$ thay vì $10^{-4}$), do lỗi tính lũy thừa. Không dùng máy tính nên phải cẩn thận với số mũ.
\end{dapan}

\begin{dapan}[13 — Đáp án A]
Khi dây dẫn \textbf{vuông góc} với từ trường thì lực từ đạt cực đại: $F = BIL$.
\[
I = \frac{F}{BL} = \frac{0{,}08}{0{,}8 \times 0{,}2} = 0{,}5 \text{ A}
\]
\end{dapan}

\begin{dapan}[14 — Đáp án D]
\textbf{Trước khi tiếp xúc}, lực Coulomb tỉ lệ với tích điện tích:
\[
F_1 \propto |8q \times (-4q)| = 32q^2
\]
\textbf{Sau khi tiếp xúc}, hai quả cầu giống hệt nhau nên chia đều điện tích:
\[
q' = \frac{8q + (-4q)}{2} = 2q
\]
\[
F_2 \propto (2q)^2 = 4q^2
\]
Tỉ số: $\dfrac{F_2}{F_1} = \dfrac{4q^2}{32q^2} = \dfrac{1}{8}$.

\nhan{Chú ý} Sau khi tiếp xúc, hai điện tích \textbf{cùng dấu} nên lực đổi từ hút sang đẩy. Đề chỉ hỏi độ lớn nên không cần xét dấu.
\end{dapan}

\sectiondautrang{Phần 3 — Dao động, sóng, quang, nhiệt}

\begin{dapan}[15 — Đáp án B]
Tần số là nghịch đảo chu kỳ:
\[
f = \frac{1}{T} = \frac{1}{2} = 0{,}5 \text{ Hz}
\]
\end{dapan}

\begin{dapan}[16 — Đáp án A]
Công thức sóng: $v = \lambda f$.
\[
\lambda = \frac{v}{f} = \frac{340}{200} = 1{,}7 \text{ m}
\]
\end{dapan}

\begin{dapan}[17 — Đáp án C]
Theo định luật phản xạ, góc phản xạ bằng góc tới: $r = i = 30^\circ$.

Tia tới và tia phản xạ nằm \textbf{hai bên} pháp tuyến, nên góc giữa chúng:
\[
i + r = 30^\circ + 30^\circ = 60^\circ
\]

\nhan{Bẫy} Đáp án A ($30^\circ$) là góc tới, không phải góc giữa hai tia. Đọc kỹ đề hỏi góc nào.
\end{dapan}

\begin{dapan}[18 — Đáp án D]
Định luật Boyle--Mariotte: khi nhiệt độ không đổi thì $pV = \text{const}$.

Vì $p$ giảm nên $V$ phải \textbf{tăng} để tích không đổi.
\end{dapan}

\begin{dapan}[19 — Đáp án C]
Vật rơi tự do từ độ cao 5 m, vận tốc đầu bằng 0:
\[
v = \sqrt{2gh} = \sqrt{2 \times 10 \times 5} = \sqrt{100} = 10 \text{ m/s}
\]
\end{dapan}

\begin{dapan}[20 — Đáp án A]
Công của lực kéo chỉ phụ thuộc lực kéo và quãng đường:
\[
W_F = F \cdot s = 20 \times 5 = 100 \text{ J}
\]

\nhan{Chú ý} Lực ma sát $f = 5$ N không ảnh hưởng đến công của \textbf{lực kéo}, nó chỉ làm giảm công \textbf{toàn phần}. Đáp án B (75 J) là công toàn phần $(F-f)s = 15\times5$.
\end{dapan}

\section{Nhận xét về đề gốc}

Đề này rút từ đề chính thức CSCA tháng 12.2025 (48 câu). Khi dùng đề gốc để luyện tập, cần lưu ý ba điểm.

\textbf{Câu 43 có đáp án sai.} Đề ghi A (200 m/s), nhưng tính đúng phải là B (100 m/s). Đạn xuyên hai tấm gỗ giống hệt nhau, mỗi tấm hấp thụ cùng một lượng động năng:
\[
\frac{1}{2}m(700^2 - 500^2) = \frac{1}{2}m(500^2 - v_2^2)
\]
\[
700^2 - 500^2 = 240\,000 = 500^2 - v_2^2
\]
\[
v_2^2 = 250\,000 - 240\,000 = 10\,000 \implies v_2 = 100 \text{ m/s}
\]

\textbf{Chín câu phụ thuộc hình vẽ}, không dùng được nếu tách khỏi đề gốc: câu 13, 16, 25, 37, 38, 40, 45, 46, 47.
