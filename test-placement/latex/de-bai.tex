\section*{Phần 1 — Tập hợp và bất đẳng thức (2 câu)}
\addcontentsline{toc}{section}{Phần 1 — Tập hợp và bất đẳng thức}

\begin{baitap}[1]
设集合 $A=\{x \mid -2<x\le 3\}$。下列哪个说法是正确的？（ ）\\[4pt]
A. $2\notin A$ \quad B. $-3\in A$ \quad C. $3\in A$ \quad D. $-2\in A$
\end{baitap}

\begin{baitap}[2]
不等式 $x^2-x-6>0$ 的解集是（ ）\\[4pt]
A. $\{x \mid -2<x<3\}$ \quad B. $\{x \mid x<-2 \text{ 或 } x>3\}$ \quad C. $\{x \mid x<-2\}$ \quad D. $\{x \mid x>3\}$
\end{baitap}

\section*{Phần 2 — Hàm số và dãy số (10 câu)}
\addcontentsline{toc}{section}{Phần 2 — Hàm số và dãy số}

\begin{baitap}[3]
设 $\alpha=\dfrac{5\pi}{6}$，则 $\cos\alpha=$（ ）\\[4pt]
A. $-\dfrac{1}{2}$ \quad B. $\dfrac{\sqrt{3}}{2}$ \quad C. $\dfrac{1}{2}$ \quad D. $-\dfrac{\sqrt{3}}{2}$
\end{baitap}

\begin{baitap}[4]
在其定义域上，函数 $f(x)=\ln(x^2)$ 是（ ）\\[4pt]
A. 奇函数 \quad B. 奇偶性不确定 \quad C. 既不是奇函数也不是偶函数 \quad D. 偶函数
\end{baitap}

\begin{baitap}[5]
函数 $f(x)=\sin x$ 单调递增的区间是（ ）\\[4pt]
A. $[-\pi,\pi]$ \quad B. $[0,\pi]$ \quad C. $\left[-\dfrac{\pi}{2},\dfrac{\pi}{2}\right]$ \quad D. $(-\infty,+\infty)$
\end{baitap}

\begin{baitap}[6]
设 $\{a_n\}$ 是等差数列，首项 $a_1=2$，公差 $d=3$。则 $a_{100}=$（ ）\\[4pt]
A. 298 \quad B. 302 \quad C. 305 \quad D. 299
\end{baitap}

\begin{baitap}[7]
若指数函数 $y=a^x$ 在 $\mathbb{R}$ 上单调递增，则 $a$ 的取值范围是（ ）\\[4pt]
A. $a>1$ \quad B. $a<0$ 或 $a>1$ \quad C. $0<a<1$ \quad D. $a>0$
\end{baitap}

\begin{baitap}[8]
设 $a=\lg 4$，$b=\lg 5$。则 $a+2b=$（ ）\\[4pt]
A. 100 \quad B. 4 \quad C. 2 \quad D. 5
\end{baitap}

\begin{baitap}[9]
函数 $f(x)=3\cos\left(4x-\dfrac{\pi}{3}\right)$ 的最小正周期是（ ）\\[4pt]
A. $\dfrac{\pi}{3}$ \quad B. $\dfrac{\pi}{2}$ \quad C. $\dfrac{\pi}{4}$ \quad D. $2\pi$
\end{baitap}

\begin{baitap}[10]
已知 $\cos\alpha=-\dfrac{1}{2}$，$\alpha\in\left(\dfrac{\pi}{2},\pi\right)$，则 $\sin\dfrac{\alpha}{2}$ 的值为（ ）\\[4pt]
A. $\dfrac{\sqrt{3}}{2}$ \quad B. $-\dfrac{1}{2}$ \quad C. $-\dfrac{\sqrt{3}}{2}$ \quad D. $\dfrac{1}{2}$
\end{baitap}

\begin{baitap}[11]
若 $\cos\alpha=\dfrac{1}{3}$ 且 $-\dfrac{\pi}{2}<\alpha<0$，则 $\tan\dfrac{\alpha}{2}=$（ ）\\[4pt]
A. $\dfrac{\sqrt{2}}{2}$ \quad B. $-\sqrt{2}$ \quad C. $-\dfrac{\sqrt{2}}{2}$ \quad D. $\sqrt{2}$
\end{baitap}

\begin{baitap}[12]
设 $f(x)=\log_5(2x+1)$。则不等式 $f(x)\ge 0$ 的解集是（ ）\\[4pt]
A. $\{x \mid x\ge 1\}$ \quad B. $\{x \mid x>0\}$ \quad C. $\{x \mid x>1\}$ \quad D. $\{x \mid x\ge 0\}$
\end{baitap}

\sectiondautrang{Phần 3 — Hình học và đại số (7 câu)}
\addcontentsline{toc}{section}{Phần 3 — Hình học và đại số}

\begin{baitap}[13]
直线 $x+\sqrt{3}y-2=0$ 的倾斜角是（ ）\\[4pt]
A. $60^\circ$ \quad B. $120^\circ$ \quad C. $150^\circ$ \quad D. $30^\circ$
\end{baitap}

\begin{baitap}[14]
已知点 $A(-2,-1)$ 和 $B(a,3)$，且 $|AB|=5$。则 $a$ 的值为（ ）\\[4pt]
A. $1$ 或 $-5$ \quad B. $-1$ 或 $5$ \quad C. $\pm 5$ \quad D. $\pm 1$
\end{baitap}

\begin{baitap}[15]
若直线 $l$ 经过两点 $M(0,-1)$ 和 $N(-2,0)$，则 $l$ 的方程为（ ）\\[4pt]
A. $x+y+2=0$ \quad B. $x+2y+2=0$ \quad C. $2x+y+2=0$ \quad D. $2x+y+1=0$
\end{baitap}

\begin{baitap}[16]
圆心在 $A(-3,2)$，半径为 $3$ 的圆的方程是（ ）\\[4pt]
A. $(x+3)^2+(y-2)^2=3$ \quad B. $(x+3)^2+(y-2)^2=9$ \quad C. $(x-3)^2+(y+2)^2=9$ \quad D. $(x-3)^2+(y+2)^2=3$
\end{baitap}

\begin{baitap}[17]
已知椭圆的长轴长为 $10$，两焦点 $F_1$、$F_2$ 分别在 $(3,0)$ 和 $(-3,0)$。则该椭圆的方程为（ ）\\[4pt]
A. $\dfrac{x^2}{9}+\dfrac{y^2}{25}=1$ \quad B. $\dfrac{x^2}{25}+\dfrac{y^2}{16}=1$ \quad C. $\dfrac{x^2}{16}+\dfrac{y^2}{25}=1$ \quad D. $\dfrac{x^2}{16}+\dfrac{y^2}{9}=1$
\end{baitap}

\begin{baitap}[18]
设向量 $\vec{a}=(-2,3)$，向量 $\vec{b}=(2,1)$。则 $\vec{a}-2\vec{b}=$（ ）\\[4pt]
A. $(-6,-1)$ \quad B. $(6,-1)$ \quad C. $(-6,1)$ \quad D. $(6,1)$
\end{baitap}

\begin{baitap}[19]
若直线 $l$ 垂直于直线 $l_1:x+2y+4=0$，且经过直线 $l_2:x+y+1=0$ 和 $l_3:2x+y-1=0$ 的交点，则 $l$ 的方程为（ ）\\[4pt]
A. $2x-y-7=0$ \quad B. $x+2y-3=0$ \quad C. $x-2y-1=0$ \quad D. $2x+y-5=0$
\end{baitap}

\begin{baitap}[20]
一个袋子里有五个大小形状完全相同的小球，包括两个红球（标号 1、2）、两个黑球（标号 1、2）和一个白球（标号 1）。现从中随机抽取两个球（不放回）。求两个球颜色不同的概率，以及至少有一个球标号为 1 的概率，这两个概率分别为（ ）\\[4pt]
A. $\dfrac{4}{5}$，$\dfrac{7}{10}$ \quad B. $\dfrac{3}{5}$，$\dfrac{7}{10}$ \quad C. $\dfrac{3}{5}$，$\dfrac{9}{10}$ \quad D. $\dfrac{4}{5}$，$\dfrac{9}{10}$
\end{baitap}
