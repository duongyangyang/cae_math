\section{KIẾN THỨC TRỌNG TÂM}

\subsection{Vector trong mặt phẳng}

\begin{dinhnghia}[{1 — Vector và môđun}]
Vector là đại lượng có cả độ lớn và hướng. Vector $a = (x, y)$ có môđun $\lvert a \rvert = \sqrt{x^2 + y^2}$.
\end{dinhnghia}
\begin{dinhnghia}[{2 — Tích vô hướng}]
Với $a = (x_1, y_1)$ và $b = (x_2, y_2)$:

$$a \cdot b = x_1x_2 + y_1y_2 = \lvert a \rvert \lvert b \rvert \cos\theta$$

trong đó $\theta$ là góc giữa hai vector.
\end{dinhnghia}
\begin{dinhly}[{1 — Các phép toán tọa độ}]
Với $a = (x_1, y_1)$, $b = (x_2, y_2)$ và số thực $\lambda$:

$$a + b = (x_1 + x_2,\ y_1 + y_2), \qquad a - b = (x_1 - x_2,\ y_1 - y_2), \qquad \lambda a = (\lambda x_1,\ \lambda y_1)$$
\end{dinhly}
\begin{dinhly}[{2 — Điều kiện vuông góc}]
Với hai vector khác vector không:

$$a \perp b \iff a \cdot b = 0 \iff x_1x_2 + y_1y_2 = 0$$
\end{dinhly}
\begin{dinhly}[{3 — Điều kiện song song}]
Với $b$ khác vector không:

$$a \parallel b \iff x_1y_2 - x_2y_1 = 0$$
\end{dinhly}
\begin{dinhly}[{4 — Công thức góc}]
Góc $\theta$ giữa hai vector khác vector không xác định bởi:

$$\cos\theta = \frac{a \cdot b}{\lvert a \rvert \lvert b \rvert} = \frac{x_1x_2 + y_1y_2}{\sqrt{x_1^2 + y_1^2}\sqrt{x_2^2 + y_2^2}}$$
\end{dinhly}
\begin{dinhly}[{5 — Hình chiếu}]
Hình chiếu vô hướng của $a$ lên phương $b$ bằng:

$$\frac{a \cdot b}{\lvert b \rvert}$$

\nhan{Chú ý} Hình chiếu là một \textbf{số} có thể âm, không phải một vector. Đề dùng chữ 投影 mà hỏi một giá trị trong bốn phương án số thì đó là hình chiếu vô hướng.
\end{dinhly}
\begin{dinhly}[{6 — Tọa độ trung điểm}]
Trung điểm $M$ của đoạn $AB$ với $A(x_1, y_1)$, $B(x_2, y_2)$ có tọa độ:

$$M = \left(\frac{x_1 + x_2}{2},\ \frac{y_1 + y_2}{2}\right)$$

\nhan{Chú ý} Vector $\overrightarrow{AB} = B - A$, tức lấy tọa độ điểm cuối trừ tọa độ điểm đầu. Đảo thứ tự là sai dấu cả hai thành phần, và đây là lỗi mất điểm phổ biến nhất ở dạng tìm tọa độ điểm.

\nhan{Cách nhanh} Hai vector cùng phương khi tỉ số các thành phần tương ứng bằng nhau. Với $a = (1,2)$, $b = (2,4)$: $\dfrac{2}{1} = \dfrac{4}{2} = 2$ nên cùng phương. Cách này nhanh hơn viết phương trình $x_1y_2 = x_2y_1$ nhưng phải kiểm tra trường hợp có thành phần bằng $0$.
\end{dinhly}
\subsection{Số phức}

\begin{dinhnghia}[{3 — Số phức và các thành phần}]
Số phức có dạng $z = a + bi$ với $a, b \in \mathbb{R}$ và $i^2 = -1$. Số $a$ gọi là \textbf{phần thực}, số $b$ gọi là \textbf{phần ảo}.
\end{dinhnghia}
\begin{dinhnghia}[{4 — Số phức liên hợp}]
Số phức liên hợp của $z = a + bi$ là $\overline{z} = a - bi$.
\end{dinhnghia}
\begin{dinhnghia}[{5 — Môđun số phức}]
Môđun của $z = a + bi$ là:

$$\lvert z \rvert = \sqrt{a^2 + b^2}$$
\end{dinhnghia}
\begin{dinhly}[{7 — Bốn phép toán}]
Với $z_1 = a + bi$ và $z_2 = c + di$:

$$z_1 \pm z_2 = (a \pm c) + (b \pm d)i$$

$$z_1 \cdot z_2 = (ac - bd) + (ad + bc)i$$

$$\frac{z_1}{z_2} = \frac{(a + bi)(c - di)}{c^2 + d^2} \quad (z_2 \ne 0)$$
\end{dinhly}
\begin{dinhly}[{8 — Chu kì lũy thừa của $i$}]
Với mọi số nguyên $n$:

$$i^{4n} = 1, \qquad i^{4n+1} = i, \qquad i^{4n+2} = -1, \qquad i^{4n+3} = -i$$

\nhan{Cách nhanh} Chia số mũ cho $4$ và chỉ quan tâm số dư. Với $i^{2025}$: $2025 = 4 \times 506 + 1$ nên $i^{2025} = i$. Không cần viết chu kì ra giấy.
\end{dinhly}
\begin{dinhly}[{9 — Các đẳng thức về môđun và liên hợp}]
$$z \cdot \overline{z} = \lvert z \rvert^2 = a^2 + b^2, \qquad \lvert \overline{z} \rvert = \lvert z \rvert, \qquad \overline{z_1 \cdot z_2} = \overline{z_1} \cdot \overline{z_2}$$
\end{dinhly}
\begin{dinhly}[{10 — Điều kiện số thực và số thuần ảo}]
Cho $z = a + bi$ với $a, b \in \mathbb{R}$:

$$z \in \mathbb{R} \iff b = 0; \qquad z \ \text{là số thuần ảo} \iff a = 0 \ \text{và} \ b \ne 0$$

\nhan{Bẫy} Số thuần ảo yêu cầu \textbf{hai} điều kiện. Nếu chỉ đặt $a = 0$ mà quên $b \ne 0$, ta nhận luôn cả $z = 0$, mà số $0$ là số thực chứ không phải số thuần ảo.

\nhan{Chú ý} Phần ảo của $z = a + bi$ là số thực $b$, \textbf{không phải} $bi$. Đề hỏi 虚部 thì đáp án là một số thực; phương án dạng $4i$ là phương án nhiễu.

\nhan{Cách nhanh} Khi đề cho $z = \dfrac{a + bi}{c + di}$, đừng khai triển ngay. Nhân tử và mẫu với $c - di$ rồi tính $c^2 + d^2$ trước để biết mẫu số, sau đó mới rút gọn tử.

\nhan{Nhận xét} Phép chia số phức và phép tính môđun là hai dạng chiếm phần lớn số câu về số phức trong đề thi thật. Cả hai đều chỉ dùng một công thức duy nhất, không đòi hỏi biến đổi sáng tạo.

\end{dinhly}
\section{CÁC DẠNG BÀI}

\subsection{Dạng 1 — Tích vô hướng và góc giữa hai vector}

\nhan{Cách nhận dạng} Đề cho tọa độ hai vector và hỏi 数量积 (tích vô hướng), 夹角 (góc), hoặc 余弦值 (giá trị cosin).

\nhan{Cách làm} Tính $a \cdot b = x_1x_2 + y_1y_2$, tính hai môđun, thay vào $\cos\theta = \dfrac{a \cdot b}{\lvert a \rvert \lvert b \rvert}$. Nếu đề cho độ dài và góc thì đi ngược lại: dùng $a \cdot b = \lvert a \rvert \lvert b \rvert \cos\theta$.

\begin{vidu}[{1}]
Cho $u = (1, \sqrt{3})$ và $v = (\sqrt{3}, 1)$. Tìm góc giữa hai vector.

Tích vô hướng: $u \cdot v = 1 \cdot \sqrt{3} + \sqrt{3} \cdot 1 = 2\sqrt{3}$.

Hai môđun: $\lvert u \rvert = \sqrt{1 + 3} = 2$ và $\lvert v \rvert = \sqrt{3 + 1} = 2$.

$$\cos\theta = \frac{2\sqrt{3}}{2 \cdot 2} = \frac{\sqrt{3}}{2}$$

Vậy $\theta = 30^\circ$.

\end{vidu}
\subsection{Dạng 2 — Tìm tham số để hai vector vuông góc}

\nhan{Cách nhận dạng} Đề có tham số $k$, $m$, $t$ và cụm 垂直 (vuông góc).

\nhan{Cách làm} Lập phương trình $x_1x_2 + y_1y_2 = 0$, giải phương trình bậc nhất hoặc bậc hai theo tham số. Chỉ cần một bước, không vẽ hình.

\begin{vidu}[{2}]
Cho $a = (5, 1)$ và $b = (-2, 2k)$. Tìm $k$ để $a \perp b$.

Điều kiện vuông góc:

$$5 \cdot (-2) + 1 \cdot 2k = 0 \iff -10 + 2k = 0 \iff k = 5$$

\end{vidu}
\subsection{Dạng 3 — Tìm tham số để hai vector song song}

\nhan{Cách nhận dạng} Đề có cụm 平行 (song song) hoặc 共线 (cùng phương).

\nhan{Cách làm} Lập phương trình $x_1y_2 - x_2y_1 = 0$. Với vector hai chiều, phương trình này là bậc nhất theo tham số và cho một nghiệm duy nhất. Với vector ba chiều thì xét tỉ số ba thành phần.

\begin{vidu}[{3}]
Cho $a = (k, 1)$ và $b = (4, k)$. Tìm $k$ để $a \parallel b$.

Điều kiện song song:

$$k \cdot k - 4 \cdot 1 = 0 \iff k^2 = 4 \iff k = \pm 2$$

\nhan{Bẫy} Đáp án chỉ ghi $k = 2$ là thiếu nghiệm âm. Phương trình bậc hai theo tham số luôn cho hai nghiệm, phải viết đủ.

\end{vidu}
\subsection{Dạng 4 — Môđun, hình chiếu và tọa độ trung điểm}

\nhan{Cách nhận dạng} Đề hỏi 模 (môđun), 投影 (hình chiếu), 中点 (trung điểm), hoặc cho tọa độ điểm đầu và vector rồi hỏi tọa độ điểm cuối.

\nhan{Cách làm} Môđun dùng căn của tổng bình phương. Hình chiếu của $a$ lên $b$ bằng $\dfrac{a \cdot b}{\lvert b \rvert}$. Tọa độ điểm cuối bằng tọa độ điểm đầu cộng vector.

\begin{vidu}[{4}]
Cho $\overrightarrow{AB} = (3, -6)$ và $A(-1, 2)$. Tìm tọa độ $B$.

Vì $\overrightarrow{AB} = B - A$ nên $B = A + \overrightarrow{AB}$:

$$B = (-1 + 3,\ 2 + (-6)) = (2, -4)$$

\nhan{Chú ý} Công thức này viết dưới dạng tọa độ: điểm cuối bằng điểm đầu cộng vector. Nếu đề cho điểm cuối và hỏi điểm đầu thì làm phép trừ.

\end{vidu}
\subsection{Dạng 5 — Phép toán cộng, trừ, nhân, chia số phức}

\nhan{Cách nhận dạng} Đề cho hai hoặc ba số phức và yêu cầu tính $z_1 \pm z_2$, $z_1 z_2$, hoặc $\dfrac{z_1}{z_2}$.

\nhan{Cách làm} Cộng trừ theo từng phần. Nhân như nhân đa thức rồi thay $i^2 = -1$. Chia bằng cách nhân tử và mẫu với số phức liên hợp của mẫu.

\begin{vidu}[{5}]
Cho $z_1 = 4 - i$ và $z_2 = (2 + 5i)i$. Tính $z_1 - z_2$.

Trước hết rút gọn $z_2$:

$$z_2 = (2 + 5i)i = 2i + 5i^2 = 2i - 5 = -5 + 2i$$

$$z_1 - z_2 = (4 - i) - (-5 + 2i) = 9 - 3i$$

\nhan{Cách nhanh} Luôn rút gọn số phức có chứa $i$ nhân bên ngoài trước khi thực hiện phép toán chính. Bỏ bước này là phải nhân hai lần.

\end{vidu}
\subsection{Dạng 6 — Phần thực, phần ảo, số phức liên hợp}

\nhan{Cách nhận dạng} Đề hỏi 实部, 虚部, 共轭复数, hoặc cho một biểu thức và hỏi phần ảo của kết quả.

\nhan{Cách làm} Đưa số phức về dạng chuẩn $a + bi$ bằng cách thực hiện hết các phép toán, sau đó đọc trực tiếp $a$ và $b$.

\begin{vidu}[{6}]
Cho $z = \dfrac{2+i}{1-i}$. Tìm số phức liên hợp $\overline{z}$.

Nhân tử và mẫu với $1 + i$:

$$z = \frac{(2+i)(1+i)}{(1-i)(1+i)} = \frac{2 + 2i + i + i^2}{1 - i^2} = \frac{1 + 3i}{2} = \frac{1}{2} + \frac{3}{2}i$$

$$\overline{z} = \frac{1}{2} - \frac{3}{2}i$$

\end{vidu}
\subsection{Dạng 7 — Tham số để số phức là số thực hoặc số thuần ảo}

\nhan{Cách nhận dạng} Đề cho $z$ chứa tham số $a$, $m$, $k$ và cụm 纯虚数 (thuần ảo) hoặc 实数 (số thực).

\nhan{Cách làm} Đưa $z$ về dạng $a + bi$ với $a$ và $b$ là biểu thức chứa tham số, rồi áp dụng Tính chất 10. Số thuần ảo cần hai điều kiện, số thực chỉ cần một.

\begin{vidu}[{7}]
Cho $z = (a + i)(1 - i)$. Tìm $a$ để $z$ là số thuần ảo.

Khai triển:

$$z = (a + i)(1 - i) = a - ai + i - i^2 = (a + 1) + (1 - a)i$$

Phần thực $a + 1$, phần ảo $1 - a$. Điều kiện số thuần ảo:

$$a + 1 = 0 \implies a = -1$$

Kiểm tra điều kiện thứ hai: $1 - (-1) = 2 \ne 0$ nên $a = -1$ thỏa mãn.

\end{vidu}
\subsection{Dạng 8 — Môđun số phức và lũy thừa của đơn vị ảo}

\nhan{Cách nhận dạng} Đề hỏi 模, $\lvert z \rvert$, $\lvert \overline{z} \rvert$, $z \cdot \overline{z}$, hoặc yêu cầu tính $i^n$ với $n$ lớn.

\nhan{Cách làm} Với môđun, đưa $z$ về dạng $a + bi$ rồi tính $\sqrt{a^2 + b^2}$. Với $i^n$, chia $n$ cho $4$ và dùng số dư.

\begin{vidu}[{8}]
Tính môđun của $z = -3 - 4i$.

$$\lvert z \rvert = \sqrt{(-3)^2 + (-4)^2} = \sqrt{9 + 16} = 5$$

\nhan{Cách nhanh} Bộ số $3$–$4$–$5$ và $5$–$12$–$13$ xuất hiện rất nhiều trong đề. Nhìn thấy phần thực và phần ảo là bội của $3$ và $4$ thì đọc ngay đáp án mà không cần bấm căn.

\nhan{Chú ý} $\lvert \overline{z} \rvert = \lvert z \rvert$, nên đề hỏi môđun của số phức liên hợp thì tính môđun của $z$ gốc, không cần đổi dấu.

\end{vidu}
