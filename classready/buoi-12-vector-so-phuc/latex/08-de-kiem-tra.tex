\begin{baitap}[1]
已知球的方程为 $(x-1)^2 + (y+3)^2 + (z-5)^2 = 9$ ，则球心坐标和半径分别为（ ）
\par\vspace{3pt}
\noindent A. $(-1,3,-5)$，$3$ \quad B. $(-1,3,-5)$，$9$ \quad C. $(1,-3,5)$，$9$ \quad D. $(1,-3,5)$，$3$
\end{baitap}

\begin{baitap}[2]
空间中点 $A(2,3,-1)$ 到平面 $x + 2y - z - 1 = 0$ 的距离是（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{4\sqrt{6}}{3}$ \quad B. $6\sqrt{6}$ \quad C. $4\sqrt{6}$ \quad D. $8\sqrt{6}$
\end{baitap}

\begin{baitap}[3]
点 $(-2, 5, 0)$ 在空间直角坐标系中的（ ）
\par\vspace{3pt}
\noindent A. $y$ 轴上 \quad B. $xOy$ 平面上 \quad C. $xOz$ 平面上 \quad D. 第一象限内
\end{baitap}

\begin{baitap}[4]
点 $A(2,0,22)$ 关于 $y$ 轴的对称点的坐标为（ ）
\par\vspace{3pt}
\noindent A. $(-2,0,22)$ \quad B. $(2,0,-22)$ \quad C. $(-2,0,-22)$ \quad D. $(2,0,22)$
\end{baitap}

\begin{baitap}[5]
若直线 $l$ 的倾斜角为 $45^\circ$，且过点 $(0,-7)$，则直线 $l$ 的方程是（ ）
\par\vspace{3pt}
\noindent A. $y - 1 = x$ \quad B. $y + 7 = x$ \quad C. $y - 1 = -x$ \quad D. $y + 7 = -x$
\end{baitap}

\begin{baitap}[6]
在空间直角坐标系中，$A(1,2,0)$，$B(5,0,1)$，$C(1,2,3)$，若 $M$ 为 $BC$ 的中点，则 $\lvert AM \rvert$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\sqrt{3}$ \quad B. $2\sqrt{3}$ \quad C. $3\sqrt{2}$ \quad D. 3
\end{baitap}

\begin{baitap}[7]
已知 $A(2, 3 - \mu, -1 + \nu)$ 关于 $x$ 轴的对称点是 $A'(\lambda, 7, -6)$，则 $\lambda, \mu, \nu$ 的值为（ ）
\par\vspace{3pt}
\noindent A. $\lambda = 2$，$\mu = 10$，$\nu = 7$ \quad B. $\lambda = -2$，$\mu = 10$，$\nu = 8$ \quad C. $\lambda = 2$，$\mu = -4$，$\nu = -5$ \quad D. $\lambda = -2$，$\mu = -4$，$\nu = -5$
\end{baitap}

\begin{baitap}[8]
直线 $2x + 3y - 6 = 0$ 在 $y$ 轴上的截距是（ ）
\par\vspace{3pt}
\noindent A. $-3$ \quad B. $2$ \quad C. $3$ \quad D. $-2$
\end{baitap}

\begin{baitap}[9]
圆 $x^2 + y^2 - 6x + 8y = 0$ 的半径是（ ）

\noindent A. $\sqrt{5}$ \quad B. 25 \quad C. 5 \quad D. $\sqrt{7}$
\end{baitap}

\begin{baitap}[10]
直线 $L_1: 3x - 4y + 1 = 0$ 与直线 $L_2: 6x - 8y - 5 = 0$ 的位置关系是（ ）
\par\vspace{3pt}
\noindent A. 垂直 \quad B. 平行 \quad C. 重合 \quad D. 相交但不垂直
\end{baitap}

\begin{baitap}[11]
在平面直角坐标系中，点 $P(-1,2)$ 到直线 $4x - 3y + 5 = 0$ 的距离是（ ）
\par\vspace{3pt}
\noindent A. 1 \quad B. $\dfrac{2}{5}$ \quad C. $\dfrac{13}{5}$ \quad D. $\dfrac{7}{5}$
\end{baitap}

\begin{baitap}[12]
过点 $(0,3)$ 且与直线 $y = 2x + 1$ 垂直的直线方程是（ ）
\par\vspace{3pt}
\noindent A. $2x - y + 3 = 0$ \quad B. $x + 2y - 6 = 0$ \quad C. $x - 2y + 6 = 0$ \quad D. $2x + y - 3 = 0$
\end{baitap}

\begin{baitap}[13]
设 $F_1, F_2$ 为双曲线 $C: \dfrac{x^2}{3} - y^2 = 1$ 的左、右焦点，$Q$ 为双曲线右支上一点，点 $P(0,2)$。当 $\lvert QF_1 \rvert + \lvert PQ \rvert$ 取最小值时，$\lvert QF_2 \rvert$ 的值为（ ）
\par\vspace{3pt}
\noindent A. $\sqrt{3} - \sqrt{2}$ \quad B. $\sqrt{3} + \sqrt{2}$ \quad C. $\sqrt{6} - 2$ \quad D. $\sqrt{6} + 2$
\end{baitap}

\begin{baitap}[14]

\noindent A. 3 \quad B. 6 \quad C. 7 \quad D. 9

\begin{baitap}[15]

\begin{baitap}[16]

\begin{baitap}[17]

\begin{baitap}[18]
双曲线 $\dfrac{x^2}{16} - \dfrac{y^2}{9} = 1$ 的离心率是（ ）

\noindent A. $\dfrac{3}{5}$ \quad B. $\dfrac{5}{3}$ \quad C. $\dfrac{4}{5}$ \quad D. $\dfrac{5}{4}$

\begin{baitap}[19]
已知椭圆 $\dfrac{x^2}{25} + \dfrac{y^2}{16} = 1$ 上一点 $P$ 到其中一个焦点的距离为 $3$，则 $P$ 到另一个焦点的距离是（ ）

\begin{baitap}[20]

\begin{baitap}[21]

\begin{baitap}[22]

\end{baitap}

\begin{baitap}[24]
已知平面向量 $\vec{a}, \vec{b}$ 的夹角为 $60^\circ$，且 $\lvert\vec{a}\rvert = 2$，$\lvert\vec{a} + \vec{b}\rvert = 2\sqrt{3}$，则 $\lvert\vec{b}\rvert$ 等于（ ） \emph{Hết giờ. Đáp án và lời giải: giáo viên phát sau buổi kiểm tra.}

\end{baitap}
