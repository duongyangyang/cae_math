\section{PHẦN C — BÀI TẬP CHUẨN BỊ}

Làm 10 câu dưới đây \textbf{trước buổi học}. Sáu câu đầu chỉ cần đọc hiểu thuật ngữ là làm được; các câu cuối đòi hỏi biến đổi, nếu chưa làm được thì ghi lại chỗ vướng mắc để hỏi trên lớp.

\subsection{Mức dễ}

\begin{baitap}[1]
设向量 $\vec{a}$, $\vec{b}$ 之间的夹角为 $\alpha$ ，则 $\vec{a} \cdot \vec{b} = \lvert\vec{a}\rvert\lvert\vec{b}\rvert\cos\alpha$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\lvert\vec{a}\rvert\lvert\vec{b}\rvert\sin\alpha$ \quad B. $\lvert\vec{a}\rvert\lvert\vec{b}\rvert\cos\alpha$ \quad C. $\lvert\vec{a}\rvert + \lvert\vec{b}\rvert$ \quad D. $\lvert\vec{a}\rvert\lvert\vec{b}\rvert$
\end{baitap}
\begin{baitap}[2]
若复数 $z = a + bi\ (a, b \in \mathbb{R})$ 是纯虚数，则（ ）
\par\vspace{3pt}
\noindent A. $a = 0$ 且 $b \ne 0$ \quad B. $a \ne 0$ 且 $b = 0$ \quad C. $a = 0$ 且 $b = 0$ \quad D. $a \ne 0$ 且 $b \ne 0$
\end{baitap}
\begin{baitap}[3]
已知向量 $a = (1, -5)$, $b = (2,4)$ 。求 $(a + b) \cdot (a - b)$ 等于（ ）
\par\vspace{3pt}
\noindent A. 12 \quad B. -5 \quad C. -6 \quad D. 6
\end{baitap}
\begin{baitap}[4]
已知两个平面向量 $u = (3,4)$, $v = (1,0)$ 。求向量 $u$ 在 $v$ 方向上的投影是（ ）
\par\vspace{3pt}
\noindent A. 1 \quad B. 3 \quad C. 4 \quad D. 5
\end{baitap}
\begin{baitap}[5]
已知向量 $u = (x,15)$ ， $v = (5,-3)$ ，若 $u \parallel v$ ，则 $x$ 的值是（ ）
\par\vspace{3pt}
\noindent A. -25 \quad B. 25 \quad C. -5 \quad D. 5
\end{baitap}
\begin{baitap}[6]
复数 $z = i^{2025}$ 等于（ ）
\par\vspace{3pt}
\noindent A. 1 \quad B. -1 \quad C. $i$ \quad D. $-i$
\end{baitap}

\begin{baitap}[8]
下列各组向量中，是共线向量的是（ ）
\par\vspace{3pt}

\end{baitap}
