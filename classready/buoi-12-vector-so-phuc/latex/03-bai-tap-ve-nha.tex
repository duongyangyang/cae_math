\section{Bài tập bắt buộc}

Làm toàn bộ 25 câu. Đây là phần tính vào điểm bài tập về nhà.

\subsection*{NHÓM A — VECTOR}

\begin{baitap}[1]
若向量 $a = (5,1)$ ， $b = (-2,2k)$ 且 $a \perp b$ ，则 $k$ 的值为（ ）
\par\vspace{3pt}
\noindent A. -10 \quad B. 10 \quad C. 5 \quad D. -5
\end{baitap}

\begin{baitap}[2]
设向量 $a = (1,3,5)$, $b = (-1,3,2)$, 则 $3a + 2b$ 等于（ ）
\par\vspace{3pt}
\noindent A. $(2,12,17)$ \quad B. $(0,6,7)$ \quad C. $(-1,15,16)$ \quad D. $(1,15,19)$
\end{baitap}

\begin{baitap}[3]
两个向量 $a = (1,2,3)$ 和 $b = (-2,-4,-6)$ 的位置关系是（ ）
\par\vspace{3pt}
\noindent A. 平行 \quad B. 垂直 \quad C. 相交但不垂直 \quad D. 既不平行也不垂直
\end{baitap}

\begin{baitap}[4]
在空间直角坐标系中，向量 $\overrightarrow{AB} = (2,-1,3)$，若点 $A = (1,2,0)$，则点 $B$ 的坐标是（ ）
\par\vspace{3pt}
\noindent A. $(2,-1,3)$ \quad B. $(-1,3,-3)$ \quad C. $(1,-1,3)$ \quad D. $(3,1,3)$
\end{baitap}

\begin{baitap}[5]
若直线 $L_1$ 的方向向量是 $d_1 = (1,-1,2)$ ，直线 $L_2$ 的方向向量是 $d_2 = (2,2,1)$ 。求 $L_1$ 和 $L_2$ 的夹角的余弦值是（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{1}{3}$ \quad B. $\dfrac{\sqrt{6}}{3}$ \quad C. $\dfrac{1}{9}$ \quad D. $\dfrac{\sqrt{6}}{9}$
\end{baitap}

\begin{baitap}[6]
已知 $A(10,0,0)$ 和 $B(0,15,0)$ ，则 $AB$ 所在直线的方向向量可以是（ ）
\par\vspace{3pt}
\noindent A. $(10,15,0)$ \quad B. $(10,-15,0)$ \quad C. $(-10,-15,0)$ \quad D. $(0,0,0)$
\end{baitap}

\begin{baitap}[7]
已知 $M(5, -1, 2)$，$A(4, 2, -1)$，$O$ 为坐标原点，若 $\overrightarrow{OM} = \overrightarrow{AB}$，则点 $B$ 的坐标应为（ ）
\par\vspace{3pt}
\noindent A. $(-1, 3, -3)$ \quad B. $(9, 1, 1)$ \quad C. $(1, -3, 3)$ \quad D. $(-9, -1, -1)$
\end{baitap}

\begin{baitap}[8]
已知空间四边形 $ABCD$，向量 $\overrightarrow{AB} = a$，$\overrightarrow{AD} = b$，$\overrightarrow{AC} = c$ 。求 $\overrightarrow{DB}$ 等于（ ）
\par\vspace{3pt}
\noindent A. $a - b$ \quad B. $b - a$ \quad C. $a + b$ \quad D. $-a - b$
\end{baitap}

\subsection*{NHÓM B — SỐ PHỨC}

\begin{baitap}[9]
若复数 $(a + i)(1 - i)$ 是纯虚数，则实数 $a$ 的值为（ ）
\par\vspace{3pt}
\noindent A. -1 \quad B. 0 \quad C. 1 \quad D. 2
\end{baitap}

\begin{baitap}[10]
复数 $\dfrac{-3-i}{1+i} + i$ 等于（ ）
\par\vspace{3pt}
\noindent A. $2 + i$ \quad B. $-2 + i$ \quad C. $-2 + 2i$ \quad D. $-4 + 2i$
\end{baitap}

\begin{baitap}[11]
复数 $z = \dfrac{10+5i}{2-i}$ 的虚部是（ ）
\par\vspace{3pt}
\noindent A. 3 \quad B. 4 \quad C. $4i$ \quad D. $3i$
\end{baitap}

\begin{baitap}[12]
已知复数 $z_1 = 6 + 8i$ ， $z_2 = -1 + 4i$ ，则 $\lvert z_1 + z_2 \rvert$ 等于（ ）
\par\vspace{3pt}
\noindent A. 12 \quad B. 13 \quad C. 14 \quad D. 5
\end{baitap}

\begin{baitap}[13]
若 $z = 1 + i$，则 $z^8$ 等于（ ）
\par\vspace{3pt}
\noindent A. 16 \quad B. -16 \quad C. $16i$ \quad D. $2^{56}$
\end{baitap}

\subsection*{NHÓM C — CONIC}

\begin{baitap}[14]
已知椭圆的两个焦点坐标分别为 $F_1(-3,0)$ 和 $F_2(3,0)$ ，且过点 $P(5,0)$ ，则该椭圆的标准方程是（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{x^2}{16} + \dfrac{y^2}{25} = 1$ \quad B. $\dfrac{x^2}{25} + \dfrac{y^2}{16} = 1$ \quad C. $\dfrac{x^2}{9} + \dfrac{y^2}{25} = 1$ \quad D. $\dfrac{x^2}{25} + \dfrac{y^2}{9} = 1$
\end{baitap}

\begin{baitap}[15]
抛物线 $y^2 = -12x$ 的焦点坐标是（ ）
\par\vspace{3pt}
\noindent A. $(3,0)$ \quad B. $(0,3)$ \quad C. $(-3,0)$ \quad D. $(0, -3)$
\end{baitap}

\begin{baitap}[16]
双曲线 $\dfrac{x^2}{16} - \dfrac{y^2}{9} = 1$ 的渐近线方程是（ ）
\par\vspace{3pt}
\noindent A. $y = \pm\dfrac{3}{4}x$ \quad B. $y = \pm\dfrac{4}{3}x$ \quad C. $y = \pm\dfrac{16}{9}x$ \quad D. $y = \pm\dfrac{9}{16}x$
\end{baitap}

\begin{baitap}[17]
抛物线 $x^2 = -4y$ 的准线方程是（ ）
\par\vspace{3pt}
\noindent A. $x = 1$ \quad B. $x = -1$ \quad C. $y = -1$ \quad D. $y = 1$
\end{baitap}

\begin{baitap}[18]
若抛物线 $C: x^2 = 2py\ (p>0)$ 上的点 $P$ 到焦点的距离为 8，到 $x$ 轴的距离为 6，则抛物线 $C$ 的标准方程是（ ）
\par\vspace{3pt}
\noindent A. $x^2 = 4y$ \quad B. $x^2 = 6y$ \quad C. $x^2 = 8y$ \quad D. $x^2 = 16y$
\end{baitap}

\begin{baitap}[19]
已知平面四边形 $ABCD$ 的四个顶点都在抛物线 $y^2 = ax\ (a \neq 0)$ 上，其中顶点 $D(1,-2)$，$F$ 为抛物线的焦点。若 $\overrightarrow{AF} = \dfrac{1}{3}(\overrightarrow{AB} + \overrightarrow{AC})$，则 $\lvert\overrightarrow{AF}\rvert + \lvert\overrightarrow{BF}\rvert + \lvert\overrightarrow{CF}\rvert$ 等于（ ）
\par\vspace{3pt}
\noindent A. 12 \quad B. 9 \quad C. 6 \quad D. 3
\end{baitap}

\subsection*{NHÓM D — ĐƯỜNG THẲNG VÀ ĐƯỜNG TRÒN}

\begin{baitap}[20]
直线 $2x - 3y + 6 = 0$ 在 $x$ 轴上的截距是（ ）
\par\vspace{3pt}
\noindent A. -3 \quad B. 3 \quad C. 2 \quad D. -2
\end{baitap}

\begin{baitap}[21]
圆 $x^2 + y^2 - 4x + 2y - 4 = 0$ 的半径是（ ）
\par\vspace{3pt}
\noindent A. 2 \quad B. $\sqrt{5}$ \quad C. 3 \quad D. 4
\end{baitap}

\begin{baitap}[22]
两直线 $kx - 4y + 1 = 0$ 和 $(k + 4)x - y - 3 = 0$ 互相垂直，则 $k$ 的值是（ ）
\par\vspace{3pt}
\noindent A. 0 \quad B. -2 \quad C. $\dfrac{5}{2}$ \quad D. 2
\end{baitap}

\begin{baitap}[23]
过点 $(1,2)$ 且与直线 $y = -3x + 1$ 平行的直线方程是（ ）
\par\vspace{3pt}
\noindent A. $3x + y - 7 = 0$ \quad B. $x - 3y + 5 = 0$ \quad C. $3x + y - 5 = 0$ \quad D. $3x - y + 1 = 0$
\end{baitap}

\subsection*{NHÓM E — ÔN TẬP TÍCH LŨY}

\begin{baitap}[24]
已知集合 $A=\{a,b,c\}$，则下列不正确的关系是（ ）
\par\vspace{3pt}
\noindent A. $a \in A$ \quad B. $\{a\} \subseteq A$ \quad C. $\varnothing \subseteq A$ \quad D. $c \notin A$
\end{baitap}

\begin{baitap}[25]
若 $a > b > 0$ ，则下列不等式中正确的是（ ）
\par\vspace{3pt}
\noindent A. $a^2 < b^2$ \quad B. $\dfrac{1}{a} > \dfrac{1}{b}$ \quad C. $a - b < 0$ \quad D. $\dfrac{a}{b} > 1$
\end{baitap}

\section{Bài tập tự chọn}

Phần này luyện cùng nội dung nhưng ở mức khó hơn rõ rệt. Chỉ làm sau khi đã hoàn thành 25 câu bắt buộc. Không tính vào điểm bài tập về nhà.

\begin{baitap}[1]
复数 $z$ 满足 $z(1 + i) = 2i$，则 $z$ 等于（ ）
\par\vspace{3pt}
\noindent A. $1 - i$ \quad B. $1 + i$ \quad C. $-1 + i$ \quad D. $-1 - i$
\end{baitap}

\begin{baitap}[2]
在 $\triangle ABC$ 中，$D$ 为 $BC$ 的中点，$\overrightarrow{EB} = 2\overrightarrow{AE}$，$\overrightarrow{AF} = 2\overrightarrow{FC}$，$EF$ 与 $AD$ 交于 $G$，$\overrightarrow{AG} = \lambda \overrightarrow{AD}$，则 $\lambda$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{1}{2}$ \quad B. $\dfrac{4}{9}$ \quad C. $\dfrac{3}{5}$ \quad D. $\dfrac{2}{3}$
\end{baitap}

\begin{baitap}[3]
设向量 $a = (k,1)$，$b = (4,k)$ 且 $a \parallel b$，则 $k$ 的值是（ ）
\par\vspace{3pt}
\noindent A. 2 \quad B. -2 \quad C. $\pm 2$ \quad D. 4
\end{baitap}

\begin{baitap}[4]
复数 $z$ 满足 $z \cdot \overline{z} = 9$，则 $\lvert z \rvert$ 等于（ ）
\par\vspace{3pt}
\noindent A. 1 \quad B. 2 \quad C. 3 \quad D. $\sqrt{3}$
\end{baitap}

\begin{baitap}[5]
复数 $z = -3 - 4i$ 的模 $\lvert \overline{z} \rvert$ 等于（ ）
\par\vspace{3pt}
\noindent A. 3 \quad B. 4 \quad C. 5 \quad D. $\sqrt{7}$
\end{baitap}

\begin{baitap}[6]
平面 $x - 2y + 3z - 6 = 0$ 的一个法向量是（ ）
\par\vspace{3pt}
\noindent A. $(1,2,3)$ \quad B. $(1,-2,3)$ \quad C. $(-1,2,-3)$ \quad D. $(2,4,-6)$
\end{baitap}

\begin{baitap}[7]
设平面向量 $\overrightarrow{AB} = (3,-6)$，点 $A(-1,2)$，则点 $B$ 的坐标为（ ）
\par\vspace{3pt}
\noindent A. $(-2,4)$ \quad B. $(2,-4)$ \quad C. $(-4,8)$ \quad D. $(4,-8)$
\end{baitap}

\begin{baitap}[8]
复数 $(3 - i) \cdot i - (1 + 2i)$ 等于（ ）
\par\vspace{3pt}
\noindent A. $-2 - 3i$ \quad B. $-2 + i$ \quad C. $i$ \quad D. $2 - i$
\end{baitap}

\begin{baitap}[9]
已知 $\lvert\overrightarrow{a}\rvert = 2\sqrt{2}$，$\lvert\overrightarrow{b}\rvert = 3$，$\overrightarrow{a}$ 与 $\overrightarrow{b}$ 的夹角为 $\dfrac{\pi}{4}$。若 $\overrightarrow{AB} = 5\overrightarrow{a} + 2\overrightarrow{b}$，$\overrightarrow{AC} = \overrightarrow{a} - 3\overrightarrow{b}$，且 $D$ 为 $BC$ 的中点，则 $\lvert\overrightarrow{AD}\rvert$ 的长度为（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{15}{2}$ \quad B. $\dfrac{\sqrt{15}}{2}$ \quad C. 7 \quad D. 8
\end{baitap}

\begin{baitap}[10]
复数 $z = \cos\dfrac{\pi}{15} + i\sin\dfrac{\pi}{15}$ 是方程 $x^5 + \alpha = 0$ 的一个根，那么 $\alpha$ 的值为（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{\sqrt{3}}{2} + \dfrac{1}{2}i$ \quad B. $\dfrac{1}{2} + \dfrac{\sqrt{3}}{2}i$ \quad C. $-\dfrac{\sqrt{3}}{2} - \dfrac{1}{2}i$ \quad D. $-\dfrac{1}{2} - \dfrac{\sqrt{3}}{2}i$
\end{baitap}

\begin{baitap}[11]
已知向量 $a = (3,-2)$，$b = (-1,4)$，则 $a \cdot (a + b)$ 等于（ ）
\par\vspace{3pt}
\noindent A. 2 \quad B. -2 \quad C. 11 \quad D. -11
\end{baitap}

\begin{baitap}[12]
已知向量 $a = (2,-1)$，$b = (1,3)$，若 $(a + tb) \perp a$，则实数 $t$ 的值为（ ）
\par\vspace{3pt}
\noindent A. 5 \quad B. -5 \quad C. 1 \quad D. -1
\end{baitap}

\begin{baitap}[13]
已知复数 $z = (1+i)(3-i)$，则 $z$ 的共轭复数是（ ）
\par\vspace{3pt}
\noindent A. $4 + 2i$ \quad B. $-4 + 2i$ \quad C. $-4 - 2i$ \quad D. $4 - 2i$
\end{baitap}

\begin{baitap}[14]
已知复数 $z = \dfrac{3 + 4i}{1 - 2i}$，则 $\lvert z \rvert$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\sqrt{5}$ \quad B. 5 \quad C. $\sqrt{3}$ \quad D. 3
\end{baitap}

\begin{baitap}[15]
已知向量 $a = (1,2)$，$b = (2,-3)$，若 $a \perp (a + \lambda b)$，则实数 $\lambda$ 的值为（ ）
\par\vspace{3pt}
\noindent A. $-\dfrac{5}{4}$ \quad B. $\dfrac{4}{5}$ \quad C. $-\dfrac{4}{5}$ \quad D. $\dfrac{5}{4}$
\end{baitap}
