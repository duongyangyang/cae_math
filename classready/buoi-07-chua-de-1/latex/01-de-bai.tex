\section{PHẦN I — TẬP HỢP VÀ BẤT ĐẲNG THỨC (5 câu)}

\begin{baitap}[1]
如果集合 $A=\{1,a\}$，$B=\{1,3,6,7,9\}$，且 $A\subseteq B$，则实数 $a\ne1$ 可以取值的个数是（ ）
\par\vspace{3pt}
\noindent A. 0 \quad B. 3 \quad C. 4 \quad D. 5
\end{baitap}

\begin{baitap}[2]
设集合 $A=\{x\in\mathbb{R}\mid x^2-4<0\}$，集合 $B=\{x\in\mathbb{Z}\mid -3<x\le1\}$。求 $A\cap B$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\{-2,-1,0,1\}$ \quad B. $\{-1,0,1\}$ \quad C. $(-2,1]$ \quad D. $\{-2,-1,0\}$
\end{baitap}

\begin{baitap}[3]
不等式 $|x-2|+|x+1|<5$ 的解集是（ ）
\par\vspace{3pt}
\noindent A. $[-2,3]$ \quad B. $(-\infty,-2)\cup(3,+\infty)$ \quad C. $(-2,3)$ \quad D. $(-\infty,-2]\cup[3,+\infty)$
\end{baitap}

\begin{baitap}[4]
设 $A=\{x\mid x^2-(a+1)x+a\le0\}$，$B=\{x\mid x^2-4x+3<0\}$。若 $A\subseteq B$，则实数 $a$ 的取值范围是（ ）
\par\vspace{3pt}
\noindent A. 无实数 $a$ 满足条件 \quad B. $[1,3]$ \quad C. $(1,3)$ \quad D. $(0,3)$
\end{baitap}

\begin{baitap}[5]
已知 $x>0$，$y>0$，且 $x+2y=4$。求 $xy$ 的最大值等于（ ）
\par\vspace{3pt}
\noindent A. 2 \quad B. 4 \quad C. 1 \quad D. 8
\end{baitap}

\section{PHẦN III — LƯỢNG GIÁC (10 câu)}

\begin{baitap}[13]

\begin{baitap}[15]

\begin{baitap}[16]

\end{baitap}

\begin{baitap}[17]
已知 $\sin\dfrac{\pi}{4}=\dfrac{\sqrt{2}}{2}$，$\sin\dfrac{\pi}{6}=\dfrac{1}{2}$。利用公式 $\cos(\alpha-\beta)=\cos\alpha\cos\beta+\sin\alpha\sin\beta$ 可得 $\cos\dfrac{\pi}{12}$ 等于（ ）

\noindent A. $\dfrac{\sqrt{6}}{4}$ \quad B. $\dfrac{\sqrt{6}+\sqrt{2}}{4}$ \quad C. $\dfrac{\sqrt{6}-\sqrt{2}}{4}$ \quad D. $\dfrac{1+\sqrt{3}}{4}$
\end{baitap}

\begin{baitap}[18]
若 $\sin\alpha=\dfrac{3}{5}$，且 $\alpha$ 是第一象限角，则 $\sin 2\alpha$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{7}{25}$ \quad B. $\dfrac{24}{25}$ \quad C. $\dfrac{12}{25}$ \quad D. $\dfrac{18}{25}$
\end{baitap}

\begin{baitap}[19]
已知 $\sin\alpha=-\dfrac{12}{13}$，且 $\alpha\in\left(\pi,\dfrac{3\pi}{2}\right)$，则 $\cos\dfrac{\alpha}{2}$ 等于（ ）

\noindent A. $\dfrac{3\sqrt{13}}{13}$ \quad B. $-\dfrac{2\sqrt{13}}{13}$ \quad C. $-\dfrac{3\sqrt{13}}{13}$ \quad D. $\dfrac{2\sqrt{13}}{13}$
\end{baitap}

\begin{baitap}[20]
