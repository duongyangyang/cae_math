\section{PHẦN I — LƯỢNG GIÁC}

\begin{dapan}[1 — Đáp án B]

\textbf{Giải.} $\alpha$ ở góc phần tư thứ hai nên $\sin\alpha > 0$:

$$\sin\alpha = \sqrt{1 - \cos^2\alpha} = \sqrt{1 - \dfrac{9}{25}} = \dfrac{4}{5}$$

$$\sin 2\alpha = 2\sin\alpha\cos\alpha = 2 \cdot \dfrac{4}{5} \cdot \left(-\dfrac{3}{5}\right) = -\dfrac{24}{25}$$

\nhan{Bẫy} Chọn A là quên dấu âm của $\cos\alpha$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[2 — Đáp án C]

\textbf{Giải.} $\dfrac{4\pi}{3} = 240^\circ$, nằm ở góc phần tư thứ ba nên $\cos$ âm. Góc tham chiếu là $60^\circ$:

$$\cos\dfrac{4\pi}{3} = -\cos\dfrac{\pi}{3} = -\dfrac{1}{2}$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[3 — Đáp án A]

\textbf{Giải.} Công thức nhân đôi theo $\tan$:

$$\sin 2\alpha = \dfrac{2\tan\alpha}{1 + \tan^2\alpha} = \dfrac{2 \cdot 2}{1 + 4} = \dfrac{4}{5}$$

\nhan{Cách nhanh} Nhớ công thức $\sin 2\alpha = \dfrac{2t}{1+t^2}$ với $t = \tan\alpha$. Không cần dựng tam giác.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[4 — Đáp án D]

\textbf{Giải.} Chu kì của $y = A\sin(\omega x + \varphi)$ là $T = \dfrac{2\pi}{\lvert \omega \rvert}$.

$$T = \dfrac{2\pi}{\frac{1}{4}} = 8\pi$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[5 — Đáp án B]

\textbf{Giải.} Bình phương hai vế:

$$(\sin\alpha - \cos\alpha)^2 = \dfrac{1}{4} \implies \sin^2\alpha - 2\sin\alpha\cos\alpha + \cos^2\alpha = \dfrac{1}{4}$$

$$1 - \sin 2\alpha = \dfrac{1}{4} \implies \sin 2\alpha = \dfrac{3}{4}$$

\nhan{Chú ý} Bình phương hai vế chỉ dùng để tìm $\sin 2\alpha$, không dùng để kết luận về $\sin\alpha - \cos\alpha$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[6 — Đáp án A]

\textbf{Giải.} Dạng $a\sin x + b\cos x$ có giá trị nhỏ nhất $-\sqrt{a^2 + b^2}$:

$$-\sqrt{1^2 + (\sqrt{3})^2} = -\sqrt{4} = -2$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[7 — Đáp án C]

\textbf{Giải.} Trên vòng tròn lượng giác, $\sin x = -\dfrac{1}{2}$ có hai điểm, ứng với hai họ nghiệm $x = \dfrac{7\pi}{6}$ và $x = \dfrac{11\pi}{6}$. Trên đoạn $[0, 2\pi]$ cả hai đều nằm trong đoạn, nên có 2 nghiệm.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[8 — Đáp án D]

\textbf{Giải.} Trục đối xứng của $y = \sin(\omega x + \varphi)$ thỏa mãn $\omega x + \varphi = \dfrac{\pi}{2} + k\pi$. Với $\omega = 2$, $\varphi = -\dfrac{\pi}{3}$:

$$2x - \dfrac{\pi}{3} = \dfrac{\pi}{2} \implies x = \dfrac{5\pi}{12}$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[9 — Đáp án B]

\textbf{Giải.} Dùng công thức hạ bậc $\cos 2x = 1 - 2\sin^2 x$:

$$y = 1 - 2\sin^2 x = \cos 2x$$

Chu kì là $\dfrac{2\pi}{2} = \pi$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[10 — Đáp án C]

\textbf{Giải.} Công thức nhân đôi theo $\tan$, mẫu số có dạng $1 - t^2$:

$$\tan 2\alpha = \dfrac{2\tan\alpha}{1 - \tan^2\alpha} = \dfrac{2 \cdot \frac{4}{3}}{1 - \frac{16}{9}} = \dfrac{\frac{8}{3}}{-\frac{7}{9}} = -\dfrac{24}{7}$$

\nhan{Bẫy} Chọn A là quên mẫu số có thể âm. Vì $\tan\alpha = \dfrac{4}{3} > 1$ nên $\tan 2\alpha$ bắt buộc âm.

\section{PHẦN II — HÌNH HỌC GIẢI TÍCH}


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[11 — Đáp án D]

\textbf{Giải.} Trục $x$ có $y = 0$. Thay vào phương trình:

$$3x - 12 = 0 \implies x = 4$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[12 — Đáp án A]

\textbf{Giải.} Công thức khoảng cách từ điểm $P(x_0, y_0)$ đến đường thẳng $ax + by + c = 0$:

$$d = \dfrac{\lvert ax\_0 + by\_0 + c \rvert}{\sqrt{a^2 + b^2}} = \dfrac{\lvert 5 \cdot 2 - 12 \cdot (-1) + 3 \rvert}{\sqrt{25 + 144}} = \dfrac{\lvert 25 \rvert}{13} = \dfrac{25}{13}$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[13 — Đáp án C]

\textbf{Giải.} Phương trình $x^2 + y^2 + Dx + Ey + F = 0$ có tâm $\left(-\dfrac{D}{2}, -\dfrac{E}{2}\right)$ và bán kính $R = \sqrt{\dfrac{D^2 + E^2}{4} - F}$.

$$R = \sqrt{\dfrac{36 + 64}{4}} = \sqrt{25} = 5$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[14 — Đáp án B]

\textbf{Giải.} Hai đường thẳng song song khi cùng vectơ pháp tuyến. Đường đã cho có vectơ pháp tuyến $(3, -2)$.

Đường cần tìm: $3x - 2y + c = 0$. Thay điểm $(-1, 3)$: $-3 - 6 + c = 0 \implies c = 9$.

$$3x - 2y + 9 = 0$$

\nhan{Bẫy} Đáp án A là chính đường đã cho, không đi qua điểm $(-1, 3)$ nên không thỏa mãn. Đáp án C có hệ số góc đổi dấu, đó là đường vuông góc.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[15 — Đáp án A]

\textbf{Giải.} Với $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = 1$ và $a > b > 0$: $a^2 = 100$, $b^2 = 36$.

$$c = \sqrt{a^2 - b^2} = \sqrt{64} = 8, \qquad e = \dfrac{c}{a} = \dfrac{8}{10} = \dfrac{4}{5}$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[16 — Đáp án D]

\textbf{Giải.} Với $\dfrac{x^2}{a^2} - \dfrac{y^2}{b^2} = 1$: $a^2 = 9$, $b^2 = 16$.

$$c = \sqrt{a^2 + b^2} = \sqrt{25} = 5$$

Trục thực nằm trên $Ox$ nên tiêu điểm là $(\pm 5, 0)$.

\nhan{Bẫy} Chọn C là nhầm đỉnh với tiêu điểm; đáp án B là lấy $\sqrt{b^2 - a^2}$ thay vì $\sqrt{a^2 + b^2}$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[17 — Đáp án B]

\textbf{Giải.} Parabol $y^2 = 2px$ có tham số $p = 5$, đường chuẩn $x = -\dfrac{p}{2}$:

$$x = -\dfrac{5}{2}$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[18 — Đáp án C]

\textbf{Giải.} Vectơ pháp tuyến lần lượt là $(2, 3)$ và $(4, 6)$. Vì $\dfrac{4}{2} = \dfrac{6}{3} = 2$ nên hai vectơ cùng phương, hai đường thẳng song song hoặc trùng nhau.

Kiểm tra hệ số tự do: $\dfrac{4}{2} = 2$ nhưng $\dfrac{6}{3} = 2 \neq \dfrac{1}{-6}$. Hệ số không tỉ lệ nên hai đường thẳng song song.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[19 — Đáp án A]

\textbf{Giải.} Trung điểm $M$ có tọa độ là trung bình cộng các tọa độ tương ứng:

$$M\left(\dfrac{1+3}{2}, \dfrac{2+6}{2}\right) = (2, 4)$$

\section{PHẦN III — DÃY SỐ}

\begin{dapan}[20 — Đáp án D]

\textbf{Giải.} Công thức số hạng tổng quát của cấp số cộng:

$$a\_8 = a\_1 + 7d = 5 + 7(-3) = 5 - 21 = -16$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[21 — Đáp án A]

\textbf{Giải.} Công thức số hạng tổng quát của cấp số nhân:

$$a\_3 = a\_1 q^2 = 4 \cdot (-3)^2 = 4 \cdot 9 = 36$$

\nhan{Bẫy} Chọn B là nhầm $q^2$ thành $q^3$ và giữ dấu âm.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[22 — Đáp án B]

\textbf{Giải.} Tổng $n$ số hạng đầu của cấp số cộng:

$$S\_5 = 5a\_1 + \dfrac{5 \cdot 4}{2}d = 5 \cdot 2 + 10 \cdot 3 = 40$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[23 — Đáp án C]

\textbf{Giải.} $a_3 = a_1 q^2 \implies 3q^2 = 27 \implies q^2 = 9$.

Vì đề cho các số hạng đều dương nên $q = 3$.

\nhan{Bẫy} Đáp án A là quên lấy căn bậc hai của $q^2$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[24 — Đáp án D]

\textbf{Giải.} Dãy $a_n = 2n - 1$ là cấp số cộng với $a_1 = 1$, $d = 2$, và $a_{10} = 19$.

$$S\_{10} = \dfrac{10(a\_1 + a\_{10})}{2} = 5 \cdot 20 = 100$$

\nhan{Cách nhanh} Dãy số lẻ từ 1 đến 19 có 10 số hạng, tổng bằng $10^2 = 100$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[25 — Đáp án B]

\textbf{Giải.} $a_3 = a_1 q^2 = 2 \cdot 25 = 50$.

\nhan{Bẫy} Chọn A là nhân $a_1$ với $q$ thay vì với $q^2$.

\section{PHẦN IV — HÀM SỐ}


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[26 — Đáp án A]

\textbf{Giải.} Hai điều kiện cùng lúc:

\begin{itemize}[leftmargin=1.4em, itemsep=2pt]
\item Căn thức: $5 - x \ge 0 \implies x \le 5$
\item Logarit: $x - 2 > 0 \implies x > 2$
\end{itemize}

Giao hai điều kiện: $2 < x \le 5$.

\nhan{Chú ý} Biểu thức trong logarit phải \textbf{lớn hơn} 0, không lấy dấu bằng. Đáp án B sai vì lấy $x = 2$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[27 — Đáp án C]

\textbf{Giải.} Hàm bậc hai có $a = -1 < 0$ nên đạt cực đại tại đỉnh $x = -\dfrac{b}{2a} = 2$.

$$y\_{\max} = -(2)^2 + 4 \cdot 2 - 1 = 3$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[28 — Đáp án B]

\textbf{Giải.} Thay $x = -2$: $f(-2) = (-2)^2 + 2(-2) = 4 - 4 = 0$.

\nhan{Bẫy} Chọn D là tính $2^2 + 2 \cdot 2$ và bỏ qua dấu âm.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[29 — Đáp án D]

\textbf{Giải.} Hàm mũ $y = a^x$ với $a > 0$ luôn nhận giá trị dương với mọi $x$, và không bao giờ bằng 0. Tập giá trị là $(0, +\infty)$.

\nhan{Bẫy} Chọn A là lấy cả giá trị 0, nhưng $3^x = 0$ vô nghiệm.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[30 — Đáp án A]

\textbf{Giải.} Hàm $y = x^2 + 1$ thỏa mãn $f(-x) = (-x)^2 + 1 = f(x)$ nên là hàm chẵn.

Đáp án B là hàm lẻ, C và D không có tính chẵn lẻ.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[31 — Đáp án C]

\textbf{Giải.} $f(2) = 2 + \dfrac{1}{2} = \dfrac{5}{2}$ và $f\!\left(\dfrac{1}{2}\right) = \dfrac{1}{2} + 2 = \dfrac{5}{2}$.

$$f(2) + f\!\left(\dfrac{1}{2}\right) = 5$$

\nhan{Nhận xét} Với $f(x) = x + \dfrac{1}{x}$ luôn có $f(x) = f\!\left(\dfrac{1}{x}\right)$, đây là tính chất giúp tính nhanh.

\section{PHẦN V — TẬP HỢP VÀ BẤT ĐẲNG THỨC}


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[32 — Đáp án B]

\textbf{Giải.} Giải bất phương trình $x^2 - 4x - 5 < 0$: hai nghiệm $-1$ và $5$, hệ số $a = 1 > 0$ nên lấy trong khoảng.

$$A = (-1, 5)$$

Giao với $B = [0, +\infty)$: phần chung là $[0, 5)$.

\nhan{Bẫy} Chọn A là quên rằng $x = 0$ thuộc $B$ nên phải lấy, đầu mút ở 0 phải là ngoặc vuông.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[33 — Đáp án A]

\textbf{Giải.} $\lvert 3x - 2 \rvert < 4 \iff -4 < 3x - 2 < 4$.

Cộng 2: $-2 < 3x < 6$. Chia 3: $-\dfrac{2}{3} < x < 2$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[34 — Đáp án C]

\textbf{Giải.} Tử có nghiệm $x = -2$, mẫu có nghiệm $x = 3$ (loại). Hai điểm chia trục số thành ba khoảng:

\bang{
\begin{tabularx}{\linewidth}{@{\hspace{6pt}}Xccc@{\hspace{6pt}}}
\toprule
\textbf{Khoảng} & \textbf{Tử $x+2$} & \textbf{Mẫu $x-3$} & \textbf{Thương} \\
\midrule
$x < -2$ & $-$ & $-$ & $+$ \\
$-2 < x < 3$ & $+$ & $-$ & $-$ (đúng) \\
$x > 3$ & $+$ & $+$ & $+$ \\
\bottomrule
\end{tabularx}}

Tập nghiệm: $(-2, 3)$.

\nhan{Chú ý} Đáp án B sai vì dùng dấu $\le$ trong khi đề dùng dấu $<$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[35 — Đáp án D]

\textbf{Giải.} Áp dụng bất đẳng thức Cauchy cho hai số dương $x$ và $4 - x$:

$$x(4 - x) \le \left(\dfrac{x + (4-x)}{2}\right)^2 = \left(\dfrac{4}{2}\right)^2 = 4$$

Dấu bằng xảy ra khi $x = 4 - x \iff x = 2$, thỏa mãn $0 < x < 4$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[36 — Đáp án B]

\textbf{Giải.} Hệ số $a = 1 > 0$ nên chỉ cần điều kiện $\Delta < 0$:

$$\Delta = m^2 - 4 < 0 \iff -2 < m < 2$$

\nhan{Chú ý} Đáp án A sai vì lấy cả hai đầu mút; khi $m = \pm 2$ thì $\Delta = 0$ và biểu thức bằng 0 tại một điểm, không thỏa mãn dấu $>$.

\textbf{Giải.} Theo định nghĩa logarit:

$$\log\_2(x - 3) = 2 \iff x - 3 = 2^2 = 4 \implies x = 7$$

\nhan{Chú ý} Phải kiểm tra điều kiện $x - 3 > 0$; với $x = 7$ điều kiện thỏa mãn.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[40 — Đáp án D]

\textbf{Giải.} Biểu thức trong logarit phải dương:

$$x + 2 > 0 \implies x > -2$$

Tập xác định là $(-2, +\infty)$.

\section{PHẦN VII — VECTOR VÀ SỐ PHỨC}


\end{dapan}

\begin{dapan}[41 — Đáp án B]

\textbf{Giải.} Tích vô hướng bằng tổng tích các tọa độ tương ứng:

$$a \cdot b = 3 \cdot (-2) + 1 \cdot 4 = -6 + 4 = -2$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[42 — Đáp án C]

\textbf{Giải.} Nhân cả tử và mẫu với liên hợp của mẫu:

$$z = \dfrac{(3+i)(1+i)}{(1-i)(1+i)} = \dfrac{3 + 3i + i + i^2}{2} = \dfrac{2 + 4i}{2} = 1 + 2i$$

Phần ảo là 2.

\nhan{Chú ý} Đáp án D sai vì phần ảo là số thực 2, không phải $2i$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[43 — Đáp án A]

\textbf{Giải.} Môđun của $z = a + bi$ là $\lvert z \rvert = \sqrt{a^2 + b^2}$:

$$\lvert z \rvert = \sqrt{1^2 + 2^2} = \sqrt{5}$$

\section{PHẦN IX — HÌNH HỌC PHẲNG VÀ KHÔNG GIAN}


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[46 — Đáp án A]

\textbf{Giải.} Mặt cầu $(x - a)^2 + (y - b)^2 + (z - c)^2 = R^2$ có tâm $(a, b, c)$ và bán kính $R$.

Tâm $(-1, 2, 3)$, bán kính $\sqrt{25} = 5$.

\nhan{Bẫy} Chọn C là quên lấy căn bậc hai của 25.
