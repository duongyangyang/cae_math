\subsection*{Nhóm A — Conic}

\begin{dapan}[1 — Đáp án B]
\textbf{Giải.} Tiêu điểm nằm trên trục hoành nên phương trình chính tắc có dạng $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = 1$ với $a > b > 0$.

Từ $F_1(-3, 0)$ và $F_2(3, 0)$ được $c = 3$. Điểm $P(5, 0)$ nằm trên trục hoành và thuộc elip nên là một đỉnh, suy ra $a = 5$.

$$b^2 = a^2 - c^2 = 25 - 9 = 16$$

Vậy phương trình là $\dfrac{x^2}{25} + \dfrac{y^2}{16} = 1$.

\textbf{Bẫy.} Đáp án A đảo mẫu số, biến elip nằm ngang thành elip nằm dọc. Đáp án D đúng dạng nhưng lấy $b^2 = c^2 = 9$, tức quên rằng $b^2 = a^2 - c^2$ chứ không phải $b = c$.
\end{dapan}

\begin{dapan}[2 — Đáp án C]
\textbf{Giải.} Mẫu $25$ lớn hơn nằm dưới $x^2$ nên $a^2 = 25$, $b^2 = 9$ và trục lớn nằm trên trục hoành.

$$c^2 = a^2 - b^2 = 25 - 9 = 16 \implies c = 4$$

Tiêu điểm là $(\pm 4, 0)$.

\textbf{Bẫy.} Đáp án A là $(\pm 3, 0)$, lấy $\sqrt{b^2} = 3$ làm $c$. Phải tính $c^2 = a^2 - b^2$ trước khi lấy căn.
\end{dapan}

\begin{dapan}[3 — Đáp án D]
\textbf{Giải.} Mẫu $100$ lớn hơn nằm dưới $y^2$ nên $a^2 = 100$, $b^2 = 25$ và trục lớn nằm trên trục tung.

$$c^2 = a^2 - b^2 = 100 - 25 = 75 \implies c = \sqrt{75} = 5\sqrt{3}$$

Tiêu điểm nằm trên trục tung: $(0, \pm\sqrt{75})$.

\textbf{Chú ý.} Đáp án A và C đặt tiêu điểm trên trục hoành, tức chọn nhầm trục. Mẫu lớn hơn nằm dưới biến nào thì tiêu điểm nằm trên trục của biến kia.
\end{dapan}

\begin{dapan}[4 — Đáp án C]
\textbf{Giải.} Chia hai vế cho $36$ để đưa về dạng chính tắc:

$$\frac{x^2}{9} + \frac{y^2}{4} = 1$$

Mẫu $9$ lớn hơn nằm dưới $x^2$ nên $a^2 = 9$, suy ra $a = 3$.

Đề hỏi 长轴长 (độ dài trục lớn), tức $2a = 6$.

\textbf{Bẫy.} Đáp án A là $3$, tức giá trị $a$ chứ không phải $2a$. Phân biệt 长轴长 (trục lớn, bằng $2a$) và 半长轴 (nửa trục lớn, bằng $a$).
\end{dapan}

\begin{dapan}[5 — Đáp án A]
\textbf{Giải.} Mẫu $81$ lớn hơn nằm dưới $x^2$ nên $a^2 = 81$, $b^2 = 36$ và tiêu điểm nằm trên trục hoành.

$$c^2 = a^2 - b^2 = 81 - 36 = 45 \implies c = \sqrt{45} = 3\sqrt{5}$$

Tiêu điểm là $(\pm 3\sqrt{5}, 0)$.

\textbf{Cách nhanh.} Kiểm tra bằng bất đẳng thức: $c$ phải nhỏ hơn $a = 9$ và lớn hơn $b = 6$. Giá trị $3\sqrt{5} \approx 6{,}7$ thỏa mãn cả hai, còn đáp án C là $3$ thì nhỏ hơn $b$ nên loại ngay.
\end{dapan}

\begin{dapan}[6 — Đáp án B]
\textbf{Giải.} Cho vế phải của phương trình chính tắc bằng $0$:

$$\frac{x^2}{16} - \frac{y^2}{9} = 0 \iff \frac{y^2}{9} = \frac{x^2}{16} \iff y^2 = \frac{9}{16}x^2$$

$$y = \pm\frac{3}{4}x$$

\textbf{Bẫy.} Đáp án A đảo ngược tỉ số thành $\dfrac{4}{3}$, còn đáp án C quên lấy căn nên giữ nguyên $\dfrac{9}{16}$. Viết rõ bước lấy căn để tránh cả hai.
\end{dapan}

\begin{dapan}[7 — Đáp án A]
\textbf{Giải.} Từ tiệm cận $y = \pm\dfrac{1}{2}x$ được $\dfrac{b}{a} = \dfrac{1}{2}$, tức $b = \dfrac{a}{2}$.

Điểm $(2, 0)$ thuộc hyperbol và nằm trên trục hoành, đây là một đỉnh, suy ra $a = 2$. Do đó $b = 1$.

Phương trình là $\dfrac{x^2}{4} - \dfrac{y^2}{1} = 1$.

\textbf{Cách nhanh.} Điểm cho trước nằm trên trục hoành luôn là đỉnh của hyperbol có tiêu điểm trên trục hoành, đọc ngay $a$ bằng hoành độ của điểm đó.
\end{dapan}

\begin{dapan}[8 — Đáp án A]
\textbf{Giải.} Từ phương trình được $a^2 = 64$, $b^2 = 16$, suy ra $a = 8$.

Với hyperbol:

$$c^2 = a^2 + b^2 = 64 + 16 = 80 \implies c = \sqrt{80} = 4\sqrt{5}$$

$$e = \frac{c}{a} = \frac{4\sqrt{5}}{8} = \frac{\sqrt{5}}{2}$$

\textbf{Chú ý.} Với hyperbol thì $c > a$ nên $e > 1$. Đáp án B là $\dfrac{2}{5}$ nhỏ hơn $1$, loại ngay mà không cần tính.
\end{dapan}

\begin{dapan}[9 — Đáp án D]
\textbf{Giải.} Số hạng mang dấu cộng chứa $y^2$ nên $a^2 = 25$, tức trục thực nằm trên trục tung và tiêu điểm có dạng $(0, \pm c)$. Từ $b^2 = 144$:

$$c^2 = a^2 + b^2 = 25 + 144 = 169 \implies c = 13$$

Tiêu điểm là $(0, \pm 13)$.

\textbf{Bẫy.} Đáp án C là $(\pm 13, 0)$, đúng độ lớn nhưng sai trục. Với hyperbol, tiêu điểm luôn nằm trên trục của số hạng mang dấu cộng.
\end{dapan}

\begin{dapan}[10 — Đáp án C]
\textbf{Giải.} Số hạng mang dấu cộng chứa $y^2$ nên $a^2 = 100$, suy ra $a = 10$. Từ $b^2 = 64$:

$$c^2 = a^2 + b^2 = 100 + 64 = 164 \implies c = \sqrt{164} = 2\sqrt{41}$$

$$e = \frac{c}{a} = \frac{2\sqrt{41}}{10} = \frac{\sqrt{41}}{5}$$

\textbf{Bẫy.} Đáp án A có dấu âm, mà tâm sai luôn dương nên loại ngay. Đáp án B quên chia cho $2$ ở bước rút gọn phân số.
\end{dapan}

\begin{dapan}[11 — Đáp án B]
\textbf{Giải.} Phương trình $y^2 = -2px$ với $p > 0$ có hệ số âm trước $x$. Biến bình phương là $y$ nên trục đối xứng là trục hoành, parabol mở theo chiều ngang. Dấu trừ quyết định mở về phía âm của trục hoành, tức sang trái.

\textbf{Cách nhanh.} Đối chiếu bảng bốn dạng phương trình: $y^2 = 2px$ mở phải, $y^2 = -2px$ mở trái. Chỉ cần nhớ dấu trừ đảo hướng mở.
\end{dapan}

\begin{dapan}[12 — Đáp án D]
\textbf{Giải.} Phương trình $x^2 = 16y$ có dạng $x^2 = 2py$ với $2p = 16$, suy ra $p = 8$.

Parabol mở lên trên nên đường chuẩn nằm phía dưới, cách đỉnh một khoảng $\dfrac{p}{2} = 4$:

$$y = -4$$

\textbf{Bẫy.} Đáp án A và C đặt đường chuẩn là đường thẳng đứng, tức chọn nhầm trục. Vì biến bình phương là $x$ nên đường chuẩn phải là đường nằm ngang $y = \text{hằng số}$.
\end{dapan}

\subsection*{Nhóm B — Đường thẳng và đường tròn}

\begin{dapan}[13 — Đáp án D]
\textbf{Giải.} Dùng công thức đường thẳng qua điểm $M(x_0, y_0)$ với hệ số góc $k$:

$$y - y\_0 = k(x - x\_0)$$

Thay $x_0 = 1$, $y_0 = -2$, $k = 3$:

$$y + 2 = 3(x - 1) \implies y + 2 = 3x - 3 \implies y = 3x - 5$$

\textbf{Bẫy.} Đáp án A cho $y = 3x + 5$, tức nhầm dấu của tung độ khi thay vào công thức. Chú ý $y - (-2) = y + 2$.
\end{dapan}

\begin{dapan}[14 — Đáp án A]
\textbf{Giải.} Trung điểm $M$ của $AB$:

$$M\left(\frac{1+3}{2}, \frac{2+4}{2}\right) = (2, 3)$$

Hệ số góc của $AB$:

$$k\_{AB} = \frac{4-2}{3-1} = 1$$

Đường trung trực vuông góc với $AB$ nên $k = -1$. Phương trình qua $M(2, 3)$:

$$y - 3 = -1(x - 2) \implies y = -x + 5 \implies x + y - 5 = 0$$

\textbf{Cách nhanh.} Kiểm tra bằng tính chất: mọi điểm trên trung trực cách đều $A$ và $B$. Thay $A(1,2)$ và $B(3,4)$ vào $x + y - 5 = 0$ đều được $0$, còn các đáp án khác không cho hai giá trị bằng nhau.
\end{dapan}

\begin{dapan}[15 — Đáp án C]
\textbf{Giải.} Vectơ $\overrightarrow{OP} = (3, 4)$ nghĩa là điểm $P$ có tọa độ $(3, 4)$. Khoảng cách từ $P$ đến gốc tọa độ là độ dài vectơ:

$$OP = \sqrt{3^2 + 4^2} = \sqrt{9 + 16} = \sqrt{25} = 5$$

\textbf{Chú ý.} Bộ ba $3$, $4$, $5$ là bộ số Pythagoras quen thuộc, gặp là nhận ra ngay mà không cần tính.
\end{dapan}

\begin{dapan}[16 — Đáp án C]
\textbf{Giải.} Hệ số góc của đường thẳng $y = \sqrt{3}x + 1$ là $k = \sqrt{3}$. Gọi $\alpha$ là góc nghiêng:

$$\tan\alpha = \sqrt{3} \implies \alpha = 60^\circ$$

\textbf{Chú ý.} Giá trị góc nghiêng nằm trong khoảng từ $0^\circ$ đến $180^\circ$. Bảng giá trị cần nhớ: $\tan 30^\circ = \dfrac{\sqrt{3}}{3}$, $\tan 45^\circ = 1$, $\tan 60^\circ = \sqrt{3}$.
\end{dapan}

\begin{dapan}[17 — Đáp án D]
\textbf{Giải.} Đường thẳng $3x - 5y + 6 = 0$ có hệ số góc $k_1 = \dfrac{3}{5}$. Đường thẳng cần tìm vuông góc với nó nên:

$$k = -\frac{1}{k\_1} = -\frac{5}{3}$$

Phương trình qua điểm $(4, -3)$:

$$y + 3 = -\frac{5}{3}(x - 4) \implies 3y + 9 = -5x + 20 \implies 5x + 3y - 11 = 0$$

\textbf{Cách nhanh.} Kiểm tra bằng tích các hệ số: đường thẳng $Ax + By + C = 0$ vuông góc với $3x - 5y + 6 = 0$ khi $3A - 5B = 0$. Với $5x + 3y - 11 = 0$ được $3 \cdot 5 - 5 \cdot 3 = 0$, thỏa mãn.
\end{dapan}

\begin{dapan}[18 — Đáp án A]
\textbf{Giải.} Tâm của đường tròn nằm trên trung trực của đoạn nối $(0,0)$ và $(4,0)$, tức đường thẳng $x = 2$. Đáp án A có tâm $(2, 0)$ và bán kính $r = 2$.

Kiểm tra: khoảng cách từ tâm $(2,0)$ tới $(0,0)$ là $2$, tới $(4,0)$ cũng là $2$, đều bằng bán kính.

\textbf{Bẫy.} Đáp án B có cùng tâm nhưng bán kính $4$, không đi qua hai điểm đã cho vì khoảng cách thực tế chỉ là $2$.
\end{dapan}

\begin{dapan}[19 — Đáp án D]
\textbf{Giải.} Nhóm các số hạng theo ẩn và thêm bớt để thành bình phương đủ:

$$x^2 - 4x + y^2 + 6y - 3 = 0$$

$$(x^2 - 4x + 4) + (y^2 + 6y + 9) = 4 + 9 + 3$$

$$(x - 2)^2 + (y + 3)^2 = 16$$

Vậy bán kính $r = \sqrt{16} = 4$.

\textbf{Chú ý.} Số hạng tự do sau khi nhóm là $4 + 9 + 3 = 16$, lấy dấu cộng vì số hạng $-3$ được chuyển vế. Nhầm dấu ở bước này cho $r^2 = 4 + 9 - 3 = 10$, tức đáp án A.
\end{dapan}

\subsection*{Nhóm C — Hình học phẳng và không gian}

\begin{dapan}[20 — Đáp án C]
\textbf{Giải.} Thể tích hình chóp bằng một phần ba tích của diện tích đáy và chiều cao:

$$V = \frac{1}{3}Sh$$

\textbf{Chú ý.} Đáp án A là công thức thể tích hình lăng trụ, áp dụng cho hình chóp sẽ sai vì hình chóp cùng đáy và cùng chiều cao chỉ có thể tích bằng một phần ba.
\end{dapan}

\begin{dapan}[21 — Đáp án B]
\textbf{Giải.} Diện tích đáy cho trước là $9\pi$ nên không cần tìm bán kính. Thể tích hình nón:

$$V = \frac{1}{3}Sh = \frac{1}{3} \cdot 9\pi \cdot 4 = 12\pi$$

\textbf{Cách nhanh.} Đề cho sẵn diện tích đáy thì thay thẳng vào công thức. Tìm bán kính từ $S = 9\pi$ rồi tính lại là bước thừa, dễ sai vì nhầm $r = 9$ hoặc $r = 3$ rồi bình phương sai.
\end{dapan}

\begin{dapan}[22 — Đáp án B]
\textbf{Giải.} Diện tích toàn phần của hình trụ gồm hai đáy và mặt xung quanh:

$$S = 2\pi r^2 + 2\pi rh$$

\textbf{Bẫy.} Đáp án A chỉ có phần mặt xung quanh, thiếu hai đáy. Đáp án C thiếu một đáy. Đề hỏi 表面积 (diện tích toàn phần) nên phải cộng đủ hai đáy.
\end{dapan}

\begin{dapan}[23 — Đáp án C]
\textbf{Giải.} Ba cạnh của tam giác $ABC$:

$$AB = \sqrt{(0-1)^2 + (1-0)^2 + 0} = \sqrt{2}$$

Tương tự $BC = \sqrt{2}$ và $CA = \sqrt{2}$. Vậy $ABC$ là tam giác đều cạnh $\sqrt{2}$.

$$S = \frac{\sqrt{3}}{4} \cdot \left(\sqrt{2}\right)^2 = \frac{\sqrt{3}}{4} \cdot 2 = \frac{\sqrt{3}}{2}$$

\textbf{Nhận xét.} Ba điểm $A(1,0,0)$, $B(0,1,0)$, $C(0,0,1)$ đều cách gốc tọa độ một khoảng bằng $1$ và cách đều nhau, nên tam giác đều. Nhận ra điều này dùng được công thức diện tích tam giác đều thay vì phải tính vectơ.
\end{dapan}

\subsection*{Nhóm D — Ôn tập tích lũy}

\begin{dapan}[24 — Đáp án A]
\textbf{Giải.} Xét từng cặp:

\begin{itemize}[leftmargin=1.4em, itemsep=2pt]
\item \textbf{A.} Tập $A$ gồm các bội của $12$ từ $18$ trở lên, tức $\{24, 36, 48, \dots\}$. Tập $B = \{12, 14, 18\}$. Hai tập khác nhau. (đây là cặp cần tìm)
\item \textbf{B.} $A = \{3, 9, 27, 81\}$ và $B = \{3^1, 3^2, 3^3, 3^4\} = \{3, 9, 27, 81\}$. Hai tập bằng nhau.
\item \textbf{C.} $A$ gồm các số nguyên lớn hơn $-2$ và không vượt quá $3$, tức $\{-1, 0, 1, 2, 3\}$, bằng $B$.
\item \textbf{D.} $A$ gồm các số tự nhiên nhỏ hơn $5$, tức $\{0, 1, 2, 3, 4\}$, bằng $B$.
\end{itemize}


\textbf{Chú ý.} Điều kiện $x \ge 18$ trong đáp án A loại luôn hai phần tử $12$ và $14$ của tập $B$, còn $18$ thuộc $B$ nhưng không thuộc $A$ vì $18$ không chia hết cho $12$.
\end{dapan}

\begin{dapan}[25 — Đáp án B]
\textbf{Giải.} Dùng công thức số hạng tổng quát $a_n = a_1 + (n-1)d$. Điều kiện $a_3 = 4$ cho:

$$a\_1 + 2d = 4$$

Điều kiện $a_2 + a_6 = 10$ cho:

$$(a\_1 + d) + (a\_1 + 5d) = 2a\_1 + 6d = 10 \implies a\_1 + 3d = 5$$

Trừ hai phương trình: $d = 1$. Thay lại: $a_1 = 4 - 2 = 2$.

\textbf{Cách nhanh.} Trong cấp số cộng, $a_2 + a_6 = 2a_4$, nên $2a_4 = 10$ cho $a_4 = 5$. Từ $a_3 = 4$ và $a_4 = 5$ được $d = 1$, suy ra $a_1 = a_3 - 2d = 2$.
\end{dapan}

\subsection*{Nhóm D — Ôn tập tích lũy}

\begin{dapan}[1 — Đáp án A]
\textbf{Giải.} Đề cho tiêu cự $2c = 2$ nên $c = 1$. Từ tâm sai $e = \dfrac{c}{a} = \dfrac{1}{2}$ được $a = 2$, suy ra $a^2 = 4$.

$$b^2 = a^2 - c^2 = 4 - 1 = 3$$

Điều kiện $a > b > 0$ và tiêu điểm trên trục hoành cho phương trình $\dfrac{x^2}{4} + \dfrac{y^2}{3} = 1$.

\textbf{Bẫy.} Đáp án B đảo mẫu số nên trục lớn thành trục tung, không khớp điều kiện $a > b > 0$ với $a^2 = 4$.
\end{dapan}

\begin{dapan}[2 — Đáp án B]
\textbf{Giải.} Từ $\dfrac{x^2}{36} - \dfrac{y^2}{25} = 1$ được $a^2 = 36$, $b^2 = 25$, tức $a = 6$, $b = 5$.

Tiệm cận:

$$y = \pm\frac{b}{a}x = \pm\frac{5}{6}x$$

\textbf{Chú ý.} Tỉ số là $\dfrac{b}{a} = \dfrac{5}{6}$, không phải $\dfrac{a}{b}$. Cách kiểm tra: cho vế phải bằng $0$ rồi giải, được $y^2 = \dfrac{25}{36}x^2$, tức $y = \pm\dfrac{5}{6}x$.
\end{dapan}

\begin{dapan}[3 — Đáp án C]
\textbf{Giải.} Đưa phương trình về dạng chuẩn bằng cách chia hai vế cho $4$:

$$x^2 = \frac{y}{4}$$

So sánh với $x^2 = 2py$ được $2p = \dfrac{1}{4}$, suy ra $p = \dfrac{1}{8}$. Parabol mở lên trên nên tiêu điểm nằm trên trục tung:

$$F\left(0, \frac{p}{2}\right) = \left(0, \frac{1}{16}\right)$$

\textbf{Bẫy.} Đáp án A là $\left(0, \dfrac{1}{4}\right)$, tức đọc thẳng hệ số $4$ thành $p$. Hệ số $4$ là $2p$ sau khi đảo vế, không phải $p$.
\end{dapan}

\begin{dapan}[4 — Đáp án A]
\textbf{Giải.} Mẫu $16$ lớn hơn nằm dưới $y^2$ nên $a^2 = 16$, $b^2 = 9$, tức trục lớn nằm trên trục tung.

$$c^2 = a^2 - b^2 = 16 - 9 = 7 \implies c = \sqrt{7}$$

$$e = \frac{c}{a} = \frac{\sqrt{7}}{4}$$

\textbf{Bẫy.} Đáp án C là $\dfrac{4}{3}$ lớn hơn $1$, không thể là tâm sai của elip. Đáp án D lấy $\dfrac{b}{a}$ thay vì $\dfrac{c}{a}$.
\end{dapan}

\begin{dapan}[5 — Đáp án D]
\textbf{Giải.} Phương trình $x^2 = -12y$ có dạng $x^2 = -2py$ với $2p = 12$, suy ra $p = 6$.

Parabol mở xuống dưới nên đường chuẩn nằm phía trên, cách đỉnh một khoảng $\dfrac{p}{2} = 3$:

$$y = 3$$

\textbf{Chú ý.} Đáp án A là $y = -3$, lấy nhầm dấu. Với parabol mở xuống, tiêu điểm ở dưới và đường chuẩn ở trên, tức giá trị dương.
\end{dapan}

\begin{dapan}[6 — Đáp án B]
\textbf{Giải.} Điều kiện để $\dfrac{x^2}{k+1} + \dfrac{y^2}{1-k} = 1$ biểu diễn elip là hai mẫu số cùng dương và khác nhau:

$$k + 1 > 0, \quad 1 - k > 0, \quad k + 1 \ne 1 - k$$

$$k > -1, \quad k < 1, \quad k \ne 0$$

Vậy điều kiện là $-1 < k < 1$ và $k \ne 0$.

Xét hai chiều: từ $-1 < k < 1$ không suy ra được elip vì tại $k = 0$ phương trình thành $x^2 + y^2 = 1$, tức đường tròn. Ngược lại, elip đòi hỏi $-1 < k < 1$ và $k \ne 0$, suy ra $-1 < k < 1$. Vậy $-1 < k < 1$ là điều kiện cần nhưng không đủ.

\textbf{Bẫy.} Học sinh hay quên trường hợp $k = 0$ làm hai mẫu số bằng nhau. Đây là điểm khác nhau giữa đường tròn và elip.
\end{dapan}

\begin{dapan}[7 — Đáp án B]
\textbf{Giải.} Kí hiệu $PF_1 = m$, $PF_2 = n$. Theo định nghĩa elip:

$$m + n = 2a$$

Trong tam giác $F_1PF_2$ có $\angle F_1PF_2 = 60^\circ$ và $F_1F_2 = 2c$. Định lý cosin:

$$4c^2 = m^2 + n^2 - 2mn\cos 60^\circ = (m+n)^2 - 2mn - mn = 4a^2 - 3mn$$

$$\implies mn = \frac{4a^2 - 4c^2}{3}$$

Bán kính đường tròn ngoại tiếp tam giác:

$$R = \frac{F\_1F\_2}{2\sin 60^\circ} = \frac{2c}{\sqrt{3}}$$

Diện tích tam giác là $S = \dfrac{1}{2}mn\sin 60^\circ = \dfrac{\sqrt{3}}{4}mn$, nửa chu vi là $a + c$, nên bán kính nội tiếp:

$$r = \frac{S}{a+c} = \frac{\sqrt{3}(a^2 - c^2)}{3(a+c)} = \frac{\sqrt{3}}{3}(a - c)$$

Điều kiện $R = 4r$:

$$\frac{2c}{\sqrt{3}} = \frac{4\sqrt{3}}{3}(a - c) \implies 2c = 4(a - c) \implies 6c = 4a \implies e = \frac{c}{a} = \frac{2}{3}$$

\textbf{Nhận xét.} Bước then chốt là dùng định lý cosin để rút ra tích $mn$ theo $a$ và $c$. Không cần tìm $m$, $n$ riêng lẻ.
\end{dapan}

\begin{dapan}[8 — Đáp án A]
\textbf{Giải.} Mẫu $100$ lớn hơn nằm dưới $x^2$ nên $a^2 = 100$, $b^2 = 36$ và tiêu điểm nằm trên trục hoành.

$$c^2 = a^2 - b^2 = 100 - 36 = 64 \implies c = 8$$

Tiêu điểm là $(\pm 8, 0)$.

\textbf{Cách nhanh.} Kiểm tra bằng bất đẳng thức $b < c < a$: được $6 < 8 < 10$, thỏa mãn.
\end{dapan}

\begin{dapan}[9 — Đáp án C]
\textbf{Giải.} Từ phương trình được $a^2 = 16$, $b^2 = 7$, suy ra $c^2 = 16 - 7 = 9$, tức $c = 3$ và tiêu cự $F_1F_2 = 6$.

Gọi $P(x_0, y_0)$. Điều kiện $OP = 3$ cho $x_0^2 + y_0^2 = 9$. Thay $x_0^2 = 9 - y_0^2$ vào phương trình elip:

$$\frac{9 - y\_0^2}{16} + \frac{y\_0^2}{7} = 1$$

Nhân hai vế với $112$:

$$7(9 - y\_0^2) + 16y\_0^2 = 112 \implies 63 + 9y\_0^2 = 112 \implies y\_0^2 = \frac{49}{9} \implies |y\_0| = \frac{7}{3}$$

Diện tích tam giác $PF_1F_2$ với đáy là tiêu cự và chiều cao là $|y_0|$:

$$S = \frac{1}{2} \cdot 6 \cdot \frac{7}{3} = 7$$

\textbf{Nhận xét.} Không cần tìm $x_0$ vì diện tích chỉ phụ thuộc $|y_0|$. Dừng ở $y_0^2$ là đủ.
\end{dapan}

\begin{dapan}[10 — Đáp án D]
\textbf{Giải.} Theo định nghĩa elip, tổng khoảng cách từ một điểm bất kỳ trên elip tới hai tiêu điểm bằng $2a$.

Từ phương trình $\dfrac{x^2}{169} + \dfrac{y^2}{25} = 1$ được $a^2 = 169$, tức $a = 13$.

$$2a = 26$$

\textbf{Cách nhanh.} Đây là câu hỏi định nghĩa thuần túy, không cần tìm tiêu điểm. Chỉ cần đọc $a$ và nhân đôi.
\end{dapan}

\begin{dapan}[11 — Đáp án C]
\textbf{Giải.} Gọi hai đỉnh trên trục lớn là $A(-a, 0)$ và $B(a, 0)$, tiêu điểm $F(c, 0)$ với $0 < c < a$.

$$|AF| = a + c, \qquad |BF| = a - c$$

Điều kiện $|AF| = 3|BF|$:

$$a + c = 3(a - c) \implies a + c = 3a - 3c \implies 4c = 2a \implies e = \frac{c}{a} = \frac{1}{2}$$

\textbf{Bẫy.} Nếu lấy $F(-c, 0)$ thì vai trò của $A$ và $B$ đổi chỗ, nhưng tỉ số vẫn cho cùng kết quả vì đề cho $|AF| = 3|BF|$ với $A$, $B$ là hai đỉnh đối xứng.
\end{dapan}

\begin{dapan}[12 — Đáp án A]
\textbf{Giải.} Parabol $y^2 = 4x$ có $2p = 4$, tức $p = 2$, tiêu điểm $F(1, 0)$.

Tia sáng xuất phát từ $M(3, 1)$ theo phương song song trục hoành, tức theo đường thẳng $y = 1$. Giao với parabol tại điểm $A$:

$$1^2 = 4x \implies x = \frac{1}{4}$$

Vậy $A\left(\dfrac{1}{4}, 1\right)$.

Theo tính chất phản xạ, tia phản xạ đi qua tiêu điểm $F$. Hệ số góc của đường thẳng $AF$:

$$k = \frac{0 - 1}{1 - \frac{1}{4}} = \frac{-1}{\frac{3}{4}} = -\frac{4}{3}$$

Tia này cắt parabol lần nữa tại $B$, và $A$, $B$, $F$ thẳng hàng nên đường thẳng $AB$ chính là đường $AF$.

\textbf{Cách nhanh.} Không cần tìm tọa độ $B$. Vì tia phản xạ đi qua tiêu điểm và cũng đi qua $A$, đường thẳng $AB$ chính là đường thẳng $AF$, nên hệ số góc tính được ngay từ hai điểm $A$ và $F$.
\end{dapan}

\begin{dapan}[13 — Đáp án B]
\textbf{Giải.} Viết lại điều kiện:

$$\left| \sqrt{(x-1)^2 + (y-1)^2} - \sqrt{(x+3)^2 + (y+3)^2} \right| = 4$$

Hai căn thức là khoảng cách từ $P(x, y)$ tới hai điểm $F_1(1, 1)$ và $F_2(-3, -3)$. Khoảng cách giữa hai điểm này:

$$F\_1F\_2 = \sqrt{(1+3)^2 + (1+3)^2} = \sqrt{16 + 16} = 4\sqrt{2}$$

Đặt $2a = 4$, tức $a = 2$. Vì $4\sqrt{2} \approx 5{,}66 > 4 = 2a$ nên hiệu hai khoảng cách là hằng số nhỏ hơn khoảng cách giữa hai tiêu điểm. Theo định nghĩa, quỹ tích là hyperbol.

\textbf{Bẫy.} Đáp án C là 两条射线 (hai tia), chỉ xảy ra khi $2a = F_1F_2$. Ở đây $2a < F_1F_2$ nên quỹ tích là hyperbol thật sự, không suy biến.
\end{dapan}

\begin{dapan}[14 — Đáp án D]
\textbf{Giải.} Parabol $y^2 = 4x$ có $2p = 4$, tức $p = 2$. Tiêu điểm $F(1, 0)$, đường chuẩn $x = -1$.

Theo định nghĩa, khoảng cách từ $P$ tới tiêu điểm bằng khoảng cách từ $P$ tới đường chuẩn. Với $P(x_0, y_0)$ trên parabol có $x_0 \ge 0$:

$$PF = x\_0 + \frac{p}{2} = x\_0 + 1$$

Điều kiện $PF = 5$:

$$x\_0 + 1 = 5 \implies x\_0 = 4$$

\textbf{Cách nhanh.} Dùng công thức bán kính qua tiêu $PF = x_0 + \dfrac{p}{2}$ thay vì lập phương trình khoảng cách với căn thức.
\end{dapan}

\begin{dapan}[15 — Đáp án C]
\textbf{Giải.} Giao điểm của đường thẳng $l$ với parabol có tung độ là nghiệm của phương trình thu được khi thay $x = y + 2$ (từ $kx - y - 2 = 0$ với $k$ là tham số) vào $x^2 = 2py$. Thay trực tiếp $x = \dfrac{y+2}{k}$:

$$\frac{(y+2)^2}{k^2} = 2py \implies (y+2)^2 = 2pk^2 y$$

Khai triển và sắp xếp theo $y$:

$$y^2 + (4 - 2pk^2)y + 4 = 0$$

Theo định lý Viète, tích hai nghiệm:

$$y\_1 y\_2 = 4$$

Điều kiện $y_P^2 = y_1 y_2 = 4$ với $P$ nằm trên $y$ 轴正半轴 (nửa dương trục tung) cho $y_P = 2$. Vậy $P(0, 2)$.

\textbf{Nhận xét.} Kết quả không phụ thuộc $k$ và $p$, đây là điểm đặc biệt của bài toán. Chỉ cần đọc số hạng tự do của phương trình bậc hai theo $y$ là ra ngay tích hai nghiệm.
\end{dapan}
