\subsection*{NHÓM A — CONIC}

\begin{baitap}[1]

\par\vspace{3pt}
\noindent A. $\dfrac{x^2}{16} + \dfrac{y^2}{25} = 1$ \quad B. $\dfrac{x^2}{25} + \dfrac{y^2}{16} = 1$ \quad C. $\dfrac{x^2}{9} + \dfrac{y^2}{25} = 1$ \quad D. $\dfrac{x^2}{25} + \dfrac{y^2}{9} = 1$

\begin{baitap}[9]

\section{Bài tập tự chọn}

Phần này luyện cùng nội dung nhưng ở mức khó hơn rõ rệt. Chỉ làm sau khi đã hoàn thành 25 câu bắt buộc. Không tính vào điểm bài tập về nhà.

\begin{baitap}[1]
椭圆 $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = 1$（其中 $a > b > 0$）的离心率为 $e = \dfrac{1}{2}$ ，且其焦距为 $2$ 。求该椭圆的方程是（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{x^2}{4} + \dfrac{y^2}{3} = 1$ \quad B. $\dfrac{x^2}{3} + \dfrac{y^2}{4} = 1$ \quad C. $\dfrac{x^2}{16} + \dfrac{y^2}{12} = 1$ \quad D. $\dfrac{x^2}{12} + \dfrac{y^2}{16} = 1$
\end{baitap}

\begin{baitap}[2]
双曲线 $\dfrac{x^2}{36} - \dfrac{y^2}{25} = 1$ 的渐近线方程是（ ）
\par\vspace{3pt}
\noindent A. $y = \pm\dfrac{36}{25}x$ \quad B. $y = \pm\dfrac{5}{6}x$ \quad C. $y = \pm\dfrac{25}{36}x$ \quad D. $y = \pm\dfrac{6}{5}x$
\end{baitap}

\begin{baitap}[3]
抛物线 $y = 4x^2$ 的焦点坐标是（ ）
\par\vspace{3pt}
\noindent A. $(0, \dfrac{1}{4})$ \quad B. $(\dfrac{1}{16}, 0)$ \quad C. $(0, \dfrac{1}{16})$ \quad D. $(\dfrac{1}{4}, 0)$
\end{baitap}

\begin{baitap}[4]
椭圆 $\dfrac{x^2}{9} + \dfrac{y^2}{16} = 1$ 的离心率是（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{\sqrt{7}}{4}$ \quad B. $\dfrac{\sqrt{7}}{3}$ \quad C. $\dfrac{4}{3}$ \quad D. $\dfrac{3}{4}$
\end{baitap}

\begin{baitap}[5]
抛物线 $x^2 = -12y$ 的准线方程是（ ）
\par\vspace{3pt}
\noindent A. $y = -3$ \quad B. $x = 3$ \quad C. $x = -3$ \quad D. $y = 3$
\end{baitap}

\begin{baitap}[6]
若 $k \in \mathbb{R}$ ，则 $-1 < k < 1$ 是“方程 $\dfrac{x^2}{k + 1} + \dfrac{y^2}{1 - k} = 1$ 表示椭圆”的（ ）
\par\vspace{3pt}
\noindent A. 充分不必要条件 \quad B. 必要不充分条件 \quad C. 充要条件 \quad D. 既不充分也不必要条件
\end{baitap}

\begin{baitap}[7]
设椭圆 $\dfrac{x^2}{a^2} + \dfrac{y^2}{b^2} = 1 \ (a > b > 0)$ 的焦点为 $F_1, F_2$ ， $P$ 是椭圆上一点，且 $\angle F_1 P F_2 = \dfrac{\pi}{3}$ ，若 $\Delta F_1 P F_2$ 的外接圆和内切圆的半径分别为 $R, r$ ，当 $R = 4r$ 时，椭圆的离心率为（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{4}{5}$ \quad B. $\dfrac{2}{3}$ \quad C. $\dfrac{1}{2}$ \quad D. $\dfrac{2}{5}$
\end{baitap}

\begin{baitap}[8]
椭圆 $\dfrac{x^2}{100} + \dfrac{y^2}{36} = 1$ 的焦点坐标是（ ）
\par\vspace{3pt}
\noindent A. $(\pm 8, 0)$ \quad B. $(\pm 4, 0)$ \quad C. $(0, \pm 8)$ \quad D. $(0, \pm 4)$
\end{baitap}

\begin{baitap}[9]
设 $F_1, F_2$ 分别是椭圆 $C: \dfrac{x^2}{16} + \dfrac{y^2}{7} = 1$ 的左、右焦点，$O$ 为坐标原点，点 $P$ 在 $C$ 上且 $|OP| = 3$ ，则 $\triangle PF_1F_2$ 的面积为（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{7}{2}$ \quad B. 8 \quad C. 7 \quad D. 16
\end{baitap}

\begin{baitap}[10]
椭圆 $\dfrac{x^2}{169} + \dfrac{y^2}{25} = 1$ 上任意一点到两个焦点的距离之和是（ ）
\par\vspace{3pt}
\noindent A. 12 \quad B. 13 \quad C. 24 \quad D. 26
\end{baitap}

\begin{baitap}[11]
已知椭圆 $C$ 的长轴的顶点分别为 $A$ 、 $B$ ，点 $F$ 为椭圆 $C$ 的一个焦点，若 $|AF| = 3|BF|$ ，则椭圆 $C$ 的离心率为（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{1}{3}$ \quad B. $\dfrac{\sqrt{2}}{2}$ \quad C. $\dfrac{1}{2}$ \quad D. $\dfrac{\sqrt{3}}{2}$
\end{baitap}

\begin{baitap}[12]
抛物线有如下光学性质：过焦点的光线经抛物线反射后平行于抛物线的对称轴；反之，平行于抛物线对称轴的入射光线经抛物线反射后必过抛物线的焦点。已知抛物线 $y^2 = 4x$ 的焦点为 $F$ ，一条平行于 $x$ 轴的光线从点 $M(3,1)$ 射出，经过抛物线上的点 $A$ 反射后，再经抛物线上的另一点 $B$ 射出，则直线 $AB$ 的斜率为（ ）
\par\vspace{3pt}
\noindent A. $-\dfrac{4}{3}$ \quad B. $\dfrac{4}{3}$ \quad C. $\pm\dfrac{4}{3}$ \quad D. $\dfrac{16}{9}$
\end{baitap}

\begin{baitap}[13]
已知点 $P(x, y)$ 的坐标满足 $\sqrt{(x-1)^2 + (y-1)^2} - \sqrt{(x+3)^2 + (y+3)^2} = \pm 4$ 则动点 $P$ 的轨迹是（ ）
\par\vspace{3pt}
\noindent A. 椭圆 \quad B. 双曲线 \quad C. 两条射线 \quad D. 以上都不对
\end{baitap}

\begin{baitap}[14]
已知抛物线 $y^2 = 4x$ 上一点 $P$ 到焦点的距离为 $5$ ，则点 $P$ 的横坐标是（ ）
\par\vspace{3pt}
\noindent A. 3 \quad B. $2\sqrt{5}$ \quad C. $5$ \quad D. 4
\end{baitap}

\begin{baitap}[15]
已知 $O$ 为坐标原点，直线 $l: kx - y - 2 = 0$ 交抛物线 $E: x^2 = 2py \ (p > 0)$ 于 $A$ ， $B$ 两点，$P$ 为 $y$ 轴正半轴上一点，点 $A$ ， $P$ ， $B$ 的纵坐标分别为 $y_1, y_P, y_2$ 且 $y_P^2 = y_1 y_2$ ，则 $P$ 的坐标为（ ）

\end{baitap}
