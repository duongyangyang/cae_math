\begin{dapan}[2 — Đáp án B]

\textbf{Giải.} $k=1$: phương trình thành $-6x+6=0$, có đúng một nghiệm. $k \ne 1$: có đúng một nghiệm khi $\Delta=(k+5)^2-24(k-1)=k^2-14k+49=0$, được $k=7$. Tổng là $1+7=8$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[3 — Đáp án A]

\end{dapan}
\vspace{0.3cm}
\begin{dapan}[4 — Đáp án D]

\textbf{Giải.} Tam thức đổi dấu qua hai nghiệm $1$ và $2$, nên $ax^2+bx+c>0$ đúng ở ngoài khoảng hai nghiệm.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[5 — Đáp án A]

\end{dapan}

\begin{dapan}[6 — Đáp án C]

\end{dapan}
\vspace{0.3cm}
\begin{dapan}[7 — Đáp án C]

\textbf{Giải.} $\sqrt{x^2-4x+4}=\lvert x-2 \rvert$, nên bất phương trình là $\lvert x-2 \rvert<1$, tức $1<x<3$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[8 — Đáp án C]

\textbf{Giải.} $f(x)=\displaystyle\int(3x^2-1)\,dx=x^3-x+C$. Thay $f(1)=0$ được $C=0$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[9 — Đáp án A]

\textbf{Giải.} $\displaystyle\int \dfrac{1}{3x}\,dx=\dfrac{1}{3}\int\dfrac{1}{x}\,dx=\dfrac{1}{3}\ln\lvert x \rvert+C$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[10 — Đáp án A]

\end{dapan}
\vspace{0.3cm}
\begin{dapan}[11 — Đáp án A]

\textbf{Giải.} Cần $4-x \ge 0$, $x-1>0$ và $\ln(x-1) \ne 0$. Ba điều kiện cho $1<x \le 4$ và $x \ne 2$, tức $(1,2) \cup (2,4]$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[12 — Đáp án C]

\end{dapan}

\begin{dapan}[25 — Đáp án A]

\begin{dapan}[26 — Đáp án D]

\textbf{Giải.} Từ $\overrightarrow{BA}+\overrightarrow{BC}=2\overrightarrow{BM}$ với $M$ là trung điểm $AC$, tích vô hướng bằng 0 nghĩa là trung tuyến $BM$ vuông góc với $AC$, suy ra $BA=BC$ tức $A=C$. Vectơ tổng ở điều kiện thứ hai là tổng hai vectơ đơn vị của $AB$ và $AC$, có độ lớn $2\cos\dfrac{A}{2}$; cho bằng $\sqrt{3}$ được $A=60^\circ$. Vậy $A=B=C=60^\circ$.


\end{dapan}
