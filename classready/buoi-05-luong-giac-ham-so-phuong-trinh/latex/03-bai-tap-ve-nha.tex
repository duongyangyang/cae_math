\subsection*{NHÓM A — HÀM SỐ VÀ PHƯƠNG TRÌNH LƯỢNG GIÁC}

\begin{baitap}[1]

\par\vspace{3pt}
\noindent A. $\left\{x \mid x \ne -\dfrac{\pi}{4} + k\pi,\ k \in \mathbb{Z}\right\}$ \quad B. $\left\{x \mid x \ne \dfrac{\pi}{4} + k\pi,\ k \in \mathbb{Z}\right\}$ \quad C. $\left\{x \mid x \ne \dfrac{\pi}{4} + 2k\pi,\ k \in \mathbb{Z}\right\}$ \quad D. $\left\{x \mid x \ne -\dfrac{\pi}{4} + 2k\pi,\ k \in \mathbb{Z}\right\}$
\end{baitap}

\begin{baitap}[2]
函数 $y = \sin x$ 在区间 $\left[0, \dfrac{\pi}{2}\right]$ 上是 ( )
\par\vspace{3pt}
\noindent A. 增函数 \quad B. 减函数 \quad C. 不单调 \quad D. 既不增也不减
\end{baitap}

\begin{baitap}[3]
函数 $y = 2\sin\!\left(\dfrac{\pi}{4} - x\right)$ 的最小正周期是（ ）
\par\vspace{3pt}
\noindent A. $2\pi$ \quad B. $\pi$ \quad C. $\dfrac{\pi}{2}$ \quad D. $4\pi$
\end{baitap}

\begin{baitap}[4]
方程 $\sin x = \dfrac{\sqrt{3}}{2}$ 在 $\left[0, \dfrac{\pi}{2}\right]$ 上的解是（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{\pi}{6}$ \quad B. $\dfrac{\pi}{3}$ \quad C. $\dfrac{\pi}{4}$ \quad D. $\dfrac{\pi}{2}$
\end{baitap}

\begin{baitap}[5]

\noindent A. 增函数 \quad B. 减函数 \quad C. 不单调 \quad D. 既不增也不减
\end{baitap}

\begin{baitap}[6]
声音是由于物体的振动产生的能引起听觉的波，每一个声音都是由纯音合成的，纯音的数学模型是三角函数。已知某音叉发出的纯音振动可表示为 $y=\frac{1}{3}\sin3x$，则该纯音振动的频率为（ ）
\par\vspace{3pt}
\noindent A. $\frac{1}{6\pi}$ \quad B. $\frac{2}{3\pi}$ \quad C. $\frac{3}{2\pi}$ \quad D. $\frac{2}{\pi}$
\end{baitap}

\begin{baitap}[7]
方程 $\cos x = -1$ 的解集是（ ）
\par\vspace{3pt}
\noindent A. $x = \dfrac{\pi}{2} + k\pi,\ k \in \mathbb{Z}$ \quad B. $x = \pi + 2k\pi,\ k \in \mathbb{Z}$ \quad C. $x = \pi + k\pi,\ k \in \mathbb{Z}$ \quad D. $x = 2k\pi,\ k \in \mathbb{Z}$
\end{baitap}

\begin{baitap}[8]

\end{baitap}

\begin{baitap}[9]

\begin{baitap}[10]

\par\vspace{3pt}

\begin{baitap}[11]

\noindent A. $\left[\dfrac{\pi}{12} + k\pi,\ \dfrac{7\pi}{12} + k\pi\right],\ k \in \mathbb{Z}$ \quad B. $\left[-\dfrac{\pi}{6} + k\pi,\ \dfrac{\pi}{3} + k\pi\right],\ k \in \mathbb{Z}$ \quad C. $\left[-\dfrac{\pi}{12} + k\pi,\ \dfrac{5\pi}{12} + k\pi\right],\ k \in \mathbb{Z}$ \quad D. $\left[\dfrac{\pi}{6} + 2k\pi,\ \dfrac{2\pi}{3} + 2k\pi\right],\ k \in \mathbb{Z}$

\begin{baitap}[12]

\subsection*{NHÓM C — HÀM SỐ BẬC HAI, MŨ, LOGARIT}

\begin{baitap}[20]

\par\vspace{3pt}
\noindent A. $\sqrt{2}$ \quad B. 2 \quad C. 4 \quad D. 16
\end{baitap}

\begin{baitap}[21]

\noindent A. $x^2 - x + 1$ \quad B. $2x^2 - x + 1$ \quad C. $x^2 - x$ \quad D. $2x^2 - x$
\end{baitap}

\begin{baitap}[22]

\par\vspace{3pt}

\end{baitap}

\begin{baitap}[23]

\subsection*{NHÓM D — ÔN TẬP TÍCH LŨY}

\begin{baitap}[24]

\par\vspace{3pt}
\noindent A. $\{x\in\mathbb{N}\mid x>2\}$ \quad B. $\{3\}$ \quad C. $\{x\in\mathbb{Z}\mid 2\le x\le3\}$ \quad D. $\{x\in\mathbb{N}\mid x\ge2\}$
\end{baitap}

\begin{baitap}[25]

\noindent A. 63 \quad B. 70 \quad C. 77 \quad D. 84
\end{baitap}
