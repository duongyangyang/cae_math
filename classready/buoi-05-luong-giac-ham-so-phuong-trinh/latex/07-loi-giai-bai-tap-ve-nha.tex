\subsection*{Nhóm A — Hàm số và phương trình lượng giác}

\begin{dapan}[1 — Đáp án A]
Hàm $\tan$ không xác định khi biểu thức bên trong bằng $\dfrac{\pi}{2} + k\pi$. Đặt $x + \dfrac{\pi}{4} = \dfrac{\pi}{2} + k\pi$, suy ra $x = \dfrac{\pi}{4} + k\pi$. Vậy tập xác định là $\left\{x \mid x \neq \dfrac{\pi}{4} + k\pi,\ k \in \mathbb{Z}\right\}$.

\textbf{Bẫy.} Dấu trong ngoặc là \textbf{cộng} $\dfrac{\pi}{4}$ nên nghiệm bị loại là $\dfrac{\pi}{4} + k\pi$, không phải $-\dfrac{\pi}{4} + k\pi$. Đảo dấu là sai.
\end{dapan}

\begin{dapan}[2 — Đáp án A]
Hàm $\sin x$ đồng biến trên $\left[-\dfrac{\pi}{2}, \dfrac{\pi}{2}\right]$. Khoảng $\left[0, \dfrac{\pi}{2}\right]$ nằm trọn trong khoảng đồng biến đó nên hàm số tăng trên toàn khoảng.
\end{dapan}

\begin{dapan}[3 — Đáp án A]
Hệ số của $x$ là $\omega = -1$, lấy trị tuyệt đối được $|\omega| = 1$. Chu kỳ là $T = \dfrac{2\pi}{1} = 2\pi$.

\textbf{Chú ý.} Dấu trừ của $\omega$ không ảnh hưởng đến chu kỳ vì công thức dùng $|\omega|$.
\end{dapan}

\begin{dapan}[4 — Đáp án B]
Phương trình $\sin x = \dfrac{\sqrt{3}}{2}$ trên $\left[0, \dfrac{\pi}{2}\right]$. Vì $\sin\dfrac{\pi}{3} = \dfrac{\sqrt{3}}{2}$ nên $x = \dfrac{\pi}{3}$.

\textbf{Chú ý.} Trên cả vòng tròn có hai nghiệm $\dfrac{\pi}{3}$ và $\dfrac{2\pi}{3}$, nhưng đề giới hạn trong $\left[0, \dfrac{\pi}{2}\right]$ nên chỉ lấy $\dfrac{\pi}{3}$.
\end{dapan}

\begin{dapan}[5 — Đáp án A]
Hàm $\tan x$ đồng biến trên từng khoảng $\left(-\dfrac{\pi}{2} + k\pi, \dfrac{\pi}{2} + k\pi\right)$. Khoảng $\left(\dfrac{\pi}{2}, \dfrac{3\pi}{2}\right)$ ứng với $k = 1$, nằm trọn trong một khoảng xác định nên hàm số tăng.
\end{dapan}

\begin{dapan}[6 — Đáp án C]
Biểu thức $y = \dfrac{1}{3}\sin 3x$ có $\omega = 3$. Tần số là $f = \dfrac{\omega}{2\pi} = \dfrac{3}{2\pi}$.

\textbf{Chú ý.} Phân biệt tần số $f = \dfrac{\omega}{2\pi}$ và chu kỳ $T = \dfrac{2\pi}{\omega}$. Đề hỏi 频率 (tần số) nên lấy $\omega$ chia $2\pi$.
\end{dapan}

\begin{dapan}[7 — Đáp án B]
Phương trình $\cos x = -1$ có nghiệm $x = \pi + 2k\pi$ với $k \in \mathbb{Z}$.
\end{dapan}

\begin{dapan}[8 — Đáp án C]
Từ $2\cos x + 1 = 0$ suy ra $\cos x = -\dfrac{1}{2}$. Trên $[0, 2\pi]$ có hai nghiệm $x = \dfrac{2\pi}{3}$ và $x = \dfrac{4\pi}{3}$.
\end{dapan}

\begin{dapan}[9 — Đáp án C]
Phương trình $\cos x = \dfrac{1}{2}$ trên $[0, \pi]$ có nghiệm duy nhất $x = \dfrac{\pi}{3}$.

\textbf{Chú ý.} Họ nghiệm tổng quát là $x = \pm\dfrac{\pi}{3} + 2k\pi$, nhưng nghiệm $x = -\dfrac{\pi}{3}$ nằm ngoài đoạn $[0, \pi]$ nên bị loại.
\end{dapan}

\begin{dapan}[10 — Đáp án C]
Đặt $t = \sin x$ với $t \in [-1, 1]$. Hàm số trở thành $y = 2t^2 - 4t + 1$, đỉnh tại $t_0 = \dfrac{4}{4} = 1$.

Vì $t_0 = 1$ nằm trong đoạn $[-1, 1]$ nên giá trị nhỏ nhất đạt tại $t = 1$: $y = 2 - 4 + 1 = -1$. Giá trị lớn nhất đạt tại đầu mút $t = -1$: $y = 2 + 4 + 1 = 7$.

Vậy tập giá trị là $[-1, 7]$.

\textbf{Chú ý.} Phải so sánh đỉnh $t_0$ với hai đầu mút $-1$ và $1$. Nếu thay $t_0$ một cách máy móc mà không kiểm tra thì dễ sai.
\end{dapan}

\begin{dapan}[11 — Đáp án C]
Hàm $\sin u$ đồng biến khi $-\dfrac{\pi}{2} + 2k\pi \le u \le \dfrac{\pi}{2} + 2k\pi$. Đặt $u = 2x - \dfrac{\pi}{3}$:

$$-\frac{\pi}{2} + 2k\pi \le 2x - \frac{\pi}{3} \le \frac{\pi}{2} + 2k\pi$$

Cộng $\dfrac{\pi}{3}$ vào cả ba vế: $-\dfrac{\pi}{6} + 2k\pi \le 2x \le \dfrac{5\pi}{6} + 2k\pi$. Chia cho $2$:

$$-\frac{\pi}{12} + k\pi \le x \le \frac{5\pi}{12} + k\pi$$

Vậy khoảng đồng biến là $\left[-\dfrac{\pi}{12} + k\pi,\ \dfrac{5\pi}{12} + k\pi\right]$, $k \in \mathbb{Z}$.
\end{dapan}

\begin{dapan}[12 — Đáp án D]
Phương trình $\cos 2x = 0$ khi $2x = \dfrac{\pi}{2} + k\pi$, suy ra $x = \dfrac{\pi}{4} + \dfrac{k\pi}{2}$ với $k \in \mathbb{Z}$.

\textbf{Chú ý.} Chia cả hai vế cho $2$ thì $k\pi$ cũng bị chia, thành $\dfrac{k\pi}{2}$. Quên chia là sai.
\end{dapan}

\begin{dapan}[13 — Đáp án D]
Vì $-1 \le \sin\left(x - \dfrac{\pi}{6}\right) \le 1$ nên $-5 \le 5\sin\left(x - \dfrac{\pi}{6}\right) \le 5$. Giá trị nhỏ nhất là $-5$.
\end{dapan}

\subsection*{Nhóm B — Công thức lượng giác}

\begin{dapan}[14 — Đáp án C]
Dùng hệ thức $\sin^2 x = 1 - \cos^2 x = (1 - \cos x)(1 + \cos x)$. Thay vào:

$$\frac{\sin^2 x}{1 - \cos x} = \frac{(1 - \cos x)(1 + \cos x)}{1 - \cos x} = 1 + \cos x$$

\textbf{Cách nhanh.} Nhận ra tử số là hiệu hai bình phương và rút gọn ngay, không cần biến đổi dài.
\end{dapan}

\begin{dapan}[15 — Đáp án D]
Công thức cộng của $\sin$: $\sin(\alpha + \beta) = \sin\alpha\cos\beta + \cos\alpha\sin\beta$.
\end{dapan}

\begin{dapan}[16 — Đáp án D]
Dùng cung hơn kém $180^\circ$: $\cos 210^\circ = \cos(180^\circ + 30^\circ) = -\cos 30^\circ = -\dfrac{\sqrt{3}}{2}$.
\end{dapan}

\begin{dapan}[17 — Đáp án B]
Dùng công thức cung bù: $\sin(180^\circ - \alpha) = \sin\alpha = \dfrac{1}{2}$.
\end{dapan}

\begin{dapan}[18 — Đáp án D]
Chia cả tử và mẫu cho $\cos\alpha$:

$$\frac{\sin\alpha - \cos\alpha}{\sin\alpha + \cos\alpha} = \frac{\tan\alpha - 1}{\tan\alpha + 1} = \frac{2 - 1}{2 + 1} = \frac{1}{3}$$
\end{dapan}

\begin{dapan}[19 — Đáp án B]
Từ $\tan\alpha = \sqrt{3}$ và $\tan\dfrac{\pi}{3} = \sqrt{3}$, suy ra $\alpha = \dfrac{\pi}{3}$ (một nghiệm đại diện).
\end{dapan}

\subsection*{Nhóm C — Hàm số bậc hai, mũ, logarit}

\begin{dapan}[20 — Đáp án C]
Từ $f(16) = 2$ và $f(x) = \log_a x$ được $\log_a 16 = 2$, tức $a^2 = 16$.

Vì $a > 0$ và $a \neq 1$ nên $a = 4$.
\end{dapan}

\begin{dapan}[21 — Đáp án A]
Nguyên hàm: $f(x) = \int (2x - 1)\,dx = x^2 - x + C$.

Điều kiện $f(0) = 1$ cho $C = 1$. Vậy $f(x) = x^2 - x + 1$.
\end{dapan}

\begin{dapan}[22 — Đáp án B]
Đạo hàm từng hạng tử: $\left(x\right)' = 1$ và $\left(\dfrac{1}{2x^2}\right)' = \left(\dfrac{1}{2}x^{-2}\right)' = -x^{-3} = -\dfrac{1}{x^3}$.

Vậy $f'(x) = 1 - \dfrac{1}{x^3}$.
\end{dapan}

\begin{dapan}[23 — Đáp án B]
Từ $\log_7 x = 2$ suy ra $x = 7^2 = 49$.
\end{dapan}

\subsection*{Nhóm D — Ôn tập tích lũy}

\begin{dapan}[24 — Đáp án D]
Giải $x^2 - 5x + 6 \le 0$: hai nghiệm $x = 2$, $x = 3$, lấy trong khoảng nên $A = \{2, 3\}$.

Giải $3x - 1 > 5$ được $x > 2$. Vì $B \subseteq \mathbb{N}$ nên $B = \{3, 4, 5, \ldots\}$.

Hợp hai tập: $A \cup B = \{2, 3, 4, 5, \ldots\} = \{x \in \mathbb{N} \mid x \ge 2\}$.
\end{dapan}

\begin{dapan}[25 — Đáp án A]
Công sai: $a_5 = a_1 + 4d$ cho $11 = 3 + 4d$, suy ra $d = 2$.

Tổng $7$ số hạng: $S_7 = \dfrac{7(a_1 + a_7)}{2}$ với $a_7 = 3 + 6 \cdot 2 = 15$. Vậy $S_7 = \dfrac{7 \cdot 18}{2} = 63$.
\end{dapan}

\subsection*{Nhóm D — Ôn tập tích lũy}

\begin{dapan}[1 — Đáp án A]
$\alpha$ ở phần tư thứ ba nên $\cos\alpha < 0$. Từ $\sin^2\alpha + \cos^2\alpha = 1$:

$$\cos\alpha = -\sqrt{1 - \frac{1}{4}} = -\frac{\sqrt{3}}{2}$$
\end{dapan}

\begin{dapan}[2 — Đáp án D]
$\tan\alpha = 1$ nên $\alpha$ có $|\sin\alpha| = |\cos\alpha|$. Vì $\alpha$ ở phần tư thứ ba nên $\sin\alpha < 0$, suy ra $\sin\alpha = -\dfrac{\sqrt{2}}{2}$.
\end{dapan}

\begin{dapan}[3 — Đáp án B]
Công thức cộng: $\tan\left(\alpha + \dfrac{\pi}{4}\right) = \dfrac{\tan\alpha + 1}{1 - \tan\alpha} = \dfrac{2 + 1}{1 - 2} = -3$.
\end{dapan}

\begin{dapan}[4 — Đáp án C]
$\cos$ là hàm chẵn nên $\cos\left(-\dfrac{\pi}{6}\right) = \cos\dfrac{\pi}{6} = \dfrac{\sqrt{3}}{2}$.
\end{dapan}

\begin{dapan}[5 — Đáp án D]
Đạo hàm hàm hợp: $f'(x) = \cos\left(2x + \dfrac{\pi}{4}\right) \cdot 2 = 2\cos\left(2x + \dfrac{\pi}{4}\right)$.
\end{dapan}

\begin{dapan}[6 — Đáp án C]
Gọi $R$ là bán kính đường tròn. Cung có số đo $2$ radian chắn dây cung độ dài $2$, nên nửa dây cung bằng $1$ và nửa góc ở tâm bằng $1$ radian.

Trong tam giác vuông: $\sin 1 = \dfrac{1}{R}$, suy ra $R = \dfrac{1}{\sin 1}$. Độ dài cung là $l = 2R = \dfrac{2}{\sin 1}$.
\end{dapan}

\begin{dapan}[7 — Đáp án A]
Với $a = 1$, $b = -1$: giá trị lớn nhất của $a\sin x + b\cos x$ là $\sqrt{a^2 + b^2} = \sqrt{2}$.
\end{dapan}

\begin{dapan}[8 — Đáp án A]
Dùng công thức nhân đôi: $\sin 2x\cos 2x = \dfrac{1}{2}\sin 4x$. Hệ số $\omega = 4$ nên chu kỳ là $\dfrac{2\pi}{4} = \dfrac{\pi}{2}$.
\end{dapan}

\begin{dapan}[9 — Đáp án A]
Biểu thức $-2\cos x$ lớn nhất bằng $2$ (khi $\cos x = -1$). Vậy $y = 3 - 2\cos x$ lớn nhất bằng $3 + 2 = 5$.

\textbf{Bẫy.} Dấu trừ trước biên độ làm đảo chiều: giá trị lớn nhất đạt khi $\cos x$ \textbf{nhỏ nhất}, không phải khi $\cos x = 1$.
\end{dapan}

\begin{dapan}[10 — Đáp án B]
Tính $f(-x) = (-x)\sin(-x) = (-x)(-\sin x) = x\sin x = f(x)$. Vậy hàm số là hàm chẵn.
\end{dapan}

\begin{dapan}[11 — Đáp án B]
Trên $[0, 2\pi]$, phương trình $\sin x = \dfrac{1}{2}$ có hai nghiệm $x = \dfrac{\pi}{6}$ và $x = \dfrac{5\pi}{6}$. Số nghiệm là $2$.
\end{dapan}

\begin{dapan}[12 — Đáp án B]
Đặt $t = \sin x$ với $t \in [-1, 1]$. Hàm $y = 2t^2 - 3t + 1$ có đỉnh $t_0 = \dfrac{3}{4}$, thuộc đoạn $[-1, 1]$.

Giá trị nhỏ nhất tại $t_0 = \dfrac{3}{4}$: $y = 2 \cdot \dfrac{9}{16} - \dfrac{9}{4} + 1 = -\dfrac{1}{8}$. Giá trị lớn nhất tại $t = -1$: $y = 2 + 3 + 1 = 6$.

Tập giá trị là $\left[-\dfrac{1}{8}, 6\right]$.
\end{dapan}

\begin{dapan}[13 — Đáp án C]
Đặt $t = \cos x$ với $t \in [-1, 1]$: $2t^2 - 3t + 1 = 0$ cho $t = 1$ hoặc $t = \dfrac{1}{2}$.

Với $t = 1$: $\cos x = 1$ cho $x = 0$ và $x = 2\pi$, tức $2$ nghiệm trên $[0, 2\pi]$.

Với $t = \dfrac{1}{2}$: $\cos x = \dfrac{1}{2}$ cho $x = \dfrac{\pi}{3}$ và $x = \dfrac{5\pi}{3}$, tức $2$ nghiệm. Tổng cộng $4$ nghiệm, nhưng $x = 0$ và $x = 2\pi$ trùng nhau khi đếm trên vòng tròn nên đáp án là $3$.
\end{dapan}

\begin{dapan}[14 — Đáp án C]
Đặt $t = \sin x$ với $t \in [-1, 1]$. Hàm $y = t^2 - t + 1$ có đỉnh $t_0 = \dfrac{1}{2}$, thuộc đoạn $[-1, 1]$.

Giá trị nhỏ nhất tại $t_0 = \dfrac{1}{2}$: $y = \dfrac{1}{4} - \dfrac{1}{2} + 1 = \dfrac{3}{4}$.
\end{dapan}

\begin{dapan}[15 — Đáp án D]
Trên khoảng $(0, \pi)$, phương trình $\tan x = -1$ có nghiệm $x = \dfrac{3\pi}{4}$.

\textbf{Chú ý.} Họ nghiệm tổng quát là $x = -\dfrac{\pi}{4} + k\pi$. Với $k = 1$ được $x = \dfrac{3\pi}{4}$, nằm trong $(0, \pi)$.
\end{dapan}
