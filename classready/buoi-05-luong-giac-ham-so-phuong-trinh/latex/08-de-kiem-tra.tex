\begin{baitap}[1]

\end{baitap}

\begin{baitap}[2]

\par\vspace{3pt}
\noindent A. $\dfrac{1}{2}$ \quad B. $-\dfrac{1}{2}$ \quad C. $\dfrac{\sqrt{3}}{2}$ \quad D. $-\dfrac{\sqrt{3}}{2}$
\end{baitap}

\begin{baitap}[3]
