\section{PHẦN C — BÀI TẬP CHUẨN BỊ}

Làm 10 câu dưới đây \textbf{trước buổi học}. Sáu câu đầu chỉ cần đọc hiểu thuật ngữ là làm được; các câu cuối đòi hỏi biến đổi, nếu chưa làm được thì ghi lại chỗ vướng mắc để hỏi trên lớp.

\subsection{Mức dễ}

\begin{baitap}[1]
函数 $y = \cos x$ 的图像关于 ( ) 对称
\par\vspace{3pt}
\noindent A. y 轴 \quad B. 原点 \quad C. x 轴 \quad D. 直线 $y = x$
\end{baitap}
\begin{baitap}[2]
函数 $y = 3\sin\!\left(2x + \dfrac{\pi}{6}\right)$ 的最小正周期是（ ）
\par\vspace{3pt}
\noindent A. $\pi$ \quad B. $\dfrac{\pi}{2}$ \quad C. $2\pi$ \quad D. $4\pi$
\end{baitap}
\begin{baitap}[3]
函数 $y = \cos\!\left(\dfrac{x}{3} + \dfrac{\pi}{6}\right)$ 的最小正周期是（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{\pi}{3}$ \quad B. $6\pi$ \quad C. $2\pi$ \quad D. $3\pi$
\end{baitap}
\begin{baitap}[4]
函数 $y = \sin x + \cos x$ 的最大值是 ( )
\par\vspace{3pt}
\noindent A. 1 \quad B. $\sqrt{2}$ \quad C. 2 \quad D. $\dfrac{\sqrt{2}}{2}$
\end{baitap}
\begin{baitap}[5]
方程 $\sin x = 0$ 的解集是（ ）
\par\vspace{3pt}
\noindent A. $x = \dfrac{\pi}{2} + k\pi,\ k \in \mathbb{Z}$ \quad B. $x = 2k\pi,\ k \in \mathbb{Z}$ \quad C. $x = k\pi,\ k \in \mathbb{Z}$ \quad D. $x = \dfrac{\pi}{2} + 2k\pi,\ k \in \mathbb{Z}$
\end{baitap}
\begin{baitap}[6]
方程 $\sin x = 1$ 的解集是（ ）
\par\vspace{3pt}
\noindent A. $x = \dfrac{\pi}{2} + k\pi,\ k \in \mathbb{Z}$ \quad B. $x = \pi + 2k\pi,\ k \in \mathbb{Z}$ \quad C. $x = 2k\pi,\ k \in \mathbb{Z}$ \quad D. $x = \dfrac{\pi}{2} + 2k\pi,\ k \in \mathbb{Z}$
\end{baitap}
