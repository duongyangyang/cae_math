\subsection*{NHÓM A — TÍNH CHẤT HÀM SỐ}

\begin{baitap}[1]
判断函数 $f(x) = x\ln(|x| + 1)$ 的奇偶性是 ( )
\par\vspace{3pt}
\noindent A. 奇函数 \quad B. 偶函数 \quad C. 非奇非偶函数 \quad D. 既是奇函数又是偶函数
\end{baitap}

\begin{baitap}[2]
函数 $f(x) = \sin x$ 在区间 $[0, \pi]$ 上的最大值和最小值分别为 ( )
\par\vspace{3pt}
\noindent A. $1, -1$ \quad B. $0, -1$ \quad C. 1, 0 \quad D. 0, 1
\end{baitap}

\begin{baitap}[3]
求函数 $f(x) = \dfrac{1}{\sqrt{x^2 - 3x + 2}} + \ln(4 - x)$ 的定义域是 ( )
\par\vspace{3pt}
\noindent A. $(-\infty, 1) \cup (2,4)$ \quad B. $(1,2) \cup (4, +\infty)$ \quad C. $(-\infty, 4)$ \quad D. $(1,4)$
\end{baitap}

\begin{baitap}[4]
判断函数 $f(x) = \dfrac{\sin x}{x}$ 的奇偶性是 ( )
\par\vspace{3pt}
\noindent A. 奇函数 \quad B. 偶函数 \quad C. 非奇非偶函数 \quad D. 既是奇函数又是偶函数
\end{baitap}

\begin{baitap}[5]
函数 $y = x^3 - 27x + 3$ 的单调递增区间是 ( )
\par\vspace{3pt}
\noindent A. $(-3,3)$ \quad B. $(-3, +\infty)$ \quad C. $(3, +\infty)$ \quad D. $(-\infty, 3)$
\end{baitap}

\begin{baitap}[6]
下列各组函数中，表示同一个函数的是 ( )
\par\vspace{3pt}
\noindent A. $f(x) = x$ 与 $g(x) = \dfrac{x^2}{x}$ \quad B. $f(x) = |x|$ 与 $g(x) = \sqrt{x^2}$ \quad C. $f(x) = \sqrt{x^2 - 1}$ 与 $g(x) = \dfrac{\sqrt{x} - 1}{\sqrt{x} + 1}$ \quad D. $f(x) = 1$ 与 $g(x) = x^0$
\end{baitap}

\begin{baitap}[7]
已知函数 $f(x)$ 的定义域为 $[-2,2]$, 则函数 $g(x) = f(x - 1)$ 的定义域为 ( )
\par\vspace{3pt}
\noindent A. $[-2,2]$ \quad B. $[-3,1]$ \quad C. $[-1,3]$ \quad D. $[0,4]$
\end{baitap}

\begin{baitap}[8]
函数 $f(x) = \log_{2}(x^2 - 4x + 3)$ 的定义域是 ( )
\par\vspace{3pt}
\noindent A. $(-\infty, 1) \cup (3, +\infty)$ \quad B. $(1,3)$ \quad C. $(-\infty, 1]$ \quad D. $[3, +\infty)$
\end{baitap}

\begin{baitap}[9]
函数 $f(x) = \dfrac{1}{3x-1}$ 的值域是 ( )
\par\vspace{3pt}
\noindent A. $\mathbb{R}$ \quad B. $\left(-\infty, \dfrac{1}{3}\right) \cup \left(\dfrac{1}{3}, +\infty\right)$ \quad C. $(-\infty, 0) \cup (0, +\infty)$ \quad D. $(0, +\infty)$
\end{baitap}

\begin{baitap}[10]
若函数 $f(x)$ 是奇函数，且当 $x > 0$ 时,$f(x) = 3x^2 + 1$ ，则 $f(-2)$ 等于 ( )
\par\vspace{3pt}
\noindent A. -2 \quad B. -13 \quad C. -12 \quad D. -1
\end{baitap}

\begin{baitap}[11]

\par\vspace{3pt}
\noindent A. $2x - 1$ \quad B. $2^{x-1}$ \quad C. $\dfrac{1}{2} \times 2^x$ \quad D. B 和 C 都正确
\end{baitap}

\begin{baitap}[12]
已知函数 $f(x) = \sin\!\left(-3x + \dfrac{\pi}{3}\right)$ ，则 $f(x)$ 的最小正周期是 ( )
\par\vspace{3pt}
\noindent A. $\pi$ \quad B. $\dfrac{2\pi}{3}$ \quad C. $\dfrac{3\pi}{2}$ \quad D. $6\pi$
\end{baitap}

\begin{baitap}[13]
下列函数中，在 $\mathbb{R}$ 上为减函数的是 ( )
\par\vspace{3pt}
\noindent A. $f(x) = 2x - 1$ \quad B. $f(x) = x$ \quad C. $f(x) = e^x$ \quad D. $f(x) = 2^{-x}$
\end{baitap}

\begin{baitap}[14]
函数 $f(x) = \sqrt{3x - 1} + \dfrac{1}{x-2}$ 的定义域是 ( )
\par\vspace{3pt}
\noindent A. $\left[\dfrac{1}{3}, +\infty\right)$ \quad B. $(-\infty, 2)$ \quad C. $\left(\dfrac{1}{3}, 2\right)$ \quad D. $\left[\dfrac{1}{3}, 2\right) \cup (2, +\infty)$

\begin{baitap}[15]

\begin{baitap}[16]

\par\vspace{3pt}

\begin{baitap}[17]
函数 $f(x) = x^3 - 3x^2 + 2$ 在区间 [0,3] 上的最大值和最小值之和等于 ( )
\par\vspace{3pt}
\noindent A. 0 \quad B. 1 \quad C. 2 \quad D. -2
\end{baitap}

\begin{baitap}[18]
函数 $f(x) = \sqrt{x^2 - 8x + 12} + \dfrac{1}{\ln(x-5)}$ 的定义域是 ( )

\subsection*{NHÓM B — ÔN TẬP TẬP HỢP VÀ BẤT ĐẲNG THỨC}

\begin{baitap}[19]
设集合 $A=\{x\in\mathbb{Z}\mid |x|<2\}$，则集合 $A$ 的子集个数为（ ）
\par\vspace{3pt}
\noindent A. 2 \quad B. 4 \quad C. 6 \quad D. 8
\end{baitap}

\begin{baitap}[20]
设全集 $U=\{1,2,3,4,5\}$，集合 $A=\{x\mid x^2-4x+3=0\}$，则 $C_UA$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\{1,3\}$ \quad B. $\{2,4,5\}$ \quad C. $\{1,2,3,4,5\}$ \quad D. $\{2,5\}$
\end{baitap}

\begin{baitap}[21]
不等式 $\log_{2}(x - 1) \le 1$ 的解集是 ( )
\par\vspace{3pt}
\noindent A. $(1,3)$ \quad B. $(-\infty, 3)$ \quad C. $(1,3]$ \quad D. $(3, +\infty)$
\end{baitap}

\begin{baitap}[22]
若对于任意 $x \in \mathbb{R},$ 不等式 $x^2 - 2x + 2m > 0$ 恒成立，则实数 $m$ 的取值范围是 ( )
\par\vspace{3pt}
\noindent A. $m > \dfrac{1}{2}$ \quad B. $m < \dfrac{1}{2}$ \quad C. $m \ge \dfrac{1}{2}$ \quad D. $m \le \dfrac{1}{2}$
\end{baitap}

\begin{baitap}[23]
已知集合 $A=\{x\in\mathbb Z\mid 5<x\le18\}$，则集合 $A$ 的真子集个数是（ ）
\par\vspace{3pt}
\noindent A. $2^{12}-1$ \quad B. $2^{13}-1$ \quad C. $2^{14}-1$ \quad D. $2^{15}-1$
\end{baitap}

\begin{baitap}[24]
若 $x > 2$, 则 $x + \dfrac{1}{x-2}$ 的最小值为 ( )
\par\vspace{3pt}
\noindent A. 2 \quad B. 3 \quad C. 4 \quad D. 5
\end{baitap}

\begin{baitap}[25]
不等式 $x^3 - x^2 - 2x \ge 0$ 的解集是 ( )
\par\vspace{3pt}
\noindent A. $[-1,0] \cup [2, +\infty)$ \quad B. $(-\infty, -1] \cup [0,2]$ \quad C. $(-\infty, -1) \cup (0,2)$ \quad D. $[-2,0] \cup [1, +\infty)$
\end{baitap}

\begin{baitap}[5]

\begin{baitap}[7]
