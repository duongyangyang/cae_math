\section{PHẦN C — BÀI TẬP CHUẨN BỊ}

Làm 10 câu dưới đây \textbf{trước buổi học}. Sáu câu đầu chỉ cần đọc hiểu thuật ngữ là làm được; các câu cuối đòi hỏi biến đổi, nếu chưa làm được thì ghi lại chỗ vướng mắc để hỏi trên lớp.

\subsection{Mức dễ}

\begin{baitap}[1]
下列函数中，是偶函数的是 ( )
\par\vspace{3pt}
\noindent A. $y = x^3$ \quad B. $y = \sin x$ \quad C. $y = \cos x$ \quad D. $y = \log_{2}x$
\end{baitap}
\begin{baitap}[2]
函数 $f(x) = \dfrac{x-1}{2x^2+1}$ 的定义域是 ( )
\par\vspace{3pt}
\noindent A. $(-\infty, 1) \cup (1, +\infty)$ \quad B. $\mathbb{R}$ \quad C. $(-\infty, 0) \cup (0, +\infty)$ \quad D. $(0, +\infty)$
\end{baitap}
\begin{baitap}[3]
已知函数 $f(x) = \begin{cases} x + 1, & x \ge 0 \\ x^2, & x < 0 \end{cases}$ ，则 $f(-1) + f(5)$ 的值为 ( )
\par\vspace{3pt}
\noindent A. 4 \quad B. 5 \quad C. 7 \quad D. 25
\end{baitap}
\begin{baitap}[4]
函数 $f(x) = \sin x$ 的最小正周期是 ( )
\par\vspace{3pt}
\noindent A. $\pi$ \quad B. $4\pi$ \quad C. $\dfrac{\pi}{2}$ \quad D. $2\pi$
\end{baitap}
\begin{baitap}[5]
已知函数 $f(x)$ 是定义在 $\mathbb{R}$ 上的奇函数，当 $x > 0$ 时， $f(x) = 2x^2 + 1$ ，则 $f(-1)$ 等于 ( )
\par\vspace{3pt}
\noindent A. -3 \quad B. 1 \quad C. -1 \quad D. 3
\end{baitap}
\begin{baitap}[6]
函数 $y = 2^x$ 的值域是 ( )
\par\vspace{3pt}
\noindent A. $(0, +\infty)$ \quad B. $[0, +\infty)$ \quad C. $\mathbb{R}$ \quad D. $(-\infty, 0)$
\end{baitap}
\subsection{Mức trung bình}

\begin{baitap}[7]
已知函数 $f(x) = x^2 - 2x + m$ 在区间 $[1, +\infty)$ 上有最小值 -1，则 $m$ 的值为 ( )
\par\vspace{3pt}
\noindent A. -1 \quad B. 0 \quad C. 1 \quad D. 2
\end{baitap}
\begin{baitap}[8]
求函数 $f(x) = \sqrt{x-4} - 4x - 3 + \log_{2}(5 - x)$ 的定义域等于 ( )
\par\vspace{3pt}
\noindent A. $(-\infty, 3) \cup (3,4) \cup (4,5)$ \quad B. $(-\infty, 3) \cup [4,5)$ \quad C. $[4,5)$ \quad D. $(-\infty, 3) \cup (3,4]$
\end{baitap}
\begin{baitap}[9]
求函数 $y = 2|x-1|$ 的单调递增区间是 ( )
\par\vspace{3pt}
\noindent A. $(-\infty, 1]$ \quad B. $[1, +\infty)$ \quad C. $(0, +\infty)$ \quad D. $(-\infty, 0)$
\end{baitap}

已知定义在 $\mathbb{R}$ 上的偶函数 $f(x)$ 在区间 $(-\infty, 0)$ 上递减。若 $a = f(2^{0.7})$，$b = f(-\ln 2)$，$c = f(\log_3 2)$，则 $a, b, c$ 的大小关系为（ ）
\par\vspace{3pt}
\noindent A. $c < a < b$ \quad B. $c < b < a$ \quad C. $b < a < c$ \quad D. $b < c < a$
