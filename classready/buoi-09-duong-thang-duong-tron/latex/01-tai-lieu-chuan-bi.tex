\section{PHẦN C — BÀI TẬP CHUẨN BỊ}

Làm 10 câu dưới đây \textbf{trước buổi học}. Sáu câu đầu chỉ cần đọc hiểu thuật ngữ là làm được; các câu cuối đòi hỏi biến đổi, nếu chưa làm được thì ghi lại chỗ vướng mắc để hỏi trên lớp.

\subsection{Mức dễ}

\begin{baitap}[1]
直线 $y = 5x + 4$ 的斜率是（ ）
\par\vspace{3pt}
\noindent A. $1$ \quad B. $5$ \quad C. $-1$ \quad D. $-5$
\end{baitap}
\begin{baitap}[2]
经过点 $(1, -2)$ 且斜率为 $3$ 的直线方程是（ ）
\par\vspace{3pt}
\noindent A. $y = 3x + 5$ \quad B. $y = 3x + 1$ \quad C. $y = 3x - 1$ \quad D. $y = 3x - 5$
\end{baitap}
\begin{baitap}[3]
在平面直角坐标系中，点 $P(2,-1)$ 到直线 $3x - 4y + 5 = 0$ 的距离是（ ）
\par\vspace{3pt}
\noindent A. $3$ \quad B. $5$ \quad C. $\dfrac{6}{5}$ \quad D. $\dfrac{13}{5}$
\end{baitap}
\begin{baitap}[4]
下面四个直线方程中，可以看作是直线的斜截式方程的是（ ）
\par\vspace{3pt}
\noindent A. $x = 3$ \quad B. $y = -5$ \quad C. $2y = x$ \quad D. $x = 4y - 1$
\end{baitap}
\begin{baitap}[5]
直线 $2x - 3y + 6 = 0$ 在 $x$ 轴上的截距是（ ）
\par\vspace{3pt}
\noindent A. $-3$ \quad B. $3$ \quad C. $2$ \quad D. $-2$
\end{baitap}
\begin{baitap}[6]
直线 $L_1: y = -3x + 1$ 与直线 $L_2: y = -3x - 3$ 的位置关系是（ ）
\par\vspace{3pt}
\noindent A. 相交 \quad B. 垂直 \quad C. 平行 \quad D. 重合
\end{baitap}
\subsection{Mức trung bình}

\begin{baitap}[7]
直线 $y = \sqrt{3}x + 1$ 的倾斜角是（ ）
\par\vspace{3pt}
\noindent A. $30^\circ$ \quad B. $45^\circ$ \quad C. $60^\circ$ \quad D. $120^\circ$
\end{baitap}
\begin{baitap}[8]
若直线 $L_1: y = 2x + 1$ 和直线 $L_2: y = kx - 3$ 互相平行，则 $k$ 的值为（ ）
\par\vspace{3pt}
\noindent A. $-\dfrac{1}{2}$ \quad B. $-2$ \quad C. $\dfrac{1}{2}$ \quad D. $2$
\end{baitap}
\begin{baitap}[9]
圆 $x^2 + y^2 - 4x + 6y - 3 = 0$ 的半径是（ ）
\par\vspace{3pt}
\noindent A. $\sqrt{10}$ \quad B. $\sqrt{13}$ \quad C. $\sqrt{61}$ \quad D. $4$
\end{baitap}

\noindent A. $x + y - 5 = 0$ \quad B. $x - y + 1 = 0$ \quad C. $x + y - 4 = 0$ \quad D. $x - y - 1 = 0$
