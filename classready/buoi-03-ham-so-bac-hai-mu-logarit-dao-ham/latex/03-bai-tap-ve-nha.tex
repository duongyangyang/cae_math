\section{Bài tập bắt buộc}

Làm toàn bộ 25 câu. Đây là phần tính vào điểm bài tập về nhà.

\subsection*{NHÓM A — HÀM BẬC HAI, MŨ, LOGARIT, ĐẠO HÀM}

\begin{baitap}[1]
已知函数 $f(x) = a^{3x}\ (a > 0, a \ne 1)$ 。若 $f(x)$ 在 $\mathbb{R}$ 上是单调递减函数, 且 $f(2) = \dfrac{1}{64}$, 则 $a$ 的值是（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{1}{2}$ \quad B. 2 \quad C. $\dfrac{1}{4}$ \quad D. 4
\end{baitap}

\begin{baitap}[2]
设函数 $f(x) = \log_a x\ (a > 0, a \ne 1)$ ，若 $f(4) = 2$ ，则 $f(8)$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{3}{2}$ \quad B. 2 \quad C. 3 \quad D. 4
\end{baitap}

\begin{baitap}[3]
已知 $\log_2 3 = a$ ，则 $\log_2 9$ 等于（ ）
\par\vspace{3pt}
\noindent A. $a^2$ \quad B. $2a$ \quad C. $\dfrac{a}{2}$ \quad D. $a + 1$
\end{baitap}

\begin{baitap}[4]
求函数 $f(x) = \dfrac{5x^2 - 1}{x}$ 的导数 $f'(x)$ 等于（ ）
\par\vspace{3pt}
\noindent A. 5 \quad B. $5 - \dfrac{1}{x^2}$ \quad C. $5 + \dfrac{2}{x^2}$ \quad D. $5 + \dfrac{1}{x^2}$
\end{baitap}

\begin{baitap}[5]
已知函数 $f(x) = x^3 - 3x + 1$ 。求曲线 $y = f(x)$ 在点 $(1, f(1))$ 处的切线方程是（ ）
\par\vspace{3pt}
\noindent A. $y = -1$ \quad B. $y = -2x + 1$ \quad C. $y = 0$ \quad D. $y = 2x - 1$
\end{baitap}

\begin{baitap}[6]
设函数 $f(x) = (x - c)e^x$ ，若 $f'(c) = 15$ ，则 $c$ 的值是（ ）
\par\vspace{3pt}
\noindent A. 15 \quad B. $e^{15}$ \quad C. $\ln 15$ \quad D. $15e$
\end{baitap}

\begin{baitap}[7]
函数 $f(x) = x^3 - 3x + 2$ 的极小值是（ ）
\par\vspace{3pt}
\noindent A. -1 \quad B. 0 \quad C. 1 \quad D. 4
\end{baitap}

\begin{baitap}[8]
下列函数中，在区间 (5,10) 上是减函数的是（ ）
\par\vspace{3pt}
\noindent A. $f(x) = x^2 - 12x + 30$ \quad B. $f(x) = |x - 8|$ \quad C. $f(x) = 2^{x-10}$ \quad D. $f(x) = \log_{0.5}(x - 4)$
\end{baitap}

\begin{baitap}[9]
函数 $y = 2^x$ 的值域是（ ）
\par\vspace{3pt}
\noindent A. $(0, +\infty)$ \quad B. $[0, +\infty)$ \quad C. $\mathbb{R}$ \quad D. $(-\infty, 0)$
\end{baitap}

\begin{baitap}[10]
函数 $f(x) = \left(\dfrac{1}{2}\right)^{x-10}$ 的值域是（ ）
\par\vspace{3pt}
\noindent A. $(-\infty, 0)$ \quad B. $(0, +\infty)$ \quad C. $[0, +\infty)$ \quad D. $\mathbb{R}$
\end{baitap}

\begin{baitap}[11]
若 $a^x < a^y$ ，则当 $0 < a < 1$ 时，（ ）
\par\vspace{3pt}
\noindent A. $x < y$ \quad B. $x > y$ \quad C. $x = y$ \quad D. 无法判断
\end{baitap}

\begin{baitap}[12]
若函数 $f(x) = \dfrac{1}{3}x^3 + ax^2 + bx$ 在 $x = 5$ 和 $x = 10$ 处取得极值，则 $a + b$ 的值是（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{85}{2}$ \quad B. 50 \quad C. $\dfrac{65}{2}$ \quad D. 40
\end{baitap}

\begin{baitap}[13]
函数 $y = ax^3 - x$ 在 $(-\infty, +\infty)$ 上是减函数，则（ ）
\par\vspace{3pt}
\noindent A. $a = \dfrac{1}{3}$ \quad B. $a = 1$ \quad C. $a = 2$ \quad D. $a \le 0$
\end{baitap}

\subsection*{NHÓM B — TÍNH CHẤT HÀM SỐ}

\begin{baitap}[14]
函数 $f(x) = \sqrt{3x - 1} + \dfrac{1}{x-2}$ 的定义域是（ ）
\par\vspace{3pt}
\noindent A. $\left[\dfrac{1}{3}, +\infty\right)$ \quad B. $(-\infty, 2)$ \quad C. $\left(\dfrac{1}{3}, 2\right)$ \quad D. $\left[\dfrac{1}{3}, 2\right) \cup (2, +\infty)$
\end{baitap}

\begin{baitap}[15]
下列函数中，是偶函数的是（ ）
\par\vspace{3pt}
\noindent A. $y = x^3$ \quad B. $y = \sin x$ \quad C. $y = \cos x$ \quad D. $y = \log_{2}x$
\end{baitap}

\begin{baitap}[16]
已知函数 $f(x)$ 的定义域为 $[-2,2]$, 则函数 $g(x) = f(x - 1)$ 的定义域为（ ）
\par\vspace{3pt}
\noindent A. $[-2,2]$ \quad B. $[-3,1]$ \quad C. $[-1,3]$ \quad D. $[0,4]$
\end{baitap}

\begin{baitap}[17]
函数 $f(x) = \sqrt{\log_2 x}$ 的定义域是（ ）
\par\vspace{3pt}
\noindent A. $(0, +\infty)$ \quad B. $(1, +\infty)$ \quad C. $[1, +\infty)$ \quad D. $[2, +\infty)$
\end{baitap}

\begin{baitap}[18]
已知函数 $f(x) = \begin{cases} x + 5, & x < 7 \\ x^2 - 3x + 3, & x \ge 7 \end{cases}$ ，则 $f(f(6))$ 的值是（ ）
\par\vspace{3pt}
\noindent A. 91 \quad B. 11 \quad C. 26 \quad D. 21
\end{baitap}

\begin{baitap}[19]
函数 $y = x^3$ 在 $\mathbb{R}$ 上是（ ）
\par\vspace{3pt}
\noindent A. 奇函数且增函数 \quad B. 偶函数且增函数 \quad C. 奇函数且减函数 \quad D. 偶函数且减函数
\end{baitap}

\subsection*{NHÓM C — TẬP HỢP VÀ BẤT ĐẲNG THỨC}

\begin{baitap}[20]
设全集 $U=\{1,2,3,4,5\}$，集合 $A=\{x\mid x^2-4x+3=0\}$，则 $C_UA$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\{1,3\}$ \quad B. $\{2,4,5\}$ \quad C. $\{1,2,3,4,5\}$ \quad D. $\{2,5\}$
\end{baitap}

\begin{baitap}[21]
不等式 $x^2 - 12x + 20 \ge 0$ 的解集是（ ）
\par\vspace{3pt}
\noindent A. $[2,10]$ \quad B. $(-\infty, -10] \cup [-2, +\infty)$ \quad C. $(-\infty, 2] \cup [10, +\infty)$ \quad D. $(-10, -2)$
\end{baitap}

\begin{baitap}[22]
已知集合 $A=\{x\in\mathbb Z\mid 5<x\le18\}$，则集合 $A$ 的真子集个数是（ ）
\par\vspace{3pt}
\noindent A. $2^{12}-1$ \quad B. $2^{13}-1$ \quad C. $2^{14}-1$ \quad D. $2^{15}-1$
\end{baitap}

\begin{baitap}[23]
不等式 $\dfrac{|x-1|}{x+5} \ge 1$ 的解集是（ ）
\par\vspace{3pt}
\noindent A. $[-5,2]$ \quad B. $(-5, +\infty)$ \quad C. $(-\infty, -5) \cup [2, +\infty)$ \quad D. $(-5, -2]$
\end{baitap}

\section{Bài tập tự chọn}

Phần này luyện cùng nội dung nhưng ở mức khó hơn rõ rệt. Chỉ làm sau khi đã hoàn thành 25 câu bắt buộc. Không tính vào điểm bài tập về nhà.

\begin{baitap}[1]
已知定义在 $\mathbb{R}$ 上的偶函数 $f(x)$ 在区间 $(-\infty, 0)$ 上递减。若 $a = f(2^{0.7}), b = f(-\ln 2), c = f(\log_3 2)$，则 $a, b, c$ 的大小关系为（ ）
\par\vspace{3pt}
\noindent A. $c < a < b$ \quad B. $c < b < a$ \quad C. $b < a < c$ \quad D. $b < c < a$
\end{baitap}

\begin{baitap}[2]
若函数 $f(x) = \dfrac{1}{3}x^3 - ax^2 + x + 1$ 在 $[-3a, 3a] \ (a > 0)$ 上的最大值与最小值之和不小于 $-\dfrac{1}{4}$，则实数 $a$ 的取值范围为（ ）
\par\vspace{3pt}
\noindent A. $(0, \dfrac{1}{3}]$ \quad B. $(0, \dfrac{1}{2}]$ \quad C. $(0, 1]$ \quad D. $(0, 2]$
\end{baitap}

\begin{baitap}[3]
函数 $f(x) = \dfrac{1}{3x-1}$ 的值域是（ ）
\par\vspace{3pt}
\noindent A. $\mathbb{R}$ \quad B. $\left(-\infty, \dfrac{1}{3}\right) \cup \left(\dfrac{1}{3}, +\infty\right)$ \quad C. $(-\infty, 0) \cup (0, +\infty)$ \quad D. $(0, +\infty)$
\end{baitap}

\begin{baitap}[4]
已知函数 $f(x) = (x+1)^2 + \cos(x+1) + a$ 的最小值是4，则 $a =$（ ）
\par\vspace{3pt}
\noindent A. 3 \quad B. 4 \quad C. 5 \quad D. 6
\end{baitap}

\begin{baitap}[5]
若直线 $y = ax + b$ 是函数 $f(x) = \ln x - \dfrac{1}{x}$ 图像的切线，则 $a + b$ 的最小值为（ ）
\par\vspace{3pt}
\noindent A. -1 \quad B. -2 \quad C. -e \quad D. e
\end{baitap}

\begin{baitap}[6]
求函数 $f(x) = \dfrac{1}{\sqrt{x^2 - 3x + 2}} + \ln(4 - x)$ 的定义域是（ ）
\par\vspace{3pt}
\noindent A. $(-\infty, 1) \cup (2,4)$ \quad B. $(1,2) \cup (4, +\infty)$ \quad C. $(-\infty, 4)$ \quad D. $(1,4)$
\end{baitap}

\begin{baitap}[7]
函数 $f(x) = x^3 - 12x + 3$ 的极大值是（ ）
\par\vspace{3pt}
\noindent A. -13 \quad B. -12 \quad C. 0 \quad D. 19
\end{baitap}

\begin{baitap}[8]
下列函数中，在区间 $(0, +\infty)$ 上是增函数的是（ ）
\par\vspace{3pt}
\noindent A. $y = \dfrac{1}{x}$ \quad B. $y = -x^2$ \quad C. $y = \sqrt{x}$ \quad D. $y = \cos x$
\end{baitap}

\begin{baitap}[9]
函数 $f(x)$ 在点 $x_0$ 处的导数 $f'(x_0)$ 表示（ ）
\par\vspace{3pt}
\noindent A. 函数值 \quad B. 曲线 $y = f(x)$ 在点 $(x_0, f(x_0))$ 处的切线斜率 \quad C. 函数的平均变化率 \quad D. 函数的定义域
\end{baitap}

\begin{baitap}[10]
函数 $y = x^2 + \dfrac{2}{x}$ 的单调递增区间为（ ）
\par\vspace{3pt}
\noindent A. $(-\infty, 7)$ \quad B. $(\sqrt{2}, +\infty)$ \quad C. $(1, +\infty)$ \quad D. $(-\infty, 0)$
\end{baitap}

\begin{baitap}[11]
若函数 $f(x)$ 在 $x_0$ 处可导，且 $\lim_{\Delta x \to 0} \dfrac{f(x_0 + 2\Delta x) - f(x_0)}{2\Delta x} = 1$ 则 $f'(x_0) =$（ ）
\par\vspace{3pt}
\noindent A. 1 \quad B. -1 \quad C. 2 \quad D. $\dfrac{1}{2}$
\end{baitap}

\begin{baitap}[12]
一笔本金为 10000 元的存款，年利率为 3\%，按复利计算，则 $n$ 年后的本息和 $A(n)$ 的函数表达式是（ ）
\par\vspace{3pt}
\noindent A. $A(n) = 10000 \times (1 + 0.03n)$ \quad B. $A(n) = 10000 \times (0.03)n$ \quad C. $A(n) = 10000 + 0.03n$ \quad D. $A(n) = 10000 \times (1 + 0.03)^n$
\end{baitap}

\begin{baitap}[13]
某种细菌在培养过程中，每小时分裂一次，分裂后数量是原来的 2 倍。如果最初有 100 个细菌，则 $t$ 小时后细菌的数量 $N(t)$ 的函数表达式是（ ）
\par\vspace{3pt}
\noindent A. $N(t) = 100t$ \quad B. $N(t) = 100 + 2^t$ \quad C. $N(t) = 100t^2$ \quad D. $N(t) = 100 \times 2^t$
\end{baitap}

\begin{baitap}[14]
若 $f(x) = 3x^3 + 5x + a - 1$ 为奇函数，则 $a$ 的值为（ ）
\par\vspace{3pt}
\noindent A. 0 \quad B. -1 \quad C. 1 \quad D. 2
\end{baitap}

\begin{baitap}[15]
若 $f''(x) < 0$, 则函数 $f(x)$ 的图像是（ ）
\par\vspace{3pt}
\noindent A. 凹的 \quad B. 凸的 \quad C. 直线 \quad D. 不确定
\end{baitap}
