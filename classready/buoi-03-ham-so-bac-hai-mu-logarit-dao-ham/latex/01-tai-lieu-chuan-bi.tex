\section{PHẦN C — BÀI TẬP CHUẨN BỊ}

Làm 10 câu dưới đây \textbf{trước buổi học}. Sáu câu đầu chỉ cần đọc hiểu thuật ngữ là làm được; các câu cuối đòi hỏi biến đổi, nếu chưa làm được thì ghi lại chỗ vướng mắc để hỏi trên lớp.

\subsection{Mức dễ}

\begin{baitap}[1]
已知 $\log_a x = 3$ ，则 $x$ 等于（ ）
\par\vspace{3pt}
\noindent A. $a^3$ \quad B. $3a$ \quad C. $\dfrac{3}{a}$ \quad D. $\sqrt[3]{a}$
\end{baitap}
\begin{baitap}[2]
$2^{\log_2 8}$ 等于（ ）
\par\vspace{3pt}
\noindent A. 2 \quad B. 3 \quad C. 8 \quad D. $2^{\log_2 8}$
\end{baitap}
\begin{baitap}[3]
函数 $f(x) = \ln x$ 的导数 $f'(x)$ 是（ ）
\par\vspace{3pt}
\noindent A. $x$ \quad B. $e^x$ \quad C. $\dfrac{1}{x}$ \quad D. 1
\end{baitap}
\begin{baitap}[4]
下列幂函数中，是偶函数的是（ ）
\par\vspace{3pt}
\noindent A. $y = x^{\frac{1}{2}}$ \quad B. $y = x^2$ \quad C. $y = x^3$ \quad D. $y = x-1$
\end{baitap}
\begin{baitap}[5]
比较大小：$\log_{0.5} 0.2$ 与 $\log_{0.5} 0.4$ （ ）
\par\vspace{3pt}
\noindent A. $\log_{0.5} 0.2 > \log_{0.5} 0.4$ \quad B. $\log_{0.5} 0.2 < \log_{0.5} 0.4$ \quad C. $\log_{0.5} 0.2 = \log_{0.5} 0.4$ \quad D. 无法比较
\end{baitap}
\begin{baitap}[6]
设函数 $f(x) = x^2 - 10x + 10$ 在区间 [-10,10] 上的最小值为（ ）
\par\vspace{3pt}
\noindent A. 15 \quad B. 10 \quad C. $2^{10}$ \quad D. -15
\end{baitap}
\subsection{Mức trung bình}

\par\vspace{3pt}
\noindent A. -1 \quad B. 0 \quad C. 1 \quad D. 2
\end{baitap}
\begin{baitap}[8]
已知 $f(2^x) = x$，则 $f(3) =$（ ）
\par\vspace{3pt}

\end{baitap}
