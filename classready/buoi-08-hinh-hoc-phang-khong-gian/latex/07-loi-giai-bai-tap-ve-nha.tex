\subsection*{Nhóm A — Hình học phẳng và không gian}

\begin{dapan}[1 — Đáp án D]
\textbf{Giải.} Đáy là hình vuông cạnh $a'$. Gọi $d$ là đường chéo của hình vuông đáy. Hình hộp đứng có đường chéo khối $D = \sqrt{d^2 + h^2}$ với $h$ là chiều cao.

Đề cho đường chéo khối $D = 5a$ và chiều cao $h = 4a$:

$$d^2 = D^2 - h^2 = 25a^2 - 16a^2 = 9a^2 \implies d = 3a$$

Đường chéo hình vuông bằng $a'\sqrt{2}$, suy ra $a'\sqrt{2} = 3a$, tức $a' = \dfrac{3a}{\sqrt{2}}$. Diện tích đáy:

$$S = a'^2 = \frac{9a^2}{2}$$

Thể tích:

$$V = S \cdot h = \frac{9a^2}{2} \cdot 4a = 18a^3$$

\textbf{Bẫy.} Đáp án B và C là số trần, mất đơn vị $a^3$. Đáp án A là $9a^3$, tức quên nhân chiều cao hoặc nhầm đường chéo đáy bằng đường chéo khối.
\end{dapan}

\begin{dapan}[2 — Đáp án D]
\textbf{Giải.} Xét từng mệnh đề:

\begin{itemize}[leftmargin=1.4em, itemsep=2pt]
\item \textbf{A.} Ba điểm \textbf{thẳng hàng} không xác định được mặt phẳng nào. Thiếu điều kiện không thẳng hàng, nên sai.
\item \textbf{B.} Một đường thẳng và một điểm \textbf{thuộc đường thẳng đó} không xác định được mặt phẳng duy nhất. Sai.
\item \textbf{C.} Hai đường thẳng chéo nhau không đồng phẳng, không xác định được mặt phẳng chứa cả hai. Sai.
\item \textbf{D.} Hai đường thẳng cắt nhau xác định đúng một mặt phẳng. (đúng)
\end{itemize}


\textbf{Chú ý.} Bốn cách xác định mặt phẳng: ba điểm không thẳng hàng; một đường thẳng và một điểm nằm ngoài đường thẳng đó; hai đường thẳng cắt nhau; hai đường thẳng song song.
\end{dapan}

\begin{dapan}[3 — Đáp án B]
\textbf{Giải.} Hình trụ có diện tích xung quanh $S_{\text{xq}} = 2\pi r l$ với $r$ là bán kính đáy và $l$ là độ dài đường sinh. Với hình trụ, đường sinh bằng chiều cao.

Đề cho $r = 2a$, $l = 4a$:

$$S\_{\text{xq}} = 2\pi \cdot 2a \cdot 4a = 16a^2\pi$$

\textbf{Bẫy.} Đáp án D là $8a^2\pi$, tức quên hệ số $2$ trong công thức. Đáp án A là $20a^2\pi$, tức cộng nhầm đáy.
\end{dapan}

\begin{dapan}[4 — Đáp án B]
\textbf{Giải.} Phương trình mặt cầu tâm $I(a;b;c)$ bán kính $R$:

$$(x-a)^2 + (y-b)^2 + (z-c)^2 = R^2$$

Đề cho tâm $I(1;-4;0)$ nên $a = 1$, $b = -4$, $c = 0$. Đường kính bằng $6$ nên bán kính $R = 3$, suy ra $R^2 = 9$.

Phương trình mặt cầu:

$$(x-1)^2 + (y+4)^2 + z^2 = 9$$

\textbf{Bẫy.} Đáp án C và D dùng $36$ là bình phương \textbf{đường kính} thay vì bình phương bán kính. Đáp án A và D sai dấu ở tâm: tâm $(1;-4;0)$ cho ra $(x-1)^2 + (y+4)^2$, không phải $(x+1)^2 + (y-4)^2$.
\end{dapan}

\begin{dapan}[5 — Đáp án A]
\textbf{Giải.} Hình nón có bán kính đáy $r = 3$ và đường sinh $l = 5$. Tính chiều cao:

$$h = \sqrt{l^2 - r^2} = \sqrt{25 - 9} = 4$$

Thể tích:

$$V = \frac{1}{3}\pi r^2 h = \frac{1}{3}\pi \cdot 9 \cdot 4 = 12\pi$$

\textbf{Bẫy.} Đáp án B là $15\pi$, tức dùng nhầm đường sinh $l = 5$ làm chiều cao. Đây là lỗi phổ biến nhất của dạng bài này.
\end{dapan}

\begin{dapan}[6 — Đáp án B]
\textbf{Giải.} Diện tích đáy $9\pi$ cho $\pi r^2 = 9\pi$, suy ra $r = 3$. Chiều cao $h = 4$.

$$V = \frac{1}{3}\pi r^2 h = \frac{1}{3} \cdot 9\pi \cdot 4 = 12\pi$$

\textbf{Bẫy.} Đáp án C là $36\pi$, tức quên hệ số $\dfrac{1}{3}$ và tính $\pi r^2 h$.
\end{dapan}

\begin{dapan}[7 — Đáp án C]
\textbf{Giải.} Ba điểm $A(1,0,0)$, $B(0,1,0)$, $C(0,0,1)$ có ba cạnh bằng nhau:

$$AB = BC = CA = \sqrt{(1-0)^2 + (0-1)^2 + 0} = \sqrt{2}$$

Vậy tam giác $ABC$ đều cạnh $\sqrt{2}$. Diện tích tam giác đều cạnh $a'$ là $\dfrac{a'^2\sqrt{3}}{4}$:

$$S = \frac{(\sqrt{2})^2 \cdot \sqrt{3}}{4} = \frac{2\sqrt{3}}{4} = \frac{\sqrt{3}}{2}$$

\textbf{Cách nhanh.} Ba điểm nằm trên ba trục tọa độ, cách gốc một đơn vị, luôn tạo tam giác đều cạnh $\sqrt{2}$. Nhận ra điều này thì khỏi kiểm tra từng cạnh.
\end{dapan}

\begin{dapan}[8 — Đáp án D]
\textbf{Giải.} Hình chóp tam giác đều $P-ABC$ có đáy là tam giác đều cạnh $AB = a'$ và các cạnh bên bằng nhau $PA = PB = PC = l$. Gọi $H$ là trung điểm $AB$.

Diện tích đáy: $S_{\text{đáy}} = \dfrac{a'^2\sqrt{3}}{4}$.

Mỗi mặt bên là tam giác cân có đáy $a'$ và hai cạnh bên $l$. Đường cao mặt bên $PH = \sqrt{l^2 - \dfrac{a'^2}{4}}$. Diện tích một mặt bên:

$$S\_{\text{bên}} = \frac{1}{2} \cdot a' \cdot \sqrt{l^2 - \frac{a'^2}{4}}$$

Đề cho tổng diện tích mặt bằng $5$ lần diện tích đáy:

$$S\_{\text{đáy}} + 3S\_{\text{bên}} = 5S\_{\text{đáy}} \implies 3S\_{\text{bên}} = 4S\_{\text{đáy}}$$

Thay vào:

$$3 \cdot \frac{a'}{2}\sqrt{l^2 - \frac{a'^2}{4}} = 4 \cdot \frac{a'^2\sqrt{3}}{4}$$

Chia cả hai vế cho $\dfrac{a'}{2}$:

$$3\sqrt{l^2 - \frac{a'^2}{4}} = 2a'\sqrt{3}$$

Bình phương hai vế:

$$9\left(l^2 - \frac{a'^2}{4}\right) = 12a'^2 \implies 9l^2 = 12a'^2 + \frac{9a'^2}{4} = \frac{57a'^2}{4}$$

Suy ra:

$$\frac{l^2}{a'^2} = \frac{57}{36} \implies \frac{l}{a'} = \frac{\sqrt{57}}{6}$$

Vậy $\dfrac{PA}{AB} = \dfrac{\sqrt{57}}{6}$.
\end{dapan}

\begin{dapan}[9 — Đáp án C]
\textbf{Giải.} Mặt xung quanh hình nón trải ra thành hình quạt có bán kính bằng đường sinh $l = 5$ và diện tích bằng diện tích xung quanh $20\pi$.

Gọi $\alpha$ là góc ở tâm của hình quạt. Diện tích hình quạt là $\dfrac{1}{2}l^2\alpha$:

$$\frac{1}{2} \cdot 25 \cdot \alpha = 20\pi \implies \alpha = \frac{8\pi}{5}$$

Chu vi cung của hình quạt bằng chu vi đáy nón: $l\alpha = 2\pi r$:

$$5 \cdot \frac{8\pi}{5} = 2\pi r \implies r = 4$$

Chiều cao nón $h = \sqrt{l^2 - r^2} = \sqrt{25-16} = 3$.

Mặt cắt qua hai đường sinh là tam giác có hai cạnh bằng $l = 5$ và góc xen giữa $\theta$ thay đổi trong khoảng $(0, \alpha]$. Diện tích tam giác:

$$S = \frac{1}{2}l^2\sin\theta = \frac{25}{2}\sin\theta$$

Diện tích lớn nhất khi $\sin\theta$ lớn nhất. Vì $\alpha = \dfrac{8\pi}{5} = 288^\circ > 90^\circ$, góc $\theta$ đạt được giá trị $90^\circ$ nên $\sin\theta$ đạt $1$:

$$S\_{\max} = \frac{25}{2}$$

\textbf{Chú ý.} Phải kiểm tra $\alpha$ có vượt quá $90^\circ$ hay không. Nếu $\alpha < 90^\circ$ thì $\sin\theta$ lớn nhất tại $\theta = \alpha$, đáp án sẽ khác.
\end{dapan}

\begin{dapan}[10 — Đáp án A]
\textbf{Giải.} Hình lập phương $ABCD.A'B'C'D'$ có cạnh $a' = a\sqrt{2}$. Đoạn $A'C$ nối đỉnh $A'$ của mặt trên với đỉnh $C$ của mặt dưới, đây là \textbf{đường chéo khối}:

$$A'C = a'\sqrt{3} = a\sqrt{2} \cdot \sqrt{3} = a\sqrt{6}$$

\textbf{Bẫy.} Đáp án D là $a\sqrt{3}$, tức lấy cạnh bằng $a$ thay vì $a\sqrt{2}$. Đáp án C là $a\sqrt{2}$, tức lấy đường chéo mặt.
\end{dapan}

\begin{dapan}[11 — Đáp án C]
\textbf{Giải.} Qua một điểm $P$ cho trước có vô số mặt phẳng. Mỗi đường thẳng qua $P$ cùng với $P$ xác định một mặt phẳng, mà có vô số đường thẳng qua $P$ trong không gian.

\textbf{Nhận xét.} Đây là câu hỏi lý thuyết thuần túy, không cần tính toán. Đọc được 无数个 (vô số) là chọn được đáp án.
\end{dapan}

\begin{dapan}[12 — Đáp án C]
\textbf{Giải.} Khoảng cách từ điểm $M(x_0;y_0;z_0)$ đến mặt phẳng $Ax + By + Cz + D = 0$:

$$d = \frac{|Ax\_0 + By\_0 + Cz\_0 + D|}{\sqrt{A^2+B^2+C^2}}$$

Với $M = O(0;0;0)$ và mặt phẳng $2x - y + z - 3 = 0$:

$$d = \frac{|2 \cdot 0 - 0 + 0 - 3|}{\sqrt{4+1+1}} = \frac{3}{\sqrt{6}} = \frac{3\sqrt{6}}{6} = \frac{\sqrt{6}}{2}$$

\textbf{Bẫy.} Đáp án B là $-\dfrac{\sqrt{6}}{2}$: khoảng cách luôn không âm nên không thể mang dấu trừ. Đáp án A là $\dfrac{3}{4}$, tức chia nhầm cho $4$ thay vì $\sqrt{6}$.
\end{dapan}

\subsection*{Nhóm B — Dãy số}

\begin{dapan}[13 — Đáp án A]
\textbf{Giải.} Cấp số nhân có $b_1 = 1$, $b_5 = 16$. Công thức số hạng tổng quát $b_n = b_1 q^{n-1}$:

$$b\_5 = 1 \cdot q^4 = 16 \implies q^2 = 4 \implies q = \pm 2$$

Với $q = 2$: $b_3 = q^2 = 4$, suy ra $b_1 + b_3 = 1 + 4 = 5$.

Với $q = -2$: $b_3 = q^2 = 4$, cũng cho $b_1 + b_3 = 5$.

\textbf{Nhận xét.} Đáp án không phụ thuộc dấu của $q$ vì $b_3$ chứa $q^2$. Đây là lý do đề không cần cho biết $q$ dương hay âm.
\end{dapan}

\begin{dapan}[14 — Đáp án D]
\textbf{Giải.} Dãy cho bởi $a_1 = 1$ và $a_{n+1} = a_n + 2n$. Tính lần lượt:

$$a\_2 = a\_1 + 2 \cdot 1 = 1 + 2 = 3$$

$$a\_3 = a\_2 + 2 \cdot 2 = 3 + 4 = 7$$

\textbf{Bẫy.} Đáp án B là $3$, tức dừng ở $a_2$. Đáp án C là $5$, tức cộng nhầm $2 \cdot 2 = 4$ thành $2$.
\end{dapan}

\begin{dapan}[15 — Đáp án C]
\textbf{Giải.} Cấp số cộng có $a_1 = 1$, $d = 3$. Tổng $n$ số hạng đầu:

$$S\_n = \frac{n(a\_1 + a\_n)}{2} = \frac{n\big(2a\_1 + (n-1)d\big)}{2}$$

Thay số:

$$S\_n = \frac{n\big(2 + 3(n-1)\big)}{2} = \frac{n(3n - 1)}{2} = \frac{3n^2 - n}{2}$$

\textbf{Bẫy.} Đáp án B là $\dfrac{3n^2+n}{2}$, tức dùng nhầm $a_1 + a_n = 1 + 3n$ thay vì $3n - 1$.
\end{dapan}

\begin{dapan}[16 — Đáp án D]
\textbf{Giải.} Từ $S_n = n^2 + 4n$, số hạng thứ $n$ là $a_n = S_n - S_{n-1}$ với $n \ge 2$:

$$S\_3 = 9 + 12 = 21, \qquad S\_2 = 4 + 8 = 12$$

$$a\_3 = S\_3 - S\_2 = 21 - 12 = 9$$

\textbf{Cách nhanh.} Với $S_n = an^2 + bn$, số hạng tổng quát là $a_n = 2an + (b - a)$. Ở đây $a = 1$, $b = 4$ nên $a_n = 2n + 3$, suy ra $a_3 = 9$.
\end{dapan}

\begin{dapan}[17 — Đáp án C]
\textbf{Giải.} Cấp số cộng có $a_5 = 15$ và $a_7 = 19$. Hiệu hai số hạng cách nhau hai bậc:

$$a\_7 - a\_5 = 2d \implies 19 - 15 = 2d \implies d = 2$$

Từ $a_3 = a_5 - 2d = 15 - 4 = 11$.

\textbf{Cách nhanh.} Trong cấp số cộng, $a_3$, $a_5$, $a_7$ lập thành cấp số cộng nên $a_5$ là trung bình cộng của $a_3$ và $a_7$: $a_3 = 2a_5 - a_7 = 30 - 19 = 11$. Không cần tìm $d$.
\end{dapan}

\begin{dapan}[18 — Đáp án B]
\textbf{Giải.} Trong cấp số cộng, $a_m + a_n = a_p + a_q$ khi $m + n = p + q$.

Vế trái: $a_3 + a_6 + a_8 + a_{11} = (a_3 + a_{11}) + (a_6 + a_8)$.

Với $a_3 + a_{11}$: chỉ số $3 + 11 = 14$, bằng $2 \cdot 7$ nên $a_3 + a_{11} = 2a_7$.
Với $a_6 + a_8$: chỉ số $6 + 8 = 14$, cũng bằng $2 \cdot 7$ nên $a_6 + a_8 = 2a_7$.

Vậy $4a_7 = 12$, suy ra $a_7 = 3$.

Biểu thức cần tính:

$$2a\_9 - a\_{11} = 2(a\_7 + 2d) - (a\_7 + 4d) = 2a\_7 + 4d - a\_7 - 4d = a\_7 = 3$$

\textbf{Cách nhanh.} Rút gọn biểu thức về đúng $a_7$ trước khi thay số, phần $d$ tự triệt tiêu.
\end{dapan}

\subsection*{Nhóm C — Lượng giác}

\begin{dapan}[19 — Đáp án A]
\textbf{Giải.} Biểu thức $y = 3\sin x + 4\cos x$ có dạng $p\sin x + q\cos x$ với $p = 3$, $q = 4$. Giá trị lớn nhất:

$$y\_{\max} = \sqrt{p^2 + q^2} = \sqrt{9 + 16} = 5$$

Dấu bằng xảy ra khi $\sin x = \dfrac{3}{5}$ và $\cos x = \dfrac{4}{5}$.

\textbf{Cách nhanh.} Với biểu thức $p\sin x + q\cos x$, giá trị lớn nhất luôn là $\sqrt{p^2+q^2}$ và giá trị nhỏ nhất là $-\sqrt{p^2+q^2}$. Nhớ công thức này để trả lời trong vài giây.

\textbf{Bẫy.} Đáp án D là $7$, tức cộng trực tiếp hai hệ số $3 + 4$. Không được cộng hệ số khi hai hàm khác nhau.
\end{dapan}

\begin{dapan}[20 — Đáp án D]
\textbf{Giải.} Áp dụng công thức nhân đôi $\cos 2\alpha = 1 - 2\sin^2\alpha$:

$$\cos 2\alpha = 1 - 2 \cdot \left(\frac{3}{5}\right)^2 = 1 - 2 \cdot \frac{9}{25} = 1 - \frac{18}{25} = \frac{7}{25}$$

\textbf{Chú ý.} Công thức $\cos 2\alpha = 1 - 2\sin^2\alpha$ chỉ chứa $\sin^2\alpha$ nên kết quả không phụ thuộc dấu của $\cos\alpha$. Đề không cần cho biết $\alpha$ thuộc góc phần tư nào.

\textbf{Bẫy.} Đáp án B là $-\dfrac{7}{25}$: sai dấu, thường do dùng $\cos 2\alpha = 2\sin^2\alpha - 1$ thay vì $1 - 2\sin^2\alpha$. Đáp án A là $\dfrac{24}{25}$, tức giá trị của $\sin 2\alpha$, không phải $\cos 2\alpha$.
\end{dapan}

\begin{dapan}[21 — Đáp án B]
\textbf{Giải.} Tỉ số $\dfrac{f(x_2) - f(x_1)}{x_2 - x_1}$ là hệ số góc của đường thẳng $AB$, đồng thời là tốc độ biến thiên trung bình của hàm số trên đoạn đó.

Đề cho tốc độ biến thiên trung bình bằng $\sqrt{3}$, vậy hệ số góc của $AB$ là $k = \sqrt{3}$.

Góc nghiêng $\alpha$ của đường thẳng thỏa $\tan\alpha = k = \sqrt{3}$, suy ra $\alpha = \dfrac{\pi}{3}$.

\textbf{Bẫy.} Đáp án A là $\dfrac{\pi}{6}$: nhầm $\tan\dfrac{\pi}{6} = \dfrac{\sqrt{3}}{3}$ với $\sqrt{3}$. Đáp án C và D sai dấu.
\end{dapan}

\begin{dapan}[22 — Đáp án C]
\textbf{Giải.} Hàm số liên tục tại $x = 0$ khi $\lim\limits_{x \to 0} f(x) = f(0) = a$.

Tính giới hạn: $\lim\limits_{x \to 0} \dfrac{\sin x}{x} = 1$.

Vậy $a = 1$.

\textbf{Chú ý.} Đây là giới hạn cơ bản phải nhớ. Không cần khai triển hay dùng quy tắc L'Hôpital.
\end{dapan}

\subsection*{Nhóm D — Ôn tập tích lũy}

\begin{dapan}[23 — Đáp án A]
\textbf{Giải.} Với $a + b = 2$, áp dụng bất đẳng thức Cauchy cho hai số không âm:

$$\sqrt{ab} \le \frac{a+b}{2} = 1 \implies ab \le 1$$

Dấu bằng xảy ra khi $a = b = 1$. Vậy giá trị lớn nhất của $ab$ là $1$.

\textbf{Cách nhanh.} Với tổng hai số không đổi, tích lớn nhất khi hai số bằng nhau. Không cần kiểm tra điều kiện không âm vì tích lớn nhất đạt tại giá trị dương.
\end{dapan}

\begin{dapan}[24 — Đáp án A]
\textbf{Giải.} Với $x > 0$, $y > 0$ và $x + 2y = 4$. Áp dụng bất đẳng thức Cauchy cho hai số dương $x$ và $2y$:

$$x + 2y \ge 2\sqrt{x \cdot 2y} = 2\sqrt{2xy}$$

$$4 \ge 2\sqrt{2xy} \implies \sqrt{2xy} \le 2 \implies 2xy \le 4 \implies xy \le 2$$

Dấu bằng xảy ra khi $x = 2y$, kết hợp $x + 2y = 4$ cho $x = 2$, $y = 1$. Vậy giá trị lớn nhất của $xy$ là $2$.

\textbf{Bẫy.} Không được chia đôi $x + 2y = 4$ thành $x = y = 2$ rồi tính $xy = 4$. Hệ số $2$ trước $y$ buộc phải tách thành hai số hạng $x$ và $2y$ bằng nhau.
\end{dapan}

\begin{dapan}[25 — Đáp án B]
\textbf{Giải.} Giải tập $A$: $x^2 - 4 < 0 \iff (x-2)(x+2) < 0$. Vì hệ số bậc hai dương và dấu $< 0$, lấy trong khoảng hai nghiệm:

$$A = (-2, 2)$$

Tập $B = \{x \in \mathbb{Z} \mid -3 < x \le 1\} = \{-2, -1, 0, 1\}$.

Giao hai tập: các số nguyên trong $(-2,2)$ là $\{-1, 0, 1\}$. Vì $-2 \notin A$ (dấu nhỏ hơn nghiêm ngặt), nên:

$$A \cap B = \{-1, 0, 1\}$$

\textbf{Bẫy.} Đáp án A chứa $-2$, nhưng $-2 \notin A$. Đáp án C là khoảng thực, không phải tập số nguyên.
\end{dapan}

\subsection*{Nhóm D — Ôn tập tích lũy}

\begin{dapan}[1 — Đáp án A]
\textbf{Giải.} Vì $SA$ vuông góc với mặt phẳng $(ABC)$ nên hình chiếu của $S$ lên mặt phẳng là $A$. Góc giữa đường thẳng $SB$ và mặt phẳng $(ABC)$ là góc $\widehat{SBA}$.

Tam giác $SAB$ vuông tại $A$ (do $SA \perp (ABC)$ nên $SA \perp AB$):

$$\tan\widehat{SBA} = \frac{SA}{AB} = \frac{a\sqrt{3}}{a} = \sqrt{3}$$

Suy ra $\widehat{SBA} = 60^\circ$.

\textbf{Chú ý.} Góc giữa đường thẳng và mặt phẳng luôn nằm trong khoảng từ $0^\circ$ đến $90^\circ$. Nếu tính ra giá trị âm thì lấy trị tuyệt đối.
\end{dapan}

\begin{dapan}[2 — Đáp án A]
\textbf{Giải.} Đặt hệ trục: $A(0;0;0)$, $B(a;0;0)$, $C(0;2a;0)$, $S(0;0;3a)$.

Mặt phẳng $(SBC)$ đi qua $S$, $B$, $C$. Hai vectơ chỉ phương:

$$\vec{SB} = (a;0;-3a), \qquad \vec{SC} = (0;2a;-3a)$$

Vectơ pháp tuyến:

$$\vec{n} = \vec{SB} \times \vec{SC} = \big(0 \cdot (-3a) - (-3a) \cdot 2a;\ (-3a) \cdot 0 - a \cdot (-3a);\ a \cdot 2a - 0 \cdot 0\big) = (6a^2;\ 3a^2;\ 2a^2)$$

Rút gọn $\vec{n} = a^2(6;3;2)$, lấy $\vec{n} = (6;3;2)$. Phương trình mặt phẳng $(SBC)$ có dạng $6x + 3y + 2z + D = 0$. Thay tọa độ $S(0;0;3a)$ để tìm $D$:

$$0 + 0 + 2 \cdot 3a + D = 0 \implies D = -6a$$

Vậy $(SBC): 6x + 3y + 2z - 6a = 0$. Khoảng cách từ $A(0;0;0)$ đến mặt phẳng:

$$d = \frac{|{-6a}|}{\sqrt{36+9+4}} = \frac{6a}{7}$$

\textbf{Chú ý.} Sau khi lập phương trình mặt phẳng, luôn thay tọa độ một điểm đã biết thuộc mặt phẳng để kiểm tra hằng số tự do. Bỏ qua bước này là ra đáp án sai ngay.
\end{dapan}

\begin{dapan}[3 — Đáp án B]
\textbf{Giải.} Đặt hệ trục với $A(0;0;0)$, $B(1;0;0)$, $C\left(\dfrac{1}{2};\dfrac{\sqrt{3}}{2};0\right)$, $A_1(0;0;1)$, $B_1(1;0;1)$, $C_1\left(\dfrac{1}{2};\dfrac{\sqrt{3}}{2};1\right)$.

Điểm $P$ thỏa $\vec{BP} = \lambda\vec{BC} + \mu\vec{BB_1}$, tức $P$ chạy trên mặt bên $BCC_1B_1$.

\textbf{(1) Sai.} Khi $\lambda = 1$, $P$ nằm trên cạnh $CC_1$. Chu vi tam giác $AB_1P$:

$$AB\_1 + B\_1P + PA = \sqrt{2} + \sqrt{\mu^2+1} + \sqrt{(\mu-1)^2+1}$$

Tại $\mu = 0$ giá trị là $1 + 2\sqrt{2}$; tại $\mu = \dfrac{1}{2}$ giá trị là $\sqrt{2} + \sqrt{5}$. Hai giá trị khác nhau nên chu vi không phải hằng số.

\textbf{(2) Đúng.} Khi $\mu = 1$, $P$ nằm trên đường $B_1C_1$. Thể tích tứ diện $P\text{-}A_1BC$ có đáy $A_1BC$ cố định và đỉnh $P$ chạy trên đường thẳng song song với mặt đáy đó, nên khoảng cách từ $P$ đến mặt phẳng $(A_1BC)$ không đổi. Thể tích bằng $\dfrac{\sqrt{3}}{12}$, là hằng số.

\textbf{(3) Sai.} Khi $\lambda = \dfrac{1}{2}$, điều kiện $A_1P \perp BP$ tương đương $\vec{A_1P} \cdot \vec{BP} = 0$, khai triển cho $\mu(\mu-1) = 0$, tức $\mu = 0$ hoặc $\mu = 1$. Có \textbf{hai} điểm thỏa mãn, không phải một.

\textbf{(4) Đúng.} Khi $\mu = \dfrac{1}{2}$, điều kiện $A_1B \perp$ mặt phẳng $(AB_1P)$ tương đương $\vec{A_1B} \perp \vec{AB_1}$ (luôn đúng) và $\vec{A_1B} \perp \vec{AP}$, khai triển cho $\lambda = 1$. Có đúng một điểm.

Vậy có \textbf{2} mệnh đề đúng là (2) và (4).

\textbf{Chú ý.} Đây là dạng bài đếm số mệnh đề đúng, phải kiểm tra từng mệnh đề bằng tọa độ. Mệnh đề (3) là bẫy: nghiệm có hai giá trị nhưng đề nói "có且仅有" (có và chỉ có một).
\end{dapan}

\begin{dapan}[4 — Đáp án C]
\textbf{Giải.} Hình "tiện trừ" là hình năm mặt có ba cạnh song song nhau và một mặt là hình thang. Xét từng nhận định:

\textbf{① Đúng.} Hình năm mặt có ba cạnh song song đôi một tạo ra đúng hai mặt tam giác ở hai đầu.

\textbf{② Đúng.} Hình chóp cụt (台体) có hai mặt đáy song song và đồng dạng. Tiện trừ chỉ có một mặt thang nên không thể là chóp cụt.

\textbf{③ Sai.} Tồn tại tiện trừ có hai mặt là hình bình hành (trường hợp đặc biệt khi mặt thang trở thành hình bình hành).

\textbf{④ Đúng.} Tiện trừ có tối đa hai mặt thang.

Vậy có \textbf{3} nhận định đúng.
\end{dapan}

\begin{dapan}[5 — Đáp án C]
\textbf{Giải.} Gọi $a' = 656$ (ft) là cạnh đáy, $h$ là chiều cao hình chóp, $PH$ là đường cao mặt bên (trung đoạn). Gọi $x = PH$.

Vì $H$ là trung điểm $BC$ nên khoảng cách từ tâm $O$ đến $H$ bằng nửa cạnh đáy: $OH = \dfrac{a'}{2} = 328$.

Tam giác vuông $POH$ cho $x^2 = h^2 + 328^2$, tức $h^2 = x^2 - 328^2$.

Mặt bên là tam giác có đáy $a' = 656$ và chiều cao $x$, diện tích $\dfrac{1}{2} \cdot 656 \cdot x = 328x$.

Đề cho diện tích mỗi mặt bên bằng bình phương chiều cao chóp:

$$328x = h^2 = x^2 - 328^2$$

$$x^2 - 328x - 328^2 = 0$$

Giải phương trình bậc hai, lấy nghiệm dương:

$$x = \frac{328 + \sqrt{328^2 + 4 \cdot 328^2}}{2} = \frac{328(1 + \sqrt{5})}{2} = 164(1+\sqrt{5}) \approx 164 \cdot 3{,}236 \approx 530{,}7$$

Vậy $PH \approx 530{,}7$ ft.

\textbf{Nhận xét.} Đề cho hằng số vàng nhưng không cần dùng trực tiếp; chỉ cần lập đúng phương trình từ điều kiện "diện tích mặt bên bằng bình phương chiều cao". Nghiệm dương của phương trình tự chứa $\dfrac{1+\sqrt{5}}{2}$.
\end{dapan}

\begin{dapan}[6 — Đáp án B]
\textbf{Giải.} Trong không gian, hai đường thẳng có ba vị trí tương đối: cắt nhau, song song, chéo nhau.

\textbf{Chú ý.} Phân biệt với mặt phẳng: trong mặt phẳng chỉ có hai vị trí (cắt nhau, song song), không có chéo nhau. Chéo nhau là khái niệm chỉ có trong không gian ba chiều.
\end{dapan}

\begin{dapan}[7 — Đáp án D]
\textbf{Giải.} Mặt cầu $(x-1)^2 + (y+2)^2 + (z+6)^2 = 16$ có tâm $I(1;-2;-6)$ và bán kính $R = \sqrt{16} = 4$.

\textbf{Bẫy.} Đáp án C dùng bán kính $16$, tức quên lấy căn bậc hai. Đáp án A và C sai dấu tâm: tâm $(1;-2;-6)$ cho ra $(x-1)^2 + (y+2)^2 + (z+6)^2$.
\end{dapan}

\begin{dapan}[8 — Đáp án B]
\textbf{Giải.} Công thức khoảng cách hai điểm trong không gian:

$$AB = \sqrt{(x\_2-x\_1)^2 + (y\_2-y\_1)^2 + (z\_2-z\_1)^2}$$

Với $A(1;1;1)$, $B(2;2;2)$:

$$AB = \sqrt{1^2 + 1^2 + 1^2} = \sqrt{3}$$

\textbf{Cách nhanh.} Hai điểm nằm trên đường chéo khối của hình lập phương đơn vị, khoảng cách là $\sqrt{3}$.
\end{dapan}

\begin{dapan}[9 — Đáp án D]
\textbf{Giải.} Phương trình $x^2 + y^2 + Dx + Ey + F = 0$ có tâm $\left(-\dfrac{D}{2};-\dfrac{E}{2}\right)$ và bán kính $r = \dfrac{1}{2}\sqrt{D^2+E^2-4F}$.

Với $D = -4$, $E = 6$, $F = -3$:

$$r = \frac{1}{2}\sqrt{16 + 36 + 12} = \frac{1}{2}\sqrt{64} = \frac{8}{2} = 4$$

\textbf{Cách nhanh.} Nhóm bình phương: $(x-2)^2 + (y+3)^2 = 4 + 9 + 3 = 16$, suy ra $r = 4$.
\end{dapan}

\begin{dapan}[10 — Đáp án C]
\textbf{Giải.} Biến đổi phương trình về dạng chính tắc:

$$x^2+y^2-2x+4y-20=0 \iff (x-1)^2 + (y+2)^2 = 25$$

Vậy quỹ tích điểm $M(x;y)$ là đường tròn tâm $I(1;-2)$, bán kính $R = 5$.

Biểu thức $x^2+y^2$ là bình phương khoảng cách từ $M$ đến gốc $O$. Khoảng cách này nhỏ nhất khi $M$ nằm trên đoạn $OI$, tại điểm gần $O$ nhất:

$$OM\_{\min} = |OI - R| = \sqrt{5} - 5$$

Vậy giá trị nhỏ nhất:

$$(x^2+y^2)\_{\min} = (5-\sqrt{5})^2 = 25 - 10\sqrt{5} + 5 = 30 - 10\sqrt{5}$$

\textbf{Bẫy.} Đáp án B là $5 - \sqrt{5}$: quên bình phương. Đáp án A là $5$, tức dùng $OI$ thay vì $OM_{\min}$.
\end{dapan}

\begin{dapan}[11 — Đáp án A]
\textbf{Giải.} Hình hộp chữ nhật có ba kích thước $3$, $2$, $4$. Diện tích toàn phần:

$$S = 2(ab + bc + ca) = 2(3 \cdot 2 + 2 \cdot 4 + 4 \cdot 3) = 2(6 + 8 + 12) = 2 \cdot 26 = 52$$

\textbf{Bẫy.} Đáp án B là $24$, tức tính thể tích $3 \cdot 2 \cdot 4$. Đáp án C là $26$, tức quên nhân $2$.
\end{dapan}

\begin{dapan}[12 — Đáp án A]
\textbf{Giải.} Hình trụ có bán kính đáy $r = 2$ và chiều cao $h = 5$:

$$V = \pi r^2 h = \pi \cdot 4 \cdot 5 = 20\pi$$

\textbf{Bẫy.} Đáp án B là $10\pi$, tức dùng nhầm $\pi r h$. Đáp án D là $25\pi$, tức dùng $r = 5$.
\end{dapan}

\begin{dapan}[13 — Đáp án B]
\textbf{Giải.} Hình nón có bán kính đáy $r = 6$ và chiều cao $h = 8$:

$$V = \frac{1}{3}\pi r^2 h = \frac{1}{3}\pi \cdot 36 \cdot 8 = 96\pi$$

\textbf{Bẫy.} Đáp án D là $288\pi$, tức quên hệ số $\dfrac{1}{3}$. Đáp án A là $48\pi$, tức lấy $r^2 = 18$.
\end{dapan}

\begin{dapan}[14 — Đáp án C]
\textbf{Giải.} Hình cầu bán kính $R = 3$. Diện tích mặt:

$$S = 4\pi R^2 = 4\pi \cdot 9 = 36\pi$$

\textbf{Bẫy.} Đáp án A là $9\pi$, tức dùng công thức diện tích đường tròn $\pi R^2$. Đáp án D là $27\pi$, tức tính thể tích $\dfrac{4}{3}\pi R^3$ rồi bỏ phần $\dfrac{4}{3}$.
\end{dapan}

\begin{dapan}[15 — Đáp án D]
\textbf{Giải.} Lục giác đều có $n = 6$ cạnh. Tổng góc trong:

$$(n-2) \cdot 180^\circ = 4 \cdot 180^\circ = 720^\circ$$

Mỗi góc trong của đa giác đều:

$$\frac{720^\circ}{6} = 120^\circ$$

\textbf{Cách nhanh.} Mỗi góc ngoài của đa giác đều $n$ cạnh là $\dfrac{360^\circ}{n}$. Với lục giác, góc ngoài là $60^\circ$, suy ra góc trong là $180^\circ - 60^\circ = 120^\circ$.
\end{dapan}
