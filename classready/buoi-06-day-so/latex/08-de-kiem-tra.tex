\begin{baitap}[1]
已知 $x \in \mathbb{R}$ ，则 $x + \dfrac{1}{x}$ 的最小值为 ( )
\par\vspace{3pt}
\noindent A. 1 \quad B. 2 \quad C. -2 \quad D. 不存在
\end{baitap}

\begin{baitap}[2]
如果某物体的速度函数是 $v(t) = 2t + 3$ (单位: m/s), 求从 $t = 0$ 到 $t = 2$ 秒内物体移动的距离等于 ( )
\par\vspace{3pt}
\noindent A. 4 m \quad B. 6 m \quad C. 8 m \quad D. 10 m
\end{baitap}

\begin{baitap}[3]
已知无穷等比数列 $\{a_n\}$ 的首项 $a_1 = 8$，公比 $q = \dfrac{1}{2}$，则该数列各项之和等于（ ）
\par\vspace{3pt}
\noindent A. $8$ \quad B. $16$ \quad C. $24$ \quad D. $12$
\end{baitap}

\begin{baitap}[4]
$\log_7 7$ 等于（ ）
\par\vspace{3pt}
\noindent A. $0$ \quad B. $7$ \quad C. $1$ \quad D. $\dfrac{1}{7}$
\end{baitap}

\begin{baitap}[5]
函数 $f(x) = x - \sqrt{1 - x^2}$ 的零点个数为 ( )
\par\vspace{3pt}
\noindent A. 0 \quad B. 1 \quad C. 2 \quad D. 3
\end{baitap}

\begin{baitap}[6]
$\lim_{x\to 3} \dfrac{x^2-9}{x-3}$ 等于 ( )
\par\vspace{3pt}
\noindent A. 0 \quad B. 3 \quad C. 6 \quad D. 不存在
\end{baitap}

\begin{baitap}[7]
已知等比数列 $\{b_n\}$ 中， $b_1 = 4, q = 3$ ，则 $b_4$ 等于（ ）
\par\vspace{3pt}
\noindent A. 12 \quad B. 13 \quad C. 28 \quad D. 108
\end{baitap}

\begin{baitap}[8]
由函数 $y = x^2 + x - 2$ 的图像和 x 轴围成的平面图形的面积是( )
\par\vspace{3pt}
\noindent A. $\dfrac{7}{3}$ \quad B. $\dfrac{9}{2}$ \quad C. $\dfrac{10}{3}$ \quad D. $\dfrac{11}{2}$
\end{baitap}

\begin{baitap}[9]
定义在 $\mathbb{N}$ 上的函数 $f(x)=x^2-2kx$ 是增函数，则系数 $k$ 的取值范围是（ ）
\par\vspace{3pt}
\noindent A. $k<\frac{1}{2}$ \quad B. $k<1$ \quad C. $k<\frac{3}{2}$ \quad D. $k<0$
\end{baitap}

\begin{baitap}[10]
已知函数 $y = f(x)$ 在 $x = x_0$ 处可导，且 $\lim_{\Delta x \to 0} \dfrac{f(x_0 + \Delta x) - f(x_0)}{\Delta x} = \dfrac{3}{2}$，则 $f'(x_0)$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{9}{4}$ \quad B. $\dfrac{3}{2}$ \quad C. 1 \quad D. $\dfrac{2}{3}$
\end{baitap}

\begin{baitap}[11]
已知等比数列 $\{a_n\}$ 的首项 $a_1 = 2$，$a_4 = 16$，则前 $5$ 项和 $S_5$ 等于（ ）
\par\vspace{3pt}
\noindent A. $32$ \quad B. $30$ \quad C. $62$ \quad D. $64$
\end{baitap}

\begin{baitap}[12]
$\int_0^1 (3x^2 + 1)\,dx$ 的值是 ( )
\par\vspace{3pt}
\noindent A. 1 \quad B. 2 \quad C. 3 \quad D. 4
\end{baitap}

\begin{baitap}[13]
函数 $f(x) = \log_2(x - 1)$ 的定义域是（ ）
\par\vspace{3pt}
\noindent A. $(0, +\infty)$ \quad B. $[1, +\infty)$ \quad C. $(1, +\infty)$ \quad D. $(-\infty, 1)$
\end{baitap}

\begin{baitap}[14]
已知点 $A(x_1,y_1)$，$B(x_2,y_2)$ 在函数 $y=f(x)$ 的图象上，若函数 $f(x)$ 在 $[x_1,x_2]$ 上的平均变化率为 $\sqrt{3}$，则下面叙述正确的是（ ）
\par\vspace{3pt}
\noindent A. 直线 $AB$ 的倾斜角为 $\frac{\pi}{6}$ \quad B. 直线 $AB$ 的倾斜角为 $\frac{\pi}{3}$ \quad C. 直线 $AB$ 的斜率为 $-\sqrt{3}$ \quad D. 直线 $AB$ 的斜率为 $-\frac{\sqrt{3}}{3}$
\end{baitap}

\begin{baitap}[15]
已知等比数列 $\{a_n\}$ 的各项均为正数，且 $a_1 + a_2 = 6$，$a_2 + a_3 = 12$，则 $a_4$ 等于（ ）
\par\vspace{3pt}
\noindent A. $8$ \quad B. $12$ \quad C. $16$ \quad D. $32$
\end{baitap}

\begin{baitap}[16]
如果曲线 $y = f(x)$ 上一点 $(1, 3)$ 处的切线过点 $(0, 2)$，则有（ ）
\par\vspace{3pt}
\noindent A. $f'(1) > 0$ \quad B. $f'(1) = 0$ \quad C. $f'(1) < 0$ \quad D. $f'(1)$ 不存在
\end{baitap}

\begin{baitap}[17]
下列函数中，是奇函数的是（ ）
\par\vspace{3pt}
\noindent A. $y = x^2$ \quad B. $y = |x|$ \quad C. $y = x^3$ \quad D. $y = x + 1$
\end{baitap}

\begin{baitap}[18]
积分 $\int (3x^2 + 3e^x + 2)\,dx$ 等于 ( )
\par\vspace{3pt}
\noindent A. $x^3 + e^x + 2x + C$ \quad B. $6x + 3e^x$ \quad C. $3x^3 + e^x + C$ \quad D. $x^3 + \ln x + 2x + C$
\end{baitap}

\begin{baitap}[19]
已知等差数列 $\{a_n\}$ 满足 $a_2 + a_5 + a_8 = 21$，则 $S_9$ 等于（ ）
\par\vspace{3pt}
\noindent A. $63$ \quad B. $21$ \quad C. $42$ \quad D. $54$
\end{baitap}

\begin{baitap}[20]
若函数 $f(x)$ 满足 $f(x) = \int (3x^2 - 1)\,dx$ ，且 $f(1) = 0$ 。求 $f(x)$ 等于 ( )
\par\vspace{3pt}
\noindent A. $x^3 - x$ \quad B. $x^3 - x + 1$ \quad C. $x^3 - x - 1$ \quad D. $x^3 - x + C$
\end{baitap}

\begin{baitap}[21]

\begin{baitap}[22]

\begin{baitap}[23]

\begin{baitap}[24]
已知三角形的三边构成等比数列，且它们的公比为 $q$，则 $q$ 的取值范围是（ ） \emph{Hết giờ. Đáp án và lời giải: giáo viên phát sau buổi kiểm tra.}

\end{baitap}
