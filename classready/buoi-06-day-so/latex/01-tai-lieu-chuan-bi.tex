\section{PHẦN C — BÀI TẬP CHUẨN BỊ}

Làm 10 câu dưới đây \textbf{trước buổi học}. Sáu câu đầu chỉ cần đọc hiểu thuật ngữ là làm được; các câu cuối đòi hỏi biến đổi, nếu chưa làm được thì ghi lại chỗ vướng mắc để hỏi trên lớp.

\subsection{Mức dễ}

\begin{baitap}[1]
数列 $a_n = n^2 + 1$ 的第 4 项是（ ）
\par\vspace{3pt}
\noindent A. 5 \quad B. 8 \quad C. 16 \quad D. 17
\end{baitap}
\begin{baitap}[2]
已知 $\{a_n\}$ 为等比数列，$a_3 = 4$，$a_5 \cdot a_7 = 36$，则 $a_9$ 的值为（ ）
\par\vspace{3pt}
\noindent A. $-9$ \quad B. 9 或 $-9$ \quad C. 8 \quad D. 9
\end{baitap}
\begin{baitap}[3]
若 $x, y, z$ 成等比数列，则下列关系式正确的是（ ）
\par\vspace{3pt}
\noindent A. $x + y = z$ \quad B. $2y = x + z$ \quad C. $y^2 = xz$ \quad D. $x + z = 2y$
\end{baitap}
\begin{baitap}[4]
已知等比数列 $\{a_n\}$ 的各项均为正数，$a_2 = 4$，$a_4 = 16$，则 $a_1 + a_3$ 等于（ ）
\par\vspace{3pt}
\noindent A. $6$ \quad B. $10$ \quad C. $12$ \quad D. $20$
\end{baitap}
\begin{baitap}[5]
等差数列 $\{a_n\}$ 的公差为 $d$，前 $n$ 项和为 $S_n$，若 $a_1+a_9=10$，$S_{11}=66$，则 $d=$（ ）
\par\vspace{3pt}
\noindent A. 1 \quad B. 2 \quad C. 3 \quad D. 4
\end{baitap}
\begin{baitap}[6]
在等比数列 $\{b_n\}$ 中， $b_1 = 1, b_5 = 16$ 。求 $b_1 + b_3$ 的值等于（ ）
\par\vspace{3pt}
\noindent A. 5 \quad B. 10 \quad C. 12 \quad D. 17
\end{baitap}
\subsection{Mức trung bình}

\begin{baitap}[7]
已知等比数列 $\{a_n\}$ 的首项 $a_1 = 1$，公比 $q = 3$，若前 $n$ 项和 $S_n = 40$，则 $n$ 等于（ ）
\par\vspace{3pt}
\noindent A. $4$ \quad B. $3$ \quad C. $5$ \quad D. $6$
\end{baitap}
\begin{baitap}[8]
已知等差数列 $\{a_n\}$ 的前 $n$ 项和为 $S_n$。若 $S_3=S_{10}$，$S_6=S_k\ (k\ne6)$，则 $k$ 的值是（ ）
\par\vspace{3pt}
\noindent A. 5 \quad B. 7 \quad C. 8 \quad D. 9
\end{baitap}
\begin{baitap}[9]
已知等比数列 $\{a_n\}$ 的首项 $a_1 = 3$，公比 $q = 2$，则 $a_4$ 等于（ ）
\par\vspace{3pt}
\noindent A. $12$ \quad B. $24$ \quad C. $48$ \quad D. $36$
\end{baitap}
