\section{PHẦN C — BÀI TẬP CHUẨN BỊ}

Làm 10 câu dưới đây \textbf{trước buổi học}. Sáu câu đầu chỉ cần đọc hiểu thuật ngữ là làm được; các câu cuối đòi hỏi biến đổi, nếu chưa làm được thì ghi lại chỗ vướng mắc để hỏi trên lớp.

\subsection{Mức dễ}

\begin{baitap}[1]
已知集合 $A=\{x\mid -1\le x<10\}$，集合 $B=\{x\mid x\ge2\}$，则 $A\cap B$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\{x\mid -1\le x<10\}$ \quad B. $\{x\mid x\ge2\}$ \quad C. $\{x\mid 2\le x<10\}$ \quad D. $\{x\mid -1\le x<2\}$
\end{baitap}
\begin{baitap}[2]
已知集合 $P=\{1,2,4,6,8\}$，集合 $Q=\{2,3,4,7,9\}$，则 $P\cup Q$ 的元素个数是（ ）
\par\vspace{3pt}
\noindent A. 7 \quad B. 8 \quad C. 9 \quad D. 10
\end{baitap}
\begin{baitap}[3]
设集合 $S=\{x\mid x=2k,\ k\in\mathbb{Z}\}$，集合 $T=\{x\mid x=2k+1,\ k\in\mathbb{Z}\}$，则 $S\cup T$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\mathbb{Z}$ \quad B. $\varnothing$ \quad C. $\mathbb{N}$ \quad D. $\mathbb{Q}$
\end{baitap}
\begin{baitap}[4]
空集 $\varnothing$ 是（ ）
\par\vspace{3pt}
\noindent A. 任何集合的子集 \quad B. 任何非空集合的子集 \quad C. 任何集合的真子集 \quad D. 任何集合都不是它的子集
\end{baitap}
\begin{baitap}[5]
区间 $[-1,3)$ 可以表示为集合（ ）
\par\vspace{3pt}
\noindent A. $\{x\mid -1<x\le3\}$ \quad B. $\{x\mid -1\le x<3\}$ \quad C. $\{x\mid -1\le x\le3\}$ \quad D. $\{x\mid -1<x<3\}$
\end{baitap}
\begin{baitap}[6]
不等式 $|2x^2-1|<7$ 的解集是（ ）
\par\vspace{3pt}
\noindent A. $(-2,2)$ \quad B. $[0,2]$ \quad C. $(-\infty,-2)\cup(0,+\infty)$ \quad D. $(-\infty,-2)\cup[2,+\infty)$
\end{baitap}
\subsection{Mức trung bình}

\begin{baitap}[7]
设集合 $A=\{x\mid x^2-2x-3>0\}$，集合 $B=\{x\mid x^2-4<0\}$，则 $A\cap B$ 等于（ ）
\par\vspace{3pt}
\noindent A. $(-\infty,-1)\cup(3,+\infty)$ \quad B. $(3,+\infty)$ \quad C. $(-2,3)$ \quad D. $(-2,-1)$
\end{baitap}
\begin{baitap}[8]
不等式 $\dfrac{3-x^2}{x+2}\le0$ 的解集是（ ）
\par\vspace{3pt}
\noindent A. $(-\infty,\sqrt{3}]$ \quad B. $(-2,\sqrt{3}]$ \quad C. $(-2,-\sqrt{3}]\cup[\sqrt{3},+\infty)$ \quad D. $[-\sqrt{3},+\infty)$
\end{baitap}
\begin{baitap}[9]
已知集合 $M=\{x\mid |x-1|\le2\}$，集合 $N=\{x\mid x^2-4x\ge0\}$，则 $M\cap N$ 等于（ ）
\par\vspace{3pt}
\noindent A. $[-1,0]$ \quad B. $[-1,4]$ \quad C. $[-1,3]$ \quad D. $\varnothing$
\end{baitap}
\subsection{Mức khó}

\begin{baitap}[10]
已知 $A=\{-1,\dfrac{1}{2}\}$，$B=\{x\mid ax+1=0\}$，若 $A\cap B=B$，则实数 $a$ 的取值构成的集合是（ ）
\par\vspace{3pt}
\noindent A. $\{-1,2\}$ \quad B. $\{-2,1\}$ \quad C. $\{-2,0,1\}$ \quad D. $\{-1,0,2\}$
\end{baitap}
