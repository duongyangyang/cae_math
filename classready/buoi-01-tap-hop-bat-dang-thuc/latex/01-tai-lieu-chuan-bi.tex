\section{Bài tập chuẩn bị}

Làm 10 câu dưới đây \textbf{trước buổi học}. Sáu câu đầu chỉ cần đọc hiểu thuật ngữ là làm được; các câu cuối đòi hỏi biến đổi, nếu chưa làm được thì ghi lại chỗ vướng mắc để hỏi trên lớp.

\begin{baitap}[1]
已知集合 $A=\{x\mid -1\le x<10\}$，集合 $B=\{x\mid x\ge2\}$，则 $A\cap B$ 等于（ ）\\[4pt]
A. $\{x\mid -1\le x<10\}$ \quad B. $\{x\mid x\ge2\}$ \quad C. $\{x\mid 2\le x<10\}$ \quad D. $\{x\mid -1\le x<2\}$
\end{baitap}

\begin{baitap}[2]
已知集合 $P=\{1,2,4,6,8\}$，集合 $Q=\{2,3,4,7,9\}$，则 $P\cup Q$ 的元素个数是（ ）\\[4pt]
A. 7 \quad B. 8 \quad C. 9 \quad D. 10
\end{baitap}

\begin{baitap}[3]
设集合 $S=\{x\mid x=2k,\ k\in\mathbb{Z}\}$，集合 $T=\{x\mid x=2k+1,\ k\in\mathbb{Z}\}$，则 $S\cup T$ 等于（ ）\\[4pt]
A. $\mathbb{Z}$ \quad B. $\varnothing$ \quad C. $\mathbb{N}$ \quad D. $\mathbb{Q}$
\end{baitap}

\begin{baitap}[4]
空集 $\varnothing$ 是（ ）\\[4pt]
A. 任何集合的子集 \quad B. 任何非空集合的子集 \quad C. 任何集合的真子集 \quad D. 任何集合都不是它的子集
\end{baitap}

\begin{baitap}[5]
区间 $[-1,3)$ 可以表示为集合（ ）\\[4pt]
A. $\{x\mid -1<x\le3\}$ \quad B. $\{x\mid -1\le x<3\}$ \quad C. $\{x\mid -1\le x\le3\}$ \quad D. $\{x\mid -1<x<3\}$
\end{baitap}

\begin{baitap}[6]
不等式 $|2x^2-1|<7$ 的解集是（ ）\\[4pt]
A. $(-2,2)$ \quad B. $[0,2]$ \quad C. $(-\infty,-2)\cup(0,+\infty)$ \quad D. $(-\infty,-2)\cup[2,+\infty)$
\end{baitap}

\begin{baitap}[7]
设集合 $A=\{x\mid x^2-2x-3>0\}$，集合 $B=\{x\mid x^2-4<0\}$，则 $A\cap B$ 等于（ ）\\[4pt]
A. $(-\infty,-1)\cup(3,+\infty)$ \quad B. $(3,+\infty)$ \quad C. $(-2,3)$ \quad D. $(-2,-1)$
\end{baitap}

\begin{baitap}[8]
不等式 $\dfrac{3-x^2}{x+2}\le0$ 的解集是（ ）\\[4pt]
A. $(-\infty,\sqrt{3}]$ \quad B. $(-2,\sqrt{3}]$ \quad C. $(-2,-\sqrt{3}]\cup[\sqrt{3},+\infty)$ \quad D. $[-\sqrt{3},+\infty)$
\end{baitap}

\begin{baitap}[9]
已知集合 $M=\{x\mid |x-1|\le2\}$，集合 $N=\{x\mid x^2-4x\ge0\}$，则 $M\cap N$ 等于（ ）\\[4pt]
A. $[-1,0]$ \quad B. $[-1,4]$ \quad C. $[-1,3]$ \quad D. $\varnothing$
\end{baitap}

\begin{baitap}[10]
已知 $A=\{-1,\dfrac{1}{2}\}$，$B=\{x\mid ax+1=0\}$，若 $A\cap B=B$，则实数 $a$ 的取值构成的集合是（ ）\\[4pt]
A. $\{-1,2\}$ \quad B. $\{-2,1\}$ \quad C. $\{-2,0,1\}$ \quad D. $\{-1,0,2\}$
\end{baitap}

\subsection*{Bản dịch}

\textbf{Câu 2.} 设全集 $U=\mathbb{R}$，集合 $M=\{x\mid -2<x\le8\}$，则 $\overline{M}$ 等于（ ）

\textbf{Dịch.} Cho tập vũ trụ $U = \mathbb{R}$, tập hợp $M = \{x \mid -2 < x \le 8\}$, khi đó phần bù $\overline{M}$ bằng ( )

\nhan{Từ khóa cần nhận ra} 设 (cho, giả sử) · \cjk{全集} (tập vũ trụ) · 则 (khi đó, thì) · 等于 (bằng).

\textbf{Câu 3.} 若关于 $x$ 的不等式 $x^2+(m-1)x+1>0$ 对于一切实数 $x$ 恒成立，则实数 $m$ 的取值范围是（ ）

\textbf{Dịch.} Nếu bất phương trình $x^2 + (m-1)x + 1 > 0$ đối với ẩn $x$ đúng với mọi số thực $x$, thì khoảng giá trị của số thực $m$ là ( )

\nhan{Từ khóa cần nhận ra} 若 (nếu) · 关于 $x$ 的不等式 (bất phương trình đối với $x$) · 对于一切实数 (với mọi số thực) · \cjk{恒成立} (đúng với mọi giá trị) · \cjk{取值范围} (khoảng giá trị).

\textbf{Nhận xét.} Gặp \cjk{恒成立} là biết ngay đây là dạng bài dùng điều kiện $a > 0$ và $\Delta < 0$. Đọc được từ khóa này tiết kiệm khoảng 20 giây so với việc đọc dịch toàn bộ đề.

\subsection*{Lời giải 10 câu}

\begin{dapan}[1]
\textbf{Đáp án C.} Biểu diễn $A = [-1, 10)$ và $B = [2, +\infty)$ trên trục số. Phần chung bắt đầu từ $2$ (thuộc cả hai tập) và kết thúc trước $10$.
\[
A \cap B = [2, 10) = \{x \mid 2 \le x < 10\}
\]
\textbf{Bẫy.} Đáp án B sai vì thiếu điều kiện $x < 10$: lấy cả phần $x \ge 10$ không thuộc $A$.
\end{dapan}

\begin{dapan}[2]
\textbf{Đáp án B.} Hợp của hai tập gồm tất cả phần tử của cả hai, phần tử trùng chỉ viết một lần.
\[
P \cup Q = \{1, 2, 3, 4, 6, 7, 8, 9\}
\]
Số phần tử là $8$.

\textbf{Cách nhanh.} $P$ có $5$ phần tử, $Q$ có $5$ phần tử, phần chung $\{2,4\}$ có $2$ phần tử. Vậy $|P \cup Q| = 5 + 5 - 2 = 8$. Không cần liệt kê.
\end{dapan}

\begin{dapan}[3]
\textbf{Đáp án A.} $S$ là tập các số nguyên chẵn, $T$ là tập các số nguyên lẻ. Mọi số nguyên hoặc chẵn hoặc lẻ, nên $S \cup T = \mathbb{Z}$.
\end{dapan}

\begin{dapan}[4]
\textbf{Đáp án A.} Tập rỗng không có phần tử nào. Mệnh đề ``mọi phần tử của $\varnothing$ đều thuộc $A$'' luôn đúng vì không có phần tử nào để phản bác. Vậy $\varnothing \subseteq A$ với mọi tập hợp $A$.

\textbf{Bẫy.} Đáp án C sai vì $\varnothing$ \emph{không} là tập con thực sự của chính nó.
\end{dapan}

\begin{dapan}[5]
\textbf{Đáp án B.} Ký hiệu $[-1, 3)$ nghĩa là đầu mút $-1$ \emph{thuộc} tập (ngoặc vuông) và đầu mút $3$ \emph{không thuộc} tập (ngoặc tròn).
\[
[-1, 3) = \{x \mid -1 \le x < 3\}
\]
\textbf{Quy tắc nhớ.} Ngoặc vuông $[\ ]$ = lấy đầu mút, ngoặc tròn $(\ )$ = không lấy.
\end{dapan}

\begin{dapan}[6]
\textbf{Đáp án A.} $|2x^2 - 1| < 7 \iff -7 < 2x^2 - 1 < 7$. Cộng $1$: $-6 < 2x^2 < 8$. Chia $2$: $-3 < x^2 < 4$.

Vì $x^2 \ge 0$ nên điều kiện $-3 < x^2$ luôn đúng; chỉ cần $x^2 < 4 \iff -2 < x < 2$.
\[
x \in (-2, 2)
\]
\textbf{Chú ý.} Dạng này dễ biến đổi sai nếu chia hai vế cho $2$ rồi bỏ qua điều kiện $x^2 \ge 0$.
\end{dapan}

\begin{dapan}[7]
\textbf{Đáp án D.} Giải $x^2 - 2x - 3 > 0$: hai nghiệm $x = -1$, $x = 3$, lấy ngoài khoảng:
\[
A = (-\infty, -1) \cup (3, +\infty)
\]
Giải $x^2 - 4 < 0$: hai nghiệm $x = \pm 2$, lấy trong khoảng:
\[
B = (-2, 2)
\]
Giao của hai tập chỉ nằm ở đoạn $(-2, -1)$.
\[
A \cap B = (-2, -1)
\]
\textbf{Nhận xét.} Nhánh $(3, +\infty)$ của $A$ không giao với $B$ vì $B$ dừng ở $2$. Đây là lý do phải vẽ trục số thay vì suy đoán.
\end{dapan}

\begin{dapan}[8]
\textbf{Đáp án C.} Tử: $3 - x^2 = 0 \implies x = \pm\sqrt{3}$. Mẫu: $x + 2 = 0 \implies x = -2$ (loại).

Ba điểm $-\sqrt{3} < -2 < \sqrt{3}$ chia trục số thành bốn khoảng:

\bang{
\begin{tabular}{@{}lccc@{}}
\toprule
\textbf{Khoảng} & \textbf{Tử} $3-x^2$ & \textbf{Mẫu} $x+2$ & \textbf{Thương} \\
\midrule
$x < -\sqrt{3}$ & $-$ & $-$ & $+$ \\
$-\sqrt{3} < x < -2$ & $+$ & $-$ & $-$ \checkmark \\
$-2 < x < \sqrt{3}$ & $+$ & $+$ & $+$ \\
$x > \sqrt{3}$ & $-$ & $+$ & $-$ \checkmark \\
\bottomrule
\end{tabular}}

Tập nghiệm: $(-2, -\sqrt{3}] \cup [\sqrt{3}, +\infty)$.

\textbf{Chú ý.} Đề dùng dấu $\le$ nên hai nghiệm của tử được lấy, còn nghiệm của mẫu $x = -2$ luôn bị loại. Thứ tự đúng trên trục số là $-\sqrt{3} \approx -1{,}73$ rồi mới đến $-2$.
\end{dapan}

\begin{dapan}[9]
\textbf{Đáp án A.} Giải tập $M$: $|x - 1| \le 2 \iff -2 \le x - 1 \le 2 \iff -1 \le x \le 3$. Vậy $M = [-1, 3]$.

Giải tập $N$: $x^2 - 4x \ge 0 \iff x(x-4) \ge 0$, lấy ngoài khoảng hai nghiệm:
\[
N = (-\infty, 0] \cup [4, +\infty)
\]
Giao hai tập chỉ nằm ở đoạn $[-1, 0]$.
\[
M \cap N = [-1, 0]
\]
\textbf{Nhận xét.} Nhánh $[4, +\infty)$ của $N$ không giao với $M$ vì $M$ dừng ở $3$.
\end{dapan}

\begin{dapan}[10]
\textbf{Đáp án C.} Điều kiện $A \cap B = B$ tương đương $B \subseteq A$.

\textbf{Trường hợp 1: $a = 0$.} Phương trình $ax + 1 = 0$ thành $1 = 0$, vô nghiệm, nên $B = \varnothing$. Tập rỗng là tập con của mọi tập hợp, vậy $a = 0$ thỏa mãn.

\textbf{Trường hợp 2: $a \neq 0$.} Phương trình có nghiệm duy nhất $x = -\dfrac{1}{a}$. Để $B \subseteq A$ thì $-\dfrac{1}{a}$ phải bằng $-1$ hoặc $\dfrac{1}{2}$.
\[
-\frac{1}{a} = -1 \implies a = 1; \qquad -\frac{1}{a} = \frac{1}{2} \implies a = -2
\]
Vậy $a \in \{-2, 0, 1\}$.

\textbf{Bẫy.} Đây là dạng bài bẫy kinh điển. Học sinh thường quên trường hợp $a = 0$ và chỉ chọn $\{-2, 1\}$ (đáp án B). \emph{Nhớ: tham số ở hệ số của $x$ thì luôn xét riêng giá trị làm hệ số đó bằng 0.}
\end{dapan}
