\section{Bài tập bắt buộc}

Làm toàn bộ 25 câu. Đây là phần tính vào điểm bài tập về nhà.

\subsection*{NHÓM A — TẬP HỢP}

\begin{baitap}[1]
设全集 $U=\mathbb{R}$，集合 $M=\{x\mid -2<x\le8\}$，则 $\overline{M}$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\{x\mid x\le-2 \text{ 或 } x>8\}$ \quad B. $\{x\mid x<-2 \text{ 或 } x\ge8\}$ \quad C. $\{x\mid x\le-2 \text{ 或 } x\ge8\}$ \quad D. $\{x\mid x<-2 \text{ 或 } x>8\}$
\end{baitap}

\begin{baitap}[2]
已知集合 $A=\{x\mid x^2-12x+27=0\}$，则集合 $A$ 的真子集的个数为（ ）
\par\vspace{3pt}
\noindent A. 1 \quad B. 2 \quad C. 3 \quad D. 4
\end{baitap}

\begin{baitap}[3]
已知全集 $U=\{1,2,3,4,5,6\}$，集合 $P=\{x\mid x^2-5x+6=0\}$，集合 $Q=\{x\mid x\text{ 是偶数}\}$。求 $C_U(P\cup Q)$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\{1,5\}$ \quad B. $\{1,3,5\}$ \quad C. $\{1,2,3,4,5,6\}$ \quad D. $\{3,5\}$
\end{baitap}

\begin{baitap}[4]
已知集合 $A=\{x\mid x^2-5x+6=0\}$，则集合 $A$ 的子集个数为（ ）
\par\vspace{3pt}
\noindent A. 1 \quad B. 2 \quad C. 3 \quad D. 4
\end{baitap}

\begin{baitap}[5]
已知集合 $A=\{1,3,11,13,15\}$，集合 $B=\{3,6,7,11\}$，则 $A\cup B$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\{1,3,6,7\}$ \quad B. $\{3,11\}$ \quad C. $\{1,3,6,7,11,13,15\}$ \quad D. $\{1,6,7,13,15\}$
\end{baitap}

\begin{baitap}[6]
集合 $A=\{x\mid x\text{ 是偶数}\}$，集合 $B=\{x\mid x\text{ 是3的倍数}\}$，则 $A\cap B$ 是（ ）
\par\vspace{3pt}
\noindent A. $\{x\mid x\text{ 是5的倍数}\}$ \quad B. $\{x\mid x\text{ 是6的倍数}\}$ \quad C. $\{x\mid x\text{ 是2或3的倍数}\}$ \quad D. $\{x\mid x\text{ 是奇数}\}$
\end{baitap}

\begin{baitap}[7]
已知集合 $A=\{x\in\mathbb Z\mid -1<x\le5\}$，$B=\{-5,-3,-1,2,3,4\}$，則 $A\cap B$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\{-1,2,3,4\}$ \quad B. $\{-1,2,4\}$ \quad C. $\{2,3,4\}$ \quad D. $\{-5,-3,-1\}$
\end{baitap}

\begin{baitap}[8]
设 $A = \{(x,y) \mid y = -4x + 6\}$，$B = \{(x,y) \mid y = 5x - 3\}$，则 $A \cap B =$（ ）
\par\vspace{3pt}
\noindent A. $\{(2,7)\}$ \quad B. $(1,2)$ \quad C. $x = 1, y = 2$ \quad D. $\{(1,2)\}$
\end{baitap}

\begin{baitap}[9]
已知集合 $A=\{x\mid x^2-x\le2\}$，$B=\{x\mid a\le x\le a+1\}$，若 $B\subseteq A$，则实数 $a$ 的取值集合为（ ）
\par\vspace{3pt}
\noindent A. $[0,1]$ \quad B. $[-1,0]$ \quad C. $[-1,2]$ \quad D. $[-1,1]$
\end{baitap}

\begin{baitap}[10]
已知 $A = \{x \mid x^2 - 3x + 2 = 0\}$，$B = \{x \mid ax - 2 = 0\}$，若 $A \cap B = B$，则实数 $a$ 的值为（ ）
\par\vspace{3pt}
\noindent A. 0或1或2 \quad B. 1或2 \quad C. 0 \quad D. 0或1
\end{baitap}

\begin{baitap}[11]
设集合 $M=\{x\mid (x-7)(x-11)\le0\}$，$N=\{x\mid 8<x<17\}$，则 $M\cap N$ 是（ ）
\par\vspace{3pt}
\noindent A. $(7,11]$ \quad B. $(8,17)$ \quad C. $(8,11]$ \quad D. $(7,17)$
\end{baitap}

\begin{baitap}[12]
设集合 $M=\{(x,y)\mid x+y=0\}$，$N=\{(x,y)\mid x^2+y^2=2\}$，则 $M\cap N$ 是（ ）
\par\vspace{3pt}
\noindent A. $\{(-1,1),(1,-1)\}$ \quad B. $\{(1,-1)\}$ \quad C. $\{(-1,1)\}$ \quad D. $\varnothing$
\end{baitap}

\subsection*{NHÓM B — BẤT ĐẲNG THỨC}

\begin{baitap}[13]
不等式 $x^2 - 5x + 6 \ge 0$ 的解集是 ( )
\par\vspace{3pt}
\noindent A. $[2,3]$ \quad B. $(-\infty, -3] \cup [-2, +\infty)$ \quad C. $(-\infty, 2] \cup [3, +\infty)$ \quad D. $[-3, -2]$
\end{baitap}

\begin{baitap}[14]
不等式 $|6x - 1| \le 3$ 的解集是 ( )
\par\vspace{3pt}
\noindent A. $[-\dfrac{2}{3}, \dfrac{1}{3}]$ \quad B. $(-1, \dfrac{2}{3})$ \quad C. $(-\infty, -\dfrac{1}{3}] \cup [\dfrac{2}{3}, +\infty)$ \quad D. $[-\dfrac{1}{3}, \dfrac{2}{3}]$
\end{baitap}

\begin{baitap}[15]
不等式 $x^2 - 4x + 3 < 0$ 的解集是 ( )
\par\vspace{3pt}
\noindent A. $(-\infty, 1) \cup (3, +\infty)$ \quad B. $(1,3)$ \quad C. $[1,3]$ \quad D. $(-\infty, 1] \cup [3, +\infty)$
\end{baitap}

\begin{baitap}[16]
不等式 $(x^2 - 1)(x - 2) < 0$ 的解集是 ( )
\par\vspace{3pt}
\noindent A. $(-\infty, -1) \cup (1,2)$ \quad B. $(-1,1) \cup (2, +\infty)$ \quad C. $(-\infty, -1) \cup (2, +\infty)$ \quad D. $(-1,2)$
\end{baitap}

\begin{baitap}[17]
不等式组 $\begin{cases} |x| < 4 \\ x^2 - x > 0 \end{cases}$ 的解集是 ( )
\par\vspace{3pt}
\noindent A. $(0,4)$ \quad B. $(-4,1)$ \quad C. $(-4,1) \cup (4, +\infty)$ \quad D. $(-4,0) \cup (1,4)$
\end{baitap}

\begin{baitap}[18]
10B 班的每个学生都踢足球或打排球。已知有 25 名学生踢足球，20 名学生打排球，10 名学生同时进行这两项运动。10B 班的学生人数是 ( )
\par\vspace{3pt}
\noindent A. 20 \quad B. 25 \quad C. 30 \quad D. 35
\end{baitap}

\begin{baitap}[19]
已知 $x, y$ 是正实数, 且 $x + y = 6$ 。求 $\dfrac{1}{x} + \dfrac{1}{y}$ 的最小值等于 ( )
\par\vspace{3pt}
\noindent A. $\dfrac{1}{6}$ \quad B. $\dfrac{1}{3}$ \quad C. $\dfrac{2}{3}$ \quad D. $\dfrac{1}{2}$
\end{baitap}

\begin{baitap}[20]
解不等式 $\dfrac{x-3}{x+2} \ge 2$ 等于 ( )
\par\vspace{3pt}
\noindent A. $(-2, +\infty)$ \quad B. $(-\infty, -2)$ \quad C. $[-7, -2)$ \quad D. $\emptyset$
\end{baitap}

\begin{baitap}[21]
已知不等式 $ax^2+bx+2>0$ 的解为 $-\frac{1}{2}<x<\frac{1}{3}$，则 $a+b=$（ ）
\par\vspace{3pt}
\noindent A. -14 \quad B. -8 \quad C. 10 \quad D. 12
\end{baitap}

\begin{baitap}[22]
不等式 $\dfrac{4}{x^2-1} < 1$ 的解集是 ( )
\par\vspace{3pt}
\noindent A. $(-1, +\infty)$ \quad B. $(-\infty, -\sqrt{5}) \cup (-1,1) \cup (\sqrt{5}, +\infty)$ \quad C. $(-\sqrt{5}, -1) \cup (1, \sqrt{5})$ \quad D. $(-\infty, \sqrt{5})$
\end{baitap}

\begin{baitap}[23]
不等式 $x^2 - x + m > 0$ 在 $\mathbb{R}$ 上恒成立的一个充要条件是（ ）
\par\vspace{3pt}
\noindent A. $m > \dfrac{1}{4}$ \quad B. $0 < m < 1$ \quad C. $m > 0$ \quad D. $m > 7$
\end{baitap}

\begin{baitap}[24]
已知 $a>b$，$c>d$，且 $cd\ne0$，则（ ）
\par\vspace{3pt}
\noindent A. $ad>bc$ \quad B. $ac>bc$ \quad C. $a-c>b-d$ \quad D. $a+c>b+d$
\end{baitap}

\begin{baitap}[25]
关于 $x$ 的不等式 $x^2+ax-2>0$ 在 $\{x\mid 1\le x\le5\}$ 上有解，则 $a$ 的取值范围是（ ）
\par\vspace{3pt}
\noindent A. $\{a\mid a>-\frac{23}{5}\}$ \quad B. $\{a\mid -\frac{23}{5}\le a\le1\}$ \quad C. $\{a\mid a>1\}$ \quad D. $\{a\mid a\le\frac{23}{5}\}$
\end{baitap}

\section{Bài tập tự chọn}

Phần này luyện cùng nội dung nhưng ở mức khó hơn rõ rệt. Chỉ làm sau khi đã hoàn thành 25 câu bắt buộc. Không tính vào điểm bài tập về nhà.

\begin{baitap}[1]
若全集 $U = \{x \in \mathbb{R} \mid -2 \leq x \leq 2\}$，集合 $A = \{x \in \mathbb{R} \mid -2 \leq x \leq 0\}$，则补集 $\complement_U A$ 为（ ）
\par\vspace{3pt}
\noindent A. $\{x \in \mathbb{R} \mid 0 < x < 2\}$ \quad B. $\{x \in \mathbb{R} \mid 0 \leq x < 2\}$ \quad C. $\{x \in \mathbb{R} \mid 0 < x \leq 2\}$ \quad D. $\{x \in \mathbb{R} \mid 0 \leq x \leq 2\}$
\end{baitap}

\begin{baitap}[2]
已知 $a, b \in \mathbb{R}$ ，且 $a < b$ ，下列不等式恒成立的是 ( )
\par\vspace{3pt}
\noindent A. $a^2 > b^2$ \quad B. $\dfrac{1}{a} < \dfrac{1}{b}$ \quad C. $ka < kb$ $(k > 0)$ \quad D. $|a| > |b|$
\end{baitap}

\begin{baitap}[3]
已知集合 $P=\{x\mid |2x-13|<5\}$，集合 $Q=\{x\mid 10-x>0\}$，则 $P\cap Q$ 为（ ）
\par\vspace{3pt}
\noindent A. $(4,9)$ \quad B. $(4,10)$ \quad C. $(-\infty,9)$ \quad D. $(-\infty,10)$
\end{baitap}

\begin{baitap}[4]
$A = \{x \mid 1 \leq x < 4\}$，$B = \{x \mid y = \lg(x^2 - 2x)\}$，则 $A \cap (\complement_{\mathbb{R}} B) =$（ ）
\par\vspace{3pt}
\noindent A. $\{x \mid 1 \leq x \leq 2\}$ \quad B. $\{x \mid 1 \leq x < 2\}$ \quad C. $\{x \mid 2 < x < 4\}$ \quad D. $\{x \mid 2 \leq x < 4\}$
\end{baitap}

\begin{baitap}[5]
已知 $M = \{x \mid x^2 - 4x + 3 < 0\}$，$N = \{x \mid y = \sqrt{x^2 - 4}\}$，则 $M \cup N =$（ ）
\par\vspace{3pt}
\noindent A. $(1, 2]$ \quad B. $(-\infty, -2] \cup (1, 3)$ \quad C. $(-\infty, -2] \cup (3, +\infty)$ \quad D. $(-\infty, -2] \cup (1, +\infty)$
\end{baitap}

\begin{baitap}[6]
下列说法正确的是（ ）
\par\vspace{3pt}
\noindent A. 由 1，2，3 组成的集合可表示为 $\{1,2,3\}$ 或 $\{3,2,1\}$ \quad B. $\varnothing$ 与 $\{0\}$ 是同一个集合 \quad C. 集合 $\{x\mid y=x^2-1\}$ 与集合 $\{y\mid y=x^2-1\}$ 是同一个集合 \quad D. 集合 $\{x\mid x^2+5x+6=0\}$ 与集合 $\{x^2+5x+6=0\}$ 是同一个集合
\end{baitap}

\begin{baitap}[7]
已知全集为 $\mathbb{R}$，集合 $A=\{x\mid -2<x<1\}$，集合 $B=\{x\mid -x^2+x<0\}$，则 $A\cup(\complement_{\mathbb{R}}B)=$（ ）
\par\vspace{3pt}
\noindent A. $(-2,1]$ \quad B. $(-1,1]$ \quad C. $(-\infty,-2)\cup[1,+\infty)$ \quad D. $(-\infty,0]\cup(1,+\infty)$
\end{baitap}

\begin{baitap}[8]
已知 $M = x^2 - x + 3, N = x + 2$，则 $M$ 与 $N$ 大小关系是（ ）
\par\vspace{3pt}
\noindent A. $M > N$ \quad B. $M < N$ \quad C. $M \ge N$ \quad D. $M \le N$
\end{baitap}

\begin{baitap}[9]
已知 $x<a<0$，下列不等式一定成立的是（ ）
\par\vspace{3pt}
\noindent A. $x^2<a^2<0$ \quad B. $x^2>ax>a^2$ \quad C. $x^2<ax<0$ \quad D. $x^2>a^2>ax$
\end{baitap}

\begin{baitap}[10]
解分式不等式 $\dfrac{x-4}{4x^2-1} < 0$ 等于 ( )
\par\vspace{3pt}
\noindent A. $(-\infty, -\dfrac{1}{2}) \cup (\dfrac{1}{2}, 4)$ \quad B. $(-\dfrac{1}{2}, \dfrac{1}{2}) \cup (4, +\infty)$ \quad C. $(-\infty, -\dfrac{1}{2}) \cup (\dfrac{1}{2}, +\infty)$ \quad D. $(-\dfrac{1}{2}, 4)$
\end{baitap}

\begin{baitap}[11]
不等式 $\sqrt{4x - x^2} < x$ 的解集是（ ）
\par\vspace{3pt}
\noindent A. $(0, 2]$ \quad B. $(2, +\infty)$ \quad C. $(2, 4]$ \quad D. $(-\infty, 0) \cup (2, +\infty)$
\end{baitap}

\begin{baitap}[12]
已知实数 $x, y$ 满足 $x^2 + y^2 = 1,$ 求 $x + y$ 的最大值等于 ( )
\par\vspace{3pt}
\noindent A. 1 \quad B. $\sqrt{2}$ \quad C. 2 \quad D. $\dfrac{\sqrt{2}}{2}$
\end{baitap}

\begin{baitap}[13]
已知 $a, b \in \mathbb{R},$ 且 $a^2 + b^2 = 1,$ 则 $2a + 3b$ 的最大值为 ( )
\par\vspace{3pt}
\noindent A. $\sqrt{13}$ \quad B. 5 \quad C. $\sqrt{5}$ \quad D. $\dfrac{\sqrt{13}}{2}$
\end{baitap}

\begin{baitap}[14]
命题“$ax^2 - 2ax + 4 > 0$ 恒成立”是假命题，则实数 $a$ 的取值范围是（ ）
\par\vspace{3pt}
\noindent A. $a < 0$ 或 $a \geq 3$ \quad B. $a \leq 0$ 或 $a \geq 4$ \quad C. $a < 0$ 或 $a > 3$ \quad D. $a < 0$ 或 $a \geq 4$
\end{baitap}

\begin{baitap}[15]
已知 $\triangle ABC$ 的三边长分别为 $a, b, c$，有以下4个命题： （1）以 $\sqrt{a}, \sqrt{b}, \sqrt{c}$ 为边长的三角形一定存在； （2）以 $a^2, b^2, c^2$ 为边长的三角形一定存在； （3）以 $\dfrac{a+b}{2}, \dfrac{b+c}{2}, \dfrac{c+a}{2}$ 为边长的三角形一定存在； （4）以 $ab, bc, ca$ 为边长的三角形一定存在。 其中正确命题的个数为（ ）
\par\vspace{3pt}
\noindent A. 1个 \quad B. 2个 \quad C. 3个 \quad D. 4个
\end{baitap}
