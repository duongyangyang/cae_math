\subsection*{Nhóm A — Tập hợp}

\begin{dapan}[1 — Đáp án A]
\textbf{Giải.} $M = (-2, 8]$. Phần bù trong $\mathbb{R}$ là những giá trị \textbf{không} thuộc $M$: nhỏ hơn hoặc bằng $-2$, hoặc lớn hơn $8$.

$$\overline{M} = (-\infty, -2] \cup (8, +\infty) = \{x \mid x \le -2 \ \text{hoặc} \ x > 8\}$$

\textbf{Chú ý.} Đầu mút $-2$ \textbf{thuộc} phần bù (vì $-2 \notin M$), đầu mút $8$ \textbf{không thuộc} phần bù (vì $8 \in M$). Đây là chỗ dễ sai nhất. Đảo hai đầu mút là mất điểm.
\end{dapan}

\begin{dapan}[2 — Đáp án C]
\textbf{Giải.} Giải $x^2 - 12x + 27 = 0$: $\Delta = 144 - 108 = 36$, hai nghiệm $x = \dfrac{12 \pm 6}{2}$, tức $x = 3$ hoặc $x = 9$. Vậy $A = \{3, 9\}$ có $n = 2$ phần tử.

Số tập con thực sự là $2^2 - 1 = 3$.
\end{dapan}

\begin{dapan}[3 — Đáp án A]
\textbf{Giải.} Tập $P$: $x^2 - 5x + 6 = 0 \implies (x-2)(x-3) = 0 \implies P = \{2, 3\}$.

Tập $Q$: các số chẵn trong $U = \{1,\dots,6\}$ là $Q = \{2, 4, 6\}$.

$$P \cup Q = \{2, 3, 4, 6\}$$

Phần bù trong $U$:

$$C\_U(P \cup Q) = \{1, 5\}$$
\end{dapan}

\begin{dapan}[4 — Đáp án D]
\textbf{Giải.} $x^2 - 5x + 6 = 0 \implies (x-2)(x-3) = 0 \implies A = \{2,3\}$, có $2$ phần tử.

Số tập con (không phải tập con thực sự) là $2^2 = 4$.

\textbf{Chú ý.} Phân biệt 子集 (tập con, gồm cả chính nó) và 真子集 (tập con thực sự, không gồm chính nó). Câu 2 và Câu 4 trong đề này là cặp đối chiếu trực tiếp. Đọc kỹ đề hỏi loại nào.
\end{dapan}

\begin{dapan}[5 — Đáp án C]
\textbf{Giải.} Gộp tất cả phần tử, phần tử chung $\{3, 11\}$ chỉ viết một lần:

$$A \cup B = \{1, 3, 6, 7, 11, 13, 15\}$$

\textbf{Cách nhanh.} $|A \cup B| = 5 + 4 - 2 = 7$ phần tử. Đáp án C là đáp án duy nhất có $7$ phần tử.
\end{dapan}

\begin{dapan}[6 — Đáp án B]
\textbf{Giải.} $A$ là tập các số chẵn, $B$ là tập các số chia hết cho $3$. Giao của hai tập là các số vừa chẵn vừa chia hết cho $3$, tức chia hết cho $6$.

$$A \cap B = \{x \mid x \ \text{là bội của} \ 6\}$$

\textbf{Nhận xét.} Vì $2$ và $3$ nguyên tố cùng nhau nên bội chung nhỏ nhất là $2 \times 3 = 6$. Đáp án C mô tả \textbf{hợp} chứ không phải giao. Đọc kỹ ký hiệu $\cap$ và $\cup$.
\end{dapan}

\begin{dapan}[7 — Đáp án C]
\textbf{Giải.} $A = \{x \in \mathbb{Z} \mid -1 < x \le 5\} = \{0, 1, 2, 3, 4, 5\}$.

Giao với $B = \{-5, -3, -1, 2, 3, 4\}$ là các phần tử thuộc cả hai:

$$A \cap B = \{2, 3, 4\}$$

\textbf{Bẫy.} Đáp án A chứa $-1$, nhưng $-1 \notin A$ vì điều kiện là $-1 < x$ (dấu nhỏ hơn nghiêm ngặt). Đáp án B chứa $-1$ và thiếu $3$.
\end{dapan}

\begin{dapan}[8 — Đáp án D]
\textbf{Giải.} Tập $A$ là đường thẳng $y = -4x + 6$; tập $B$ là đường thẳng $y = 5x - 3$. Giao của hai tập hợp là giao điểm của hai đường thẳng.

Giải $-4x + 6 = 5x - 3 \implies 9x = 9 \implies x = 1$. Thay vào: $y = -4(1) + 6 = 2$.

Vậy $A \cap B = \{(1, 2)\}$.

\textbf{Bẫy.} Đáp án B viết $(1,2)$: ký hiệu này chỉ \textbf{một khoảng} trên trục số, không phải một điểm trong mặt phẳng. Đáp án C viết $x=1, y=2$ là hai phương trình, không phải một tập hợp. Giao điểm phải viết dưới dạng cặp tọa độ trong ngoặc nhọn.
\end{dapan}

\begin{dapan}[9 — Đáp án D]
\textbf{Giải.} Giải $x^2 - x \le 2 \implies x^2 - x - 2 \le 0 \implies (x-2)(x+1) \le 0$. Vì $a > 0$ và bất phương trình $\le 0$, lấy trong khoảng hai nghiệm:

$$A = [-1, 2]$$

Điều kiện $B \subseteq A$ với $B = [a, a+1]$ là:

$$a \ge -1 \quad \text{và} \quad a + 1 \le 2 \implies a \le 1$$

Vậy $a \in [-1, 1]$.
\end{dapan}

\begin{dapan}[10 — Đáp án A]
\textbf{Giải.} $A = \{x \mid x^2 - 3x + 2 = 0\} = \{1, 2\}$. Điều kiện $A \cap B = B \iff B \subseteq A$.

\textbf{Trường hợp $a = 0$:} $B = \varnothing \subseteq A$ ✓

\textbf{Trường hợp $a \neq 0$:} $ax - 2 = 0 \implies x = \dfrac{2}{a}$. Để $B \subseteq A$ thì $\dfrac{2}{a} \in \{1, 2\}$.

$$\frac{2}{a} = 1 \implies a = 2; \qquad \frac{2}{a} = 2 \implies a = 1$$

Vậy $a \in \{0, 1, 2\}$.
\end{dapan}

\begin{dapan}[11 — Đáp án C]
\textbf{Giải.} $(x-7)(x-11) \le 0$ với hệ số dương và dấu $\le$ cho nghiệm trong khoảng hai nghiệm:

$$M = [7, 11]$$

$$N = (8, 17)$$

Giao: phần chung bắt đầu sau $8$ (không gồm $8$) và kết thúc tại $11$ (gồm $11$).

$$M \cap N = (8, 11]$$
\end{dapan}

\begin{dapan}[12 — Đáp án A]
\textbf{Giải.} Tập $M$ là đường thẳng $x + y = 0$, tức $y = -x$. Tập $N$ là đường tròn tâm gốc tọa độ, bán kính $\sqrt{2}$.

Thay $y = -x$ vào $x^2 + y^2 = 2$: $x^2 + x^2 = 2 \implies 2x^2 = 2 \implies x = \pm 1$.

Vậy $M \cap N = \{(-1, 1), (1, -1)\}$.

\textbf{Nhận xét.} Khoảng cách từ gốc tọa độ đến đường thẳng $x + y = 0$ bằng $0 < \sqrt{2}$, nên đường thẳng cắt đường tròn tại đúng hai điểm. Đây là cách kiểm tra nhanh trước khi giải.
\end{dapan}

\subsection*{Nhóm B — Bất đẳng thức}

\begin{dapan}[13 — Đáp án C]
\textbf{Giải.} $(x-2)(x-3) \ge 0$ với $a > 0$ cho nghiệm ngoài khoảng hai nghiệm: $x \le 2$ hoặc $x \ge 3$.
\end{dapan}

\begin{dapan}[14 — Đáp án D]
\textbf{Giải.} $|6x - 1| \le 3 \iff -3 \le 6x - 1 \le 3$.

Cộng $1$: $-2 \le 6x \le 4$. Chia $6$: $-\dfrac{1}{3} \le x \le \dfrac{2}{3}$.
\end{dapan}

\begin{dapan}[15 — Đáp án B]
\textbf{Giải.} $(x-1)(x-3) < 0$ với $a > 0$ và dấu $<$ cho nghiệm trong khoảng: $1 < x < 3$.
\end{dapan}

\begin{dapan}[16 — Đáp án A]
\textbf{Giải.} $(x^2-1)(x-2) = (x-1)(x+1)(x-2) < 0$. Ba nghiệm $\{-1, 1, 2\}$ chia trục số thành bốn khoảng:

\bang{
\begin{tabularx}{\linewidth}{@{\hspace{6pt}}Xc@{\hspace{6pt}}}
\toprule
\textbf{Khoảng} & \textbf{Dấu tích} \\
\midrule
$x < -1$ & $-$ ✓ \\
$-1 < x < 1$ & $+$ \\
$1 < x < 2$ & $-$ ✓ \\
$x > 2$ & $+$ \\
\bottomrule
\end{tabularx}}

Tập nghiệm: $(-\infty, -1) \cup (1, 2)$.

\textbf{Cách nhanh.} Với tích có $3$ nghiệm phân biệt, dấu luân phiên bắt đầu từ dấu của hệ số bậc cao nhất ở khoảng ngoài cùng bên phải. Hệ số bậc cao nhất dương nên khoảng $(2, +\infty)$ mang dấu $+$, rồi đổi dấu khi qua mỗi nghiệm.
\end{dapan}

\begin{dapan}[17 — Đáp án D]
\textbf{Giải.} Giải từng bất phương trình:

$|x| < 4 \implies -4 < x < 4$

$x^2 - x > 0 \implies x(x-1) > 0 \implies x < 0$ hoặc $x > 1$

Nghiệm của hệ là giao hai tập:

$$(-4, 4) \cap \big[(-\infty,0) \cup (1,+\infty)\big] = (-4, 0) \cup (1, 4)$$
\end{dapan}

\begin{dapan}[18 — Đáp án D]
\textbf{Giải.} Gọi $A$ là tập học sinh chơi bóng đá, $B$ là tập học sinh chơi bóng chuyền. Đề cho mọi học sinh đều chơi ít nhất một môn nên $A \cup B$ là cả lớp.

Công thức bao hàm – loại trừ:

$$|A \cup B| = |A| + |B| - |A \cap B| = 25 + 20 - 10 = 35$$

\textbf{Nhận xét.} Đây là dạng bài tập hợp núp dưới lời văn. Nhận ra cấu trúc 都...或 (đều... hoặc) là biết ngay dùng công thức hợp hai tập.
\end{dapan}

\begin{dapan}[19 — Đáp án C]
\textbf{Giải.} Với $x, y > 0$:

$$\frac{1}{x} + \frac{1}{y} \ge \frac{4}{x+y} = \frac{4}{6} = \frac{2}{3}$$

Dấu bằng xảy ra khi $x = y = 3$.

\textbf{Chú ý.} Dạng $\dfrac{1}{x} + \dfrac{1}{y}$ với $x + y$ cho trước luôn dùng hệ quả $\dfrac{1}{x} + \dfrac{1}{y} \ge \dfrac{4}{x+y}$ của bất đẳng thức Cauchy. Không cần biến đổi gì thêm.
\end{dapan}

\begin{dapan}[20 — Đáp án C]
\textbf{Giải.} Chuyển vế về dạng so với $0$:

$$\frac{x-3}{x+2} - 2 \ge 0 \iff \frac{x-3-2(x+2)}{x+2} \ge 0 \iff \frac{-x-7}{x+2} \ge 0$$

Nhân tử và mẫu cho $-1$ và đổi chiều:

$$\frac{x+7}{x+2} \le 0$$

Nghiệm tử: $x = -7$. Nghiệm mẫu: $x = -2$ (loại). Nghiệm trong khoảng hai nghiệm: $-7 \le x < -2$.

\textbf{Bẫy.} Không được nhân chéo $x+2$ vì chưa biết dấu của $x+2$. Phải chuyển vế về $0$ rồi xét dấu.
\end{dapan}

\begin{dapan}[21 — Đáp án A]
\textbf{Giải.} Bất phương trình $ax^2 + bx + 2 > 0$ có tập nghiệm $-\dfrac{1}{2} < x < \dfrac{1}{3}$, tức là nghiệm nằm \textbf{trong} khoảng hai nghiệm. Điều này chỉ xảy ra khi $a < 0$.

Hai nghiệm của phương trình $ax^2 + bx + 2 = 0$ là $x_1 = -\dfrac{1}{2}$ và $x_2 = \dfrac{1}{3}$. Theo định lý Viète:

$$x\_1 x\_2 = \frac{2}{a} = -\frac{1}{2} \cdot \frac{1}{3} = -\frac{1}{6} \implies a = -12$$

$$x\_1 + x\_2 = -\frac{b}{a} = -\frac{1}{2} + \frac{1}{3} = -\frac{1}{6} \implies \frac{b}{12} = -\frac{1}{6} \implies b = -2$$

Vậy $a + b = -12 + (-2) = -14$.
\end{dapan}

\begin{dapan}[22 — Đáp án B]
\textbf{Giải.} Chuyển vế:

$$\frac{4}{x^2-1} - 1 < 0 \iff \frac{4 - (x^2-1)}{x^2-1} < 0 \iff \frac{5 - x^2}{x^2-1} < 0$$

Tử: $5 - x^2 = 0 \implies x = \pm\sqrt{5}$. Mẫu: $x^2 - 1 = 0 \implies x = \pm 1$ (loại).

Bốn điểm $\{-\sqrt{5}, -1, 1, \sqrt{5}\}$ chia trục số thành năm khoảng. Xét dấu:

\bang{
\begin{tabularx}{\linewidth}{@{\hspace{6pt}}Xccc@{\hspace{6pt}}}
\toprule
\textbf{Khoảng} & \textbf{Tử $5-x^2$} & \textbf{Mẫu $x^2-1$} & \textbf{Thương} \\
\midrule
$x < -\sqrt{5}$ & $-$ & $+$ & $-$ ✓ \\
$-\sqrt{5} < x < -1$ & $+$ & $+$ & $+$ \\
$-1 < x < 1$ & $+$ & $-$ & $-$ ✓ \\
$1 < x < \sqrt{5}$ & $+$ & $+$ & $+$ \\
$x > \sqrt{5}$ & $-$ & $+$ & $-$ ✓ \\
\bottomrule
\end{tabularx}}

Tập nghiệm: $(-\infty, -\sqrt{5}) \cup (-1, 1) \cup (\sqrt{5}, +\infty)$.
\end{dapan}

\begin{dapan}[23 — Đáp án A]
\textbf{Giải.} $x^2 - x + m > 0$ đúng với mọi $x \in \mathbb{R}$ khi và chỉ khi $a = 1 > 0$ (đã thỏa) và $\Delta < 0$:

$$\Delta = 1 - 4m < 0 \implies m > \frac{1}{4}$$

\textbf{Nhận xét.} Đề hỏi 充要条件 (điều kiện cần và đủ) nên chỉ có $m > \dfrac{1}{4}$ là chính xác. Các đáp án $m > 0$ hay $0 < m < 1$ chỉ là điều kiện đủ, không phải cần và đủ.
\end{dapan}

\begin{dapan}[24 — Đáp án D]
\textbf{Giải.} Xét từng đáp án với $a > b$ và $c > d$:

\begin{itemize}[leftmargin=1.4em, itemsep=2pt]
\item \textbf{A.} $ad > bc$? Không suy ra được, vì dấu của $a, b$ chưa biết.
\item \textbf{B.} $ac > bc$? Sai khi $c < 0$.
\item \textbf{C.} $a - c > b - d$? Viết lại thành $(a-b) > (c-d)$. Ta biết $a - b > 0$ và $c - d > 0$ nhưng không so sánh được hai hiệu này.
\item \textbf{D.} $a + c > b + d$? Cộng hai bất đẳng thức cùng chiều: $a > b$ và $c > d$ cho $a + c > b + d$. ✓
\end{itemize}


\textbf{Chú ý.} Quy tắc: \textbf{cộng hai bất đẳng thức cùng chiều thì được}, nhưng \textbf{trừ hai bất đẳng thức cùng chiều thì không}. Đây là lỗi phổ biến nhất ở dạng bài này.
\end{dapan}

\begin{dapan}[25 — Đáp án A]
\textbf{Giải.} Đề hỏi bất phương trình $x^2 + ax - 2 > 0$ \textbf{có nghiệm} (有解) trong đoạn $[1, 5]$, tức tồn tại ít nhất một $x \in [1,5]$ thỏa mãn.

Điều kiện tương đương: tồn tại $x \in [1,5]$ sao cho

$$ax > 2 - x^2 \iff a > \frac{2 - x^2}{x} = \frac{2}{x} - x$$

Hàm $g(x) = \dfrac{2}{x} - x$ nghịch biến trên $[1,5]$ (vì $\dfrac{2}{x}$ giảm và $-x$ giảm). Vậy giá trị nhỏ nhất đạt tại $x = 5$:

$$g(5) = \frac{2}{5} - 5 = -\frac{23}{5}$$

Tồn tại $x$ thỏa mãn khi và chỉ khi $a > -\dfrac{23}{5}$.

\textbf{Bẫy.} Phân biệt 有解 (có nghiệm) và 恒成立 (đúng với mọi $x$). Bài này hỏi \textbf{có nghiệm} nên lấy \textbf{giá trị nhỏ nhất} của $g(x)$, không phải giá trị lớn nhất. Nếu đề đổi thành 恒成立 thì đáp án sẽ là $a > 1$.
\end{dapan}

\subsection*{Nhóm B — Bất đẳng thức}

\begin{dapan}[1 — Đáp án C]
\textbf{Giải.} $U = [-2, 2]$, $A = [-2, 0]$. Phần bù là phần của $U$ không thuộc $A$: từ lớn hơn $0$ đến $2$.

$$\complement\_U A = (0, 2] = \{x \in \mathbb{R} \mid 0 < x \le 2\}$$
\end{dapan}

\begin{dapan}[2 — Đáp án C]
\textbf{Giải.} $a < b$ và $k > 0$: nhân cả hai vế với số dương thì bất đẳng thức \textbf{không đổi chiều}, nên $ka < kb$. ✓

Các đáp án khác: A và D phụ thuộc dấu và giá trị tuyệt đối; B sai khi $a, b$ trái dấu hoặc khi một trong hai bằng $0$.
\end{dapan}

\begin{dapan}[3 — Đáp án A]
\textbf{Giải.} Giải tập $P$: $|2x - 13| < 5 \iff -5 < 2x - 13 < 5$. Cộng $13$: $8 < 2x < 18$. Chia $2$: $4 < x < 9$.

$$P = (4, 9)$$

Giải tập $Q$: $10 - x > 0 \iff x < 10$.

$$Q = (-\infty, 10)$$

Giao hai tập: vì $(4,9) \subset (-\infty, 10)$ nên giao chính là $P$.

$$P \cap Q = (4, 9)$$

\textbf{Nhận xét.} Khi một tập hợp là tập con của tập kia thì giao chính là tập nhỏ hơn. Nhận ra điều này tiết kiệm thời gian vẽ trục số.
\end{dapan}

\begin{dapan}[4 — Đáp án A]
\textbf{Giải.} Tìm tập xác định của $y = \lg(x^2 - 2x)$: biểu thức trong logarit phải dương.

$$x^2 - 2x > 0 \implies x(x-2) > 0 \implies x < 0 \ \text{hoặc} \ x > 2$$

Vậy $B = (-\infty, 0) \cup (2, +\infty)$, suy ra $\complement_{\mathbb{R}} B = [0, 2]$.

Giao với $A = [1, 4)$:

$$A \cap \complement\_{\mathbb{R}} B = [1, 4) \cap [0, 2] = [1, 2]$$
\end{dapan}

\begin{dapan}[5 — Đáp án D]
\textbf{Giải.} $M$: $x^2 - 4x + 3 < 0 \implies (x-1)(x-3) < 0 \implies M = (1, 3)$.

$N$: tập xác định của $y = \sqrt{x^2-4}$ cần $x^2 - 4 \ge 0 \implies x \le -2$ hoặc $x \ge 2$.

$$N = (-\infty, -2] \cup [2, +\infty)$$

Hợp: $M$ lấp đầy khoảng $(1,3)$, còn $N$ phủ hai đầu. Hợp của chúng là $(-\infty, -2]$ gộp với $(1, +\infty)$, vì phần $[2, 3)$ đã nằm trong $M$, phần $(3, +\infty)$ nằm trong $N$, và phần $(1,2)$ nằm trong $M$.

$$M \cup N = (-\infty, -2] \cup (1, +\infty)$$
\end{dapan}

\begin{dapan}[6 — Đáp án A]
\textbf{Giải.} Xét từng mệnh đề:

\begin{itemize}[leftmargin=1.4em, itemsep=2pt]
\item \textbf{A.} Tập hợp không có thứ tự, nên $\{1,2,3\} = \{3,2,1\}$. ✓
\item \textbf{B.} $\varnothing$ không có phần tử nào, còn $\{0\}$ có một phần tử là số $0$. Hai tập khác nhau. ✗
\item \textbf{C.} $\{x \mid y = x^2-1\}$ là \textbf{tập xác định}, bằng $\mathbb{R}$; còn $\{y \mid y = x^2-1\}$ là \textbf{tập giá trị}, bằng $[-1, +\infty)$. Khác nhau. ✗
\item \textbf{D.} $\{x \mid x^2+5x+6=0\}$ là tập nghiệm $\{-2,-3\}$; còn $\{x^2+5x+6=0\}$ là tập hợp chứa \textbf{một phương trình} làm phần tử. Khác nhau. ✗
\end{itemize}


\textbf{Chú ý.} Phân biệt ký hiệu tập hợp $\{x \mid \dots\}$ (tập các giá trị $x$ thỏa mãn) và $\{\dots\}$ (tập liệt kê phần tử). Đây là bẫy rất hay gặp.
\end{dapan}

\begin{dapan}[7 — Đáp án A]
\textbf{Giải.} $A = (-2, 1)$.

$B$: $-x^2 + x < 0 \iff x^2 - x > 0 \iff x(x-1) > 0 \iff x < 0$ hoặc $x > 1$.

Vậy $B = (-\infty, 0) \cup (1, +\infty)$, suy ra $\complement_{\mathbb{R}} B = [0, 1]$.

$$A \cup \complement\_{\mathbb{R}} B = (-2, 1) \cup [0, 1] = (-2, 1]$$

\textbf{Nhận xét.} Vì $[0,1] \subset (-2,1)$ trừ điểm $x = 1$, nên hợp chỉ mở rộng thêm đúng điểm $1$.
\end{dapan}

\begin{dapan}[8 — Đáp án C]
\textbf{Giải.} Xét hiệu:

$$M - N = (x^2 - x + 3) - (x + 2) = x^2 - 2x + 1 = (x-1)^2 \ge 0$$

Vậy $M \ge N$ với mọi $x$, dấu bằng xảy ra khi $x = 1$.

\textbf{Nhận xét.} Dạng bài so sánh hai biểu thức luôn giải bằng cách xét hiệu và đưa về bình phương hoặc tích. Ở đây hiệu là một bình phương hoàn hảo nên kết luận ngay được.
\end{dapan}

\begin{dapan}[9 — Đáp án B]
\textbf{Giải.} Với $x < a < 0$, cả hai đều âm và $|x| > |a|$ (vì $x$ âm hơn nên xa $0$ hơn).

Kiểm tra bằng ví dụ $x = -3$, $a = -1$:

$$x^2 = 9, \quad ax = (-1)(-3) = 3, \quad a^2 = 1$$

$$9 > 3 > 1 \implies x^2 > ax > a^2$$

Đáp án B đúng.

\textbf{Cách lý luận.} $x^2 > 0$ và $ax > 0$ (tích hai số âm). So sánh: $x^2 - ax = x(x-a)$. Vì $x < 0$ và $x - a < 0$ nên tích dương, suy ra $x^2 > ax$. Tương tự $ax - a^2 = a(x-a) > 0$ vì $a<0$ và $x-a<0$, suy ra $ax > a^2$.
\end{dapan}

\begin{dapan}[10 — Đáp án A]
\textbf{Giải.} Phân tích mẫu: $4x^2 - 1 = (2x-1)(2x+1)$, hai nghiệm $x = \dfrac{1}{2}$ và $x = -\dfrac{1}{2}$. Tử có nghiệm $x = 4$.

Ba điểm $\left\{-\dfrac{1}{2}, \dfrac{1}{2}, 4\right\}$ chia trục số thành bốn khoảng. Xét dấu thương:

\bang{
\begin{tabularx}{\linewidth}{@{\hspace{6pt}}Xccc@{\hspace{6pt}}}
\toprule
\textbf{Khoảng} & \textbf{Tử $x-4$} & \textbf{Mẫu $4x^2-1$} & \textbf{Thương} \\
\midrule
$x < -\dfrac{1}{2}$ & $-$ & $+$ & $-$ ✓ \\
$-\dfrac{1}{2} < x < \dfrac{1}{2}$ & $-$ & $-$ & $+$ \\
$\dfrac{1}{2} < x < 4$ & $-$ & $+$ & $-$ ✓ \\
$x > 4$ & $+$ & $+$ & $+$ \\
\bottomrule
\end{tabularx}}

Tập nghiệm là $\left(-\infty, -\dfrac{1}{2}\right) \cup \left(\dfrac{1}{2}, 4\right)$.

\textbf{Chú ý.} Hai nghiệm của mẫu luôn bị loại, kể cả khi đề dùng dấu $\le$. Ở đây đề dùng dấu $<$ nên không ảnh hưởng, nhưng cần nhớ quy tắc.
\end{dapan}

\begin{dapan}[11 — Đáp án C]
\textbf{Giải.} Điều kiện xác định: $4x - x^2 \ge 0 \implies x(4-x) \ge 0 \implies 0 \le x \le 4$.

Vì vế trái không âm nên phải có $x > 0$. Bình phương hai vế (cả hai đều không âm):

$$4x - x^2 < x^2 \implies 2x^2 - 4x > 0 \implies 2x(x-2) > 0 \implies x < 0 \ \text{hoặc} \ x > 2$$

Kết hợp $0 < x \le 4$: nghiệm là $2 < x \le 4$.

\textbf{Bẫy.} Đáp án D bỏ qua điều kiện xác định của căn. Đáp án A bỏ qua điều kiện $x>0$ khi bình phương.
\end{dapan}

\begin{dapan}[12 — Đáp án B]
\textbf{Giải.} Áp dụng bất đẳng thức Cauchy – Schwarz (hoặc đặt $x = \cos t$, $y = \sin t$):

$$(x+y)^2 \le 2(x^2 + y^2) = 2 \implies x + y \le \sqrt{2}$$

Dấu bằng khi $x = y = \dfrac{\sqrt{2}}{2}$.
\end{dapan}

\begin{dapan}[13 — Đáp án A]
\textbf{Giải.} Tương tự Câu 12 với cặp hệ số $(2, 3)$:

$$(2a + 3b)^2 \le (2^2 + 3^2)(a^2 + b^2) = 13 \implies 2a + 3b \le \sqrt{13}$$

Dấu bằng khi $\dfrac{a}{2} = \dfrac{b}{3}$.

\textbf{Công thức tổng quát.} Với $a^2 + b^2 = 1$, giá trị lớn nhất của $pa + qb$ là $\sqrt{p^2 + q^2}$. Nhớ công thức này để xử lý trong vài giây.
\end{dapan}

\begin{dapan}[14 — Đáp án D]
\textbf{Giải.} Mệnh đề "đúng với mọi $x$" là \textbf{sai}, nên tìm điều kiện để nó \textbf{không} đúng với mọi $x$.

Trước hết tìm điều kiện để mệnh đề \textbf{đúng}:

\begin{itemize}[leftmargin=1.4em, itemsep=2pt]
\item $a = 0$: biểu thức thành $4 > 0$, luôn đúng. ✓
\item $a \neq 0$: cần $a > 0$ và $\Delta < 0$. Với $\Delta = 4a^2 - 16a = 4a(a-4) < 0 \implies 0 < a < 4$.
\end{itemize}


Vậy mệnh đề đúng khi $0 \le a < 4$. Mệnh đề \textbf{sai} khi $a < 0$ hoặc $a \ge 4$.
\end{dapan}

\begin{dapan}[15 — Đáp án B]
\textbf{Giải.} Xét từng mệnh đề với $a, b, c$ là ba cạnh của một tam giác (tức $a+b>c$, $b+c>a$, $c+a>b$).

\textbf{(1) Đúng.} Với $a, b > 0$: $\sqrt{a} + \sqrt{b} \ge \sqrt{a+b} > \sqrt{c}$ (bất đẳng thức đầu đúng vì $(\sqrt{a}+\sqrt{b})^2 = a+b+2\sqrt{ab} \ge a+b$). Tương tự cho các cặp còn lại.

\textbf{(2) Sai.} Phản ví dụ $a = 2, b = 3, c = 4$: tam giác hợp lệ vì $2+3 > 4$, nhưng $a^2 + b^2 = 4 + 9 = 13 < 16 = c^2$. Không tạo thành tam giác.

\textbf{(3) Đúng.} Kiểm tra: $\dfrac{a+b}{2} + \dfrac{b+c}{2} = \dfrac{a + 2b + c}{2} > \dfrac{a+c}{2}$ vì $b > 0$. Vậy tổng hai cạnh bất kỳ luôn lớn hơn cạnh thứ ba.

\textbf{(4) Sai.} Phản ví dụ $a = 1{,}1$, $b = 2$, $c = 2{,}9$: tam giác hợp lệ vì $1{,}1 + 2 = 3{,}1 > 2{,}9$. Nhưng $ab + ca = 2{,}2 + 3{,}19 = 5{,}39 < 5{,}8 = bc$. Không tạo thành tam giác.

Vậy có \textbf{2} mệnh đề đúng là (1) và (3).
\end{dapan}
