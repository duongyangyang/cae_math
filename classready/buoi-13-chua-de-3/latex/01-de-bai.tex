\section{PHẦN I — VECTOR VÀ SỐ PHỨC (12 câu)}

\begin{baitap}[1]
若向量 $a=(5,1)$，$b=(-2,2k)$，且 $a\perp b$，则 $k$ 的值为（ ）
\par\vspace{3pt}
\noindent A. $5$ \quad B. $-10$ \quad C. $10$ \quad D. $-5$
\end{baitap}

\begin{baitap}[2]
若复数 $(a+i)(1-i)$ 是纯虚数，则实数 $a$ 的值为（ ）
\par\vspace{3pt}
\noindent A. $0$ \quad B. $1$ \quad C. $-1$ \quad D. $2$
\end{baitap}

\begin{baitap}[3]
已知向量 $a=(1,2,-1)$，$b=(2,-1,3)$，则 $a\times b$ (向量积) 等于（ ）
\par\vspace{3pt}
\noindent A. $(5,-5,-5)$ \quad B. $(-5,5,5)$ \quad C. $(5,5,5)$ \quad D. $(-5,-5,-5)$
\end{baitap}

\begin{baitap}[4]
已知空间四边形 $ABCD$，$\overrightarrow{AB}=a$，$\overrightarrow{AD}=b$，$\overrightarrow{AC}=c$，则 $\overrightarrow{DB}$ 等于（ ）
\par\vspace{3pt}
\noindent A. $b-a$ \quad B. $a+b$ \quad C. $-a-b$ \quad D. $a-b$
\end{baitap}

\begin{baitap}[5]
若直线 $L_1$ 的方向向量是 $d_1=(1,-1,2)$，直线 $L_2$ 的方向向量是 $d_2=(2,2,1)$，则 $L_1$ 与 $L_2$ 夹角的余弦值是（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{1}{3}$ \quad B. $\dfrac{\sqrt{6}}{3}$ \quad C. $\dfrac{\sqrt{6}}{9}$ \quad D. $\dfrac{1}{9}$
\end{baitap}

\begin{baitap}[6]
复数 $\dfrac{-3-i}{1+i}+i$ 等于（ ）
\par\vspace{3pt}
\noindent A. $2+i$ \quad B. $-2+i$ \quad C. $-4+2i$ \quad D. $-2+2i$
\end{baitap}

\begin{baitap}[7]
两个向量 $a=(1,2,3)$ 和 $b=(-2,-4,-6)$ 的位置关系是（ ）
\par\vspace{3pt}
\noindent A. 平行 \quad B. 垂直 \quad C. 相交但不垂直 \quad D. 既不平行也不垂直
\end{baitap}

\begin{baitap}[8]
设向量 $a=(k,1)$，$b=(4,k)$，且 $a\parallel b$，则 $k$ 的值是（ ）
\par\vspace{3pt}
\noindent A. $2$ \quad B. $\pm 2$ \quad C. $-2$ \quad D. $4$
\end{baitap}

\begin{baitap}[9]
复数 $(3-i)\cdot i-(1+2i)$ 等于（ ）
\par\vspace{3pt}
\noindent A. $i$ \quad B. $-2-3i$ \quad C. $-2+i$ \quad D. $2-i$
\end{baitap}

\begin{baitap}[10]
已知复数 $z_1=(2+3i)(1+i)$，$z_2=1-i$，则 $z_1+z_2$ 等于（ ）
\par\vspace{3pt}
\noindent A. $-1+5i$ \quad B. $6+4i$ \quad C. $5i$ \quad D. $4i$
\end{baitap}

\begin{baitap}[11]
已知向量 $\vec{a}=(3,4)$，$\vec{b}=(1,0)$，$\vec{c}=\vec{a}+t\vec{b}$，若 $\langle\vec{a},\vec{c}\rangle=\langle\vec{b},\vec{c}\rangle$，则 $t$ 等于（ ）
\par\vspace{3pt}
\noindent A. $5$ \quad B. $-6$ \quad C. $-5$ \quad D. $6$
\end{baitap}

\begin{baitap}[12]
已知 $\lvert\overrightarrow{a}\rvert=2\sqrt{2}$，$\lvert\overrightarrow{b}\rvert=3$，$\overrightarrow{a}$ 与 $\overrightarrow{b}$ 的夹角为 $\dfrac{\pi}{4}$。若 $\overrightarrow{AB}=5\overrightarrow{a}+2\overrightarrow{b}$，$\overrightarrow{AC}=\overrightarrow{a}-3\overrightarrow{b}$，且 $D$ 为 $BC$ 的中点，则 $\lvert\overrightarrow{AD}\rvert$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{\sqrt{15}}{2}$ \quad B. $7$ \quad C. $\dfrac{15}{2}$ \quad D. $8$
\end{baitap}

\section{PHẦN II — CONIC (13 câu)}

\begin{baitap}[13]
已知平面四边形 $ABCD$ 的四个顶点都在抛物线 $y^2=ax\ (a\neq0)$ 上，其中顶点 $D(1,-2)$，$F$ 为抛物线的焦点，若 $\overrightarrow{AF}=\dfrac{1}{3}(\overrightarrow{AB}+\overrightarrow{AC})$，则 $\lvert AF\rvert+\lvert BF\rvert+\lvert CF\rvert$ 等于（ ）
\par\vspace{3pt}
\noindent A. $12$ \quad B. $9$ \quad C. $3$ \quad D. $6$
\end{baitap}

\begin{baitap}[14]

\par\vspace{3pt}
\noindent A. $\dfrac{x^2}{2}-y^2=1$ \quad B. $\dfrac{x^2}{2}-y^2=1$ 或 $\dfrac{y^2}{2}-x^2=1$ \quad C. $\dfrac{y^2}{2}-x^2=1$ \quad D. $x^2-\dfrac{y^2}{2}=1$
\end{baitap}

\begin{baitap}[15]

\begin{baitap}[16]
设 $F_1$，$F_2$ 为双曲线 $C:\dfrac{x^2}{3}-y^2=1$ 的左、右焦点，$Q$ 为双曲线右支上一点，点 $P(0,2)$。当 $\lvert QF_1\rvert+\lvert PQ\rvert$ 取最小值时，$\lvert QF_2\rvert$ 的值为（ ）

\noindent A. $\sqrt{3}+\sqrt{2}$ \quad B. $\sqrt{3}-\sqrt{2}$ \quad C. $\sqrt{6}-2$ \quad D. $\sqrt{6}+2$
\end{baitap}

\begin{baitap}[17]

\begin{baitap}[18]

\begin{baitap}[19]

\noindent A. $\dfrac{3}{5}$ \quad B. $\dfrac{4}{5}$ \quad C. $\dfrac{5}{3}$ \quad D. $\dfrac{3}{4}$

\begin{baitap}[20]
双曲线 $\dfrac{y^2}{25}-\dfrac{x^2}{144}=1$ 的焦点坐标是（ ）

\begin{baitap}[21]

\begin{baitap}[22]

\noindent A. $(\pm 4,0)$ \quad B. $(\pm 3,0)$ \quad C. $(0,\pm 3)$ \quad D. $(0,\pm 4)$

\begin{baitap}[23]
抛物线 $y^2=-6x$ 的准线方程是（ ）
\par\vspace{3pt}

\begin{baitap}[24]
设 $F_1$，$F_2$ 是椭圆 $\dfrac{x^2}{16}+\dfrac{y^2}{7}=1$ 的两个焦点，点 $P$ 在椭圆上，则 $\lvert PF_1\rvert+\lvert PF_2\rvert$ 等于（ ）

\begin{baitap}[25]
