\section{PHẦN A — LƯỢNG GIÁC}

\begin{dapan}[1 — Đáp án B]

\textbf{Giải.} $\alpha$ thuộc góc phần tư thứ ba nên $\cos\alpha < 0$. Áp dụng $\sin^2\alpha + \cos^2\alpha = 1$:

$$\cos\alpha = -\sqrt{1 - \left(\dfrac{4}{5}\right)^2} = -\sqrt{\dfrac{9}{25}} = -\dfrac{3}{5}$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[2 — Đáp án A]

\textbf{Giải.} Công thức góc bù: $\sin 150^\circ = \sin(180^\circ - 30^\circ) = \sin 30^\circ = \dfrac{1}{2}$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[3 — Đáp án A]

\textbf{Giải.} Chu kì của $y = \sin(\omega x)$ là $T = \dfrac{2\pi}{\lvert \omega \rvert}$. Với $\omega = 3$:

$$T = \dfrac{2\pi}{3}$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[4 — Đáp án B]

\textbf{Giải.} Vì $-1 \le \sin x \le 1$, biểu thức lớn nhất khi $\sin x$ nhỏ nhất:

$$y\_{\max} = 3 - 2 \cdot (-1) = 5$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[5 — Đáp án B]

\textbf{Giải.} Chia cả tử và mẫu cho $\cos\alpha$ để quy về $\tan\alpha$:

$$\dfrac{\sin\alpha + \cos\alpha}{2\sin\alpha - \cos\alpha} = \dfrac{\tan\alpha + 1}{2\tan\alpha - 1} = \dfrac{3 + 1}{6 - 1} = \dfrac{4}{5}$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[6 — Đáp án A]

\textbf{Giải.} Dạng $a\sin x + b\cos x$ có giá trị lớn nhất $\sqrt{a^2 + b^2}$. Với $a = b = 1$:

$$\sqrt{1 + 1} = \sqrt{2}$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[7 — Đáp án B]

\textbf{Giải.} Góc phần tư thứ ba cho $\sin\alpha < 0$:

$$\sin\alpha = -\sqrt{1 - \dfrac{9}{25}} = -\dfrac{4}{5}, \qquad \tan\alpha = \dfrac{\sin\alpha}{\cos\alpha} = \dfrac{-4/5}{-3/5} = \dfrac{4}{3}$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[8 — Đáp án B]

\textbf{Giải.} Bình phương đẳng thức đã cho:

$$(\sin\alpha + \cos\alpha)^2 = 1 + \sin 2\alpha = \dfrac{1}{4} \implies \sin 2\alpha = \dfrac{1}{4} - 1 = -\dfrac{3}{4}$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[9 — Đáp án D]

\textbf{Giải.} Từ $\sin\alpha = \dfrac{3}{5}$ và $\alpha$ ở góc phần tư thứ nhất, $\cos\alpha = \dfrac{4}{5}$. Công thức nhân đôi:

$$\sin 2\alpha = 2\sin\alpha\cos\alpha = 2 \cdot \dfrac{3}{5} \cdot \dfrac{4}{5} = \dfrac{24}{25}$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[10 — Đáp án D]

\textbf{Giải.} Dùng công thức hạ bậc ở tử:

$$1 - \cos 2\alpha = 2\sin^2\alpha, \qquad \sin 2\alpha = 2\sin\alpha\cos\alpha$$

$$\dfrac{1 - \cos 2\alpha}{\sin 2\alpha} = \dfrac{2\sin^2\alpha}{2\sin\alpha\cos\alpha} = \dfrac{\sin\alpha}{\cos\alpha} = \tan\alpha$$

\begin{dapan}[13 — Đáp án B]

\begin{dapan}[14 — Đáp án D]

\begin{dapan}[15 — Đáp án B]

\begin{dapan}[16 — Đáp án C]

\textbf{Giải.} Với $\dfrac{x^2}{25} + \dfrac{y^2}{9} = 1$: $a = 5$, $b = 3$, nên $c = \sqrt{25 - 9} = 4$. Khoảng cách giữa hai tiêu điểm là $2c = 8$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[17 — Đáp án C]

\textbf{Giải.} Với $\dfrac{x^2}{9} - \dfrac{y^2}{16} = 1$: $a = 3$, $b = 4$, $c = \sqrt{9 + 16} = 5$. Tâm sai $e = \dfrac{c}{a} = \dfrac{5}{3}$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[18 — Đáp án D]

\textbf{Giải.} Tâm đường tròn $x^2 + y^2 + Dx + Ey + F = 0$ là $\left(-\dfrac{D}{2}, -\dfrac{E}{2}\right)$. Với $D = -2$, $E = 4$, tâm là $(1, -2)$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[19 — Đáp án D]

\textbf{Giải.} Đường $4x + 3y - 5 = 0$ có hệ số góc $-\dfrac{4}{3}$, nên đường vuông góc có hệ số góc $\dfrac{3}{4}$.

$$y - 1 = \dfrac{3}{4}(x - 2) \iff 3x - 4y - 2 = 0$$

\begin{dapan}[26 — Đáp án C]

\end{dapan}

\begin{dapan}[27 — Đáp án A]

\textbf{Giải.} Trục đối xứng của parabol $y = ax^2 + bx + c$ là $x = -\dfrac{b}{2a}$. Với $a = 1$, $b = -6$:

$$x = \dfrac{6}{2} = 3$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[28 — Đáp án D]

\textbf{Giải.} $f'(x) = 3x^2 - 6x = 3x(x - 2)$. Trên $[0, 2]$ đạo hàm âm trong $(0, 2)$, nên hàm nghịch biến và giá trị nhỏ nhất tại $x = 2$:

$$f(2) = 8 - 12 + 1 = -3$$


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[29 — Đáp án D]

\textbf{Giải.} $f(2) = 3 \cdot 2 - 2 = 4$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[30 — Đáp án A]

\textbf{Giải.} Hàm logarit cần biểu thức trong dấu logarit dương: $x - 1 > 0$, tức $x > 1$. Tập xác định là $(1, +\infty)$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[31 — Đáp án C]

\textbf{Giải.} Xét $y = \dfrac{2x}{x^2 + 1}$. Đặt $y = k$, phương trình $kx^2 - 2x + k = 0$ phải có nghiệm, nên $\Delta' = 1 - k^2 \ge 0$, tức $-1 \le k \le 1$. Giá trị lớn nhất là $1$ (đạt tại $x = 1$).
