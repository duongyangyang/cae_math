\noindent A. $(-1,3,-5)$，$3$ \quad B. $(-1,3,-5)$，$9$ \quad C. $(1,-3,5)$，$3$ \quad D. $(1,-3,5)$，$9$
\end{baitap}

\begin{baitap}[2]

\noindent A. $\dfrac{4\sqrt{6}}{3}$ \quad B. $6\sqrt{6}$ \quad C. $4\sqrt{6}$ \quad D. $\dfrac{8\sqrt{6}}{3}$

\begin{baitap}[3]

\begin{baitap}[5]

\begin{baitap}[6]

\begin{baitap}[8]

\begin{baitap}[9]

\section{PHẦN IV — ÔN TẬP TÍCH LŨY (22 câu)}

\begin{baitap}[27]
如果集合 $A = \{1, a\}$，$B = \{1, 3, 6, 7, 9\}$，且 $A \subseteq B$，则实数 $a$ 可以取值的个数是（ ）
\par\vspace{3pt}
\noindent A. 0 \quad B. 3 \quad C. 4 \quad D. 5
\end{baitap}

\begin{baitap}[28]
已知集合 $A = \{a, b, c\}$，则下列不正确的关系是（ ）
\par\vspace{3pt}
\noindent A. $a \in A$ \quad B. $\{a\} \subseteq A$ \quad C. $\varnothing \subseteq A$ \quad D. $c \notin A$
\end{baitap}

\begin{baitap}[29]
已知 $\cos\alpha = -\dfrac{5}{13}$，且角 $\alpha$ 属于第三象限，则下列说法正确的是（ ）
\par\vspace{3pt}
\noindent A. $\sin\alpha = \dfrac{12}{13}$ \quad B. $\sin\alpha = -\dfrac{12}{13}$ \quad C. $\tan\alpha = \dfrac{5}{12}$ \quad D. $\tan\alpha = -\dfrac{12}{5}$
\end{baitap}

\begin{baitap}[30]
在等差数列 $\{a_n\}$ 中，若 $a_3 = 4$ 且 $a_6 = 13$，则首项 $a_1$ 和公差 $d$ 分别为（ ）
\par\vspace{3pt}
\noindent A. $-2$，$3$ \quad B. $2$，$-3$ \quad C. $-5$，$3$ \quad D. $5$，$-3$
\end{baitap}

\begin{baitap}[31]
某兴趣小组有 $30$ 名成员，其中 $18$ 人会唱歌，$15$ 人会跳舞。如果 $5$ 人既会唱歌又会跳舞，则既不会唱歌也不会跳舞的人数是（ ）
\par\vspace{3pt}
\noindent A. 1 \quad B. 2 \quad C. 4 \quad D. 3
\end{baitap}

\begin{baitap}[32]
若 $a > b > 0$，则下列不等式中正确的是（ ）
\par\vspace{3pt}
\noindent A. $a^2 < b^2$ \quad B. $\dfrac{1}{a} > \dfrac{1}{b}$ \quad C. $a - b < 0$ \quad D. $\dfrac{a}{b} > 1$
\end{baitap}

\begin{baitap}[33]
已知函数 $f(x) = \ln x + x^2$。求曲线 $y = f(x)$ 在点 $(1, f(1))$ 处的切线方程是（ ）
\par\vspace{3pt}
\noindent A. $y = 2x - 1$ \quad B. $y = 3x - 1$ \quad C. $y = x + 1$ \quad D. $y = 3x - 2$
\end{baitap}

\begin{baitap}[34]
若关于 $x$ 的不等式 $x^2 + (m-1)x + 1 > 0$ 对于一切实数 $x$ 恒成立，则实数 $m$ 的取值范围是（ ）
\par\vspace{3pt}
\noindent A. $(-1,3)$ \quad B. $[-1,3]$ \quad C. $(-\infty,-1) \cup (3,+\infty)$ \quad D. $(-\infty,-1] \cup [3,+\infty)$
\end{baitap}

\begin{baitap}[35]
已知全集 $U = \{x \in \mathbb{Z} \mid 5 \le x \le 15\}$，集合 $A = \{x \mid x^2 - 6x + 5 \le 0\}$，集合 $B = \{x \mid \log_2(x-6) < 2\}$，则 $A \cup (\complement_U B)$ 的元素个数是（ ）
\par\vspace{3pt}
\noindent A. 8 \quad B. 9 \quad C. 10 \quad D. 11
\end{baitap}

\begin{baitap}[36]
若 $a, b \in \mathbb{R}$，则 $a^2 + b^2 + 1 \ge ab + a + b$ 恒成立的理由是（ ）
\par\vspace{3pt}
\noindent A. 基本不等式 \quad B. 柯西不等式 \quad C. 配方 \quad D. 均值不等式
\end{baitap}

\begin{baitap}[37]
已知全集 $U = \{1,2,3,4,5,6\}$，集合 $A = \{x \mid x^2 - 5x + 6 = 0\}$，集合 $B = \{x \mid x \text{ 是偶数}\}$，则 $\complement_U (A \cap B)$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\{1,3,4,5,6\}$ \quad B. $\{2\}$ \quad C. $\{1,3,5\}$ \quad D. $\{1,4,6\}$
\end{baitap}

\begin{baitap}[38]
已知函数 $f(x) = x + 2$（当 $x < 0$）和 $f(x) = x^2$（当 $x \ge 0$）。求不等式 $f(x) < 4$ 的解集等于（ ）

\begin{baitap}[39]

\begin{baitap}[40]
