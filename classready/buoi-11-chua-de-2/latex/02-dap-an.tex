\section{PHẦN I — HÌNH HỌC PHẲNG VÀ KHÔNG GIAN}

\begin{dapan}[1 — Đáp án C]

\textbf{Giải.} Phương trình mặt cầu $(x-a)^2 + (y-b)^2 + (z-c)^2 = R^2$ có tâm $(a,b,c)$ và bán kính $R$.

Đối chiếu: tâm $(1,-3,5)$, bán kính $\sqrt{9} = 3$. Dấu trừ trong ngoặc đổi dấu tọa độ, dấu cộng giữ nguyên.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[2 — Đáp án A]

\textbf{Giải.} Công thức khoảng cách từ điểm đến mặt phẳng: $d = \dfrac{\lvert Ax_0+By_0+Cz_0+D \rvert}{\sqrt{A^2+B^2+C^2}}$.

Tử số: $\lvert 2 + 6 + 1 - 1 \rvert = 8$. Mẫu số: $\sqrt{1+4+1} = \sqrt{6}$.

$d = \dfrac{8}{\sqrt{6}} = \dfrac{8\sqrt{6}}{6} = \dfrac{4\sqrt{6}}{3}$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[3 — Đáp án D]

\textbf{Giải.} Pháp tuyến của $P_1$ là $(1,1,1)$, của $P_2$ là $(2,2,2)$. Hai vectơ tỉ lệ nên hai mặt phẳng song song hoặc trùng nhau.

So sánh hằng số: đưa $P_2$ về dạng $x+y+z = \dfrac{3}{2}$. Vì $\dfrac{3}{2} \ne 1$ nên hai mặt phẳng không trùng. Vậy chúng song song.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[4 — Đáp án C]

\textbf{Giải.} Tử số: $\lvert 2 \cdot 1 - (-2) + 2 \cdot 2 + 3 \rvert = \lvert 2 + 2 + 4 + 3 \rvert = 11$.

Mẫu số: $\sqrt{4+1+4} = 3$.

$d = \dfrac{11}{3}$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[5 — Đáp án A]

\textbf{Giải.} Mặt cầu nhận $MN$ làm đường kính có tâm là trung điểm $I$ của $MN$ và bán kính $R = \dfrac{MN}{2}$.

$I = \left(\dfrac{3-1}{2}, \dfrac{-2+6}{2}, \dfrac{5-3}{2}\right) = (1,2,1)$.

$MN = \sqrt{(-4)^2 + 8^2 + (-8)^2} = \sqrt{16+64+64} = \sqrt{144} = 12$, suy ra $R = 6$, $R^2 = 36$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[6 — Đáp án C]

\textbf{Giải.} Đối xứng qua trục $x$ thì hoành độ giữ nguyên, tung độ và cao độ đổi dấu.

Từ $A'(\lambda, 7, -6)$ suy ra $\lambda = 2$; $-(3-\mu) = 7$ nên $\mu = 10$; $-(-1+\nu) = -6$ nên $\nu = 7$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[7 — Đáp án B]

\textbf{Giải.} Cao độ $z = 0$ nên điểm nằm trong mặt phẳng $xOy$.

Nhận dạng nhanh: tọa độ nào bằng $0$ thì điểm thuộc mặt phẳng tạo bởi hai trục còn lại. Ở đây $z = 0$ nên thuộc mặt $xOy$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[8 — Đáp án D]

\textbf{Giải.} Trung điểm $M$ của $BC$: $M = \left(\dfrac{2-2}{2}, \dfrac{2+2}{2}, \dfrac{1+1}{2}\right) = (0,2,1)$.

Trung tuyến từ $O$ có độ dài $OM = \sqrt{0 + 4 + 1} = \sqrt{5}$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[9 — Đáp án B]

\textbf{Giải.} $M$ là trung điểm $BC$: $M = \left(\dfrac{5+1}{2}, \dfrac{0+2}{2}, \dfrac{1+3}{2}\right) = (3,1,2)$.

$AM = \sqrt{(3-1)^2 + (1-2)^2 + (2-0)^2} = \sqrt{4+1+4} = 3$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[10 — Đáp án B]

\textbf{Giải.} Trung điểm $M$ của $BC$: $M = \left(\dfrac{2+2}{2}, \dfrac{2+0}{2}, \dfrac{0+2}{2}\right) = (2,1,1)$.

$AM = \sqrt{(2-1)^2 + (1-0)^2 + (1-1)^2} = \sqrt{1+1+0} = \sqrt{2}$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[11 — Đáp án C]

\textbf{Giải.} Mọi mặt phẳng cắt một mặt cầu đều cho thiết diện là hình tròn.

Ba hình còn lại đều có thiết diện dạng khác: hình trụ có thể cho hình chữ nhật, hình nón có thể cho tam giác cân, hình nón cụt có thể cho hình thang cân.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[12 — Đáp án D]

\textbf{Giải.} Pháp tuyến vuông góc với mọi vectơ nằm trong mặt phẳng, nên $\vec{n} \cdot \overrightarrow{MN} = 0$.

$\overrightarrow{MN} = (a-1, 4, 1)$. Tích vô hướng: $6(a-1) - 3 \cdot 4 + 6 \cdot 1 = 6a - 6 - 12 + 6 = 6a - 12$.

Cho bằng $0$ được $a = 2$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[13 — Đáp án C]

\textbf{Giải.} Đối xứng qua trục $y$: hoành độ và cao độ đổi dấu, tung độ giữ nguyên.

Từ $(2,0,22)$ được $(-2,0,-22)$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[14 — Đáp án B]

\textbf{Giải.} Từ $\overrightarrow{OM} = \overrightarrow{AB}$ suy ra $B = A + \overrightarrow{OM}$.

$B = (4,2,-1) + (5,-1,2) = (9,1,1)$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[15 — Đáp án C]

\textbf{Giải.} Hình lăng trụ đứng có đáy là hình vuông gọi là hình lăng trụ tứ giác đều.

Điều kiện "đáy là hình thoi và ba cạnh tại một đỉnh đôi một vuông góc" buộc đáy phải là hình vuông và cạnh bên vuông góc với đáy, tức đúng là lăng trụ tứ giác đều.

Các phương án còn lại đều thiếu: đáy hình vuông với hai mặt bên là hình chữ nhật vẫn có thể là lăng trụ xiên.

\begin{dapan}[16 — Đáp án C]

\textbf{Giải.} Giao với trục $y$ là cho $x = 0$: $-3y + 6 = 0$ suy ra $y = 2$.

Đừng nhầm với hoành độ gốc: cho $y = 0$ được $x = -3$, ứng với phương án A.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[17 — Đáp án B]

\textbf{Giải.} Góc nghiêng $45^\circ$ cho hệ số góc $k = \tan 45^\circ = 1$.

Phương trình qua $(0,-1)$: $y - (-1) = 1 \cdot (x - 0)$, tức $y + 1 = x$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[18 — Đáp án A]

\textbf{Giải.} $PQ = \sqrt{(2+1)^2 + (-4-2)^2} = \sqrt{9 + 36} = \sqrt{45} = 3\sqrt{5}$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[19 — Đáp án A]

\textbf{Giải.} Với $x^2+y^2+Dx+Ey+F=0$, tâm $\left(-\dfrac{D}{2}, -\dfrac{E}{2}\right)$ và $r = \sqrt{\dfrac{D^2+E^2}{4} - F}$.

Tâm $O(4,-1)$. Bán kính $r = \sqrt{4^2 + (-1)^2 - (-8)} = \sqrt{16+1+8} = \sqrt{25} = 5$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[20 — Đáp án C]

\textbf{Giải.} Phương trình đường tròn tâm $(a,b)$ bán kính $r$: $(x-a)^2 + (y-b)^2 = r^2$.

Với $(a,b) = (2,-5)$ và $r = 3$: $(x-2)^2 + (y+5)^2 = 9$.

Bẫy: phương án A đúng tâm nhưng quên bình phương bán kính.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[21 — Đáp án D]

\textbf{Giải.} Dây chung $AB$ vuông góc với đường nối hai tâm, nên đường trung trực của $AB$ chính là đường nối hai tâm.

Tâm $O_1(2,-3)$, tâm $O_2(1,0)$. Hệ số góc $\dfrac{0-(-3)}{1-2} = -3$.

Phương trình qua $O_2$: $y = -3(x-1)$, tức $3x + y - 3 = 0$.

\section{PHẦN III — CONIC}


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[22 — Đáp án B]

\textbf{Giải.} Parabol $x^2 = 2py$ có tiêu điểm $\left(0, \dfrac{p}{2}\right)$.

Với $2p = 12$ suy ra $p = 6$, tiêu điểm $(0,3)$.

Bẫy: hệ số $12$ là $2p$ chứ không phải $p$, nên tiêu điểm không phải $(0,6)$. Phương án A và C đặt tiêu điểm sai trục.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[23 — Đáp án D]

\textbf{Giải.} Parabol $y^2 = 2px$ có tiêu điểm $\left(\dfrac{p}{2}, 0\right)$ và đường chuẩn $x = -\dfrac{p}{2}$.

Với $2p = 8$ suy ra $p = 4$, đường chuẩn $x = -2$.

Bẫy: phương án A lấy $x = +2$ vì quên dấu trừ. Phương án B và C đặt đường chuẩn theo trục $y$, sai vì parabol này mở theo trục $x$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[24 — Đáp án C]

\textbf{Giải.} Khoảng cách từ $P$ đến trục $x$ là $6$, tức tung độ $y_P = 6$.

Theo định nghĩa parabol, khoảng cách đến tiêu điểm bằng khoảng cách đến đường chuẩn: $y_P + \dfrac{p}{2} = 8$, suy ra $\dfrac{p}{2} = 2$, $p = 4$.

Phương trình: $x^2 = 2py = 8y$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[25 — Đáp án A]

\textbf{Giải.} Hypebol $\dfrac{x^2}{a^2} - \dfrac{y^2}{b^2} = 1$ có $c^2 = a^2 + b^2$ và tâm sai $e = \dfrac{c}{a}$.

$a^2 = 9$, $b^2 = 16$, nên $c = \sqrt{25} = 5$ và $e = \dfrac{5}{3}$.

Bẫy: tâm sai luôn lớn hơn $1$ với hypebol, nên hai phương án có giá trị nhỏ hơn $1$ bị loại ngay.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[26 — Đáp án B]

\textbf{Giải.} Đưa về dạng chính tắc: $\dfrac{x^2}{9} + \dfrac{y^2}{4} = 1$.

Trục lớn trên $Oy$ vì $9 > 4$, nên $a^2 = 9$, $b^2 = 4$, $c^2 = 5$, tiêu điểm $(0,\pm\sqrt{5})$.

Ellipse cần tìm có cùng tiêu điểm nên trục lớn cũng trên $Oy$. Nửa trục nhỏ bằng $1$ (trục nhỏ dài $2$), suy ra $a^2 = c^2 + b^2 = 5 + 1 = 6$.

Phương trình: $\dfrac{y^2}{6} + x^2 = 1$.

\section{PHẦN IV — ÔN TẬP TÍCH LŨY}


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[27 — Đáp án C]

\textbf{Giải.} $A \subseteq B$ nghĩa là mọi phần tử của $A$ đều thuộc $B$.

Phần tử $1$ đã thuộc $B$, nên $a$ chỉ cần thuộc $B$ và khác $1$. Các giá trị: $3, 6, 7, 9$, tổng cộng $4$ giá trị.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[28 — Đáp án D]

\textbf{Giải.} $c$ là một phần tử của $A$, nên $c \in A$ đúng và $c \notin A$ sai. Đó là quan hệ không đúng cần tìm.

Ba quan hệ còn lại đều đúng: phần tử thuộc tập, tập con một phần tử nằm trong tập, và tập rỗng là tập con của mọi tập.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[29 — Đáp án B]

\textbf{Giải.} Góc phần tư thứ ba nên $\sin\alpha < 0$ và $\tan\alpha > 0$.

$\sin^2\alpha = 1 - \dfrac{25}{169} = \dfrac{144}{169}$, suy ra $\sin\alpha = -\dfrac{12}{13}$.

Kiểm tra nhanh $\tan\alpha$: $\dfrac{-12/13}{-5/13} = \dfrac{12}{5}$, dương, phù hợp góc phần tư thứ ba. Hai phương án C và D đều sai dấu.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[30 — Đáp án A]

\textbf{Giải.} $a_6 - a_3 = 3d$, tức $13 - 4 = 3d$, suy ra $d = 3$.

$a_1 = a_3 - 2d = 4 - 6 = -2$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[31 — Đáp án D]

\textbf{Giải.} Số người biết ít nhất một môn: $18 + 15 - 5 = 28$.

Số người không biết môn nào: $30 - 28 = 2$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[32 — Đáp án D]

\textbf{Giải.} Với $a > b > 0$, chia hai vế cho $b > 0$ được $\dfrac{a}{b} > 1$.

Ba phương án còn lại sai: bình phương hai vế của $a > b > 0$ phải cho $a^2 > b^2$; nghịch đảo phải đổi chiều thành $\dfrac{1}{a} < \dfrac{1}{b}$; và $a - b > 0$ chứ không phải $< 0$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[33 — Đáp án D]

\textbf{Giải.} $f(1) = \ln 1 + 1 = 1$, tiếp điểm $(1,1)$.

$f'(x) = \dfrac{1}{x} + 2x$, nên $f'(1) = 3$.

Phương trình tiếp tuyến: $y - 1 = 3(x-1)$, tức $y = 3x - 2$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[34 — Đáp án A]

\textbf{Giải.} Tam thức có hệ số bậc hai dương, nên dương với mọi $x$ khi và chỉ khi biệt thức âm.

$\Delta = (m-1)^2 - 4 < 0$, tức $-2 < m-1 < 2$, suy ra $-1 < m < 3$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[35 — Đáp án A]

\textbf{Giải.} $A = \{x \mid 1 \le x \le 5\}$, giao với $U$ cho $A = \{5\}$.

$B$: $\log_2(x-6) < 2$ tương đương $0 < x - 6 < 4$, tức $6 < x < 10$. Trong $U$: $B = \{7,8,9\}$, nên $\complement_U B = \{5,6,10,11,12,13,14,15\}$.

$A \cup \complement_U B = \{5,6,10,11,12,13,14,15\}$, có $8$ phần tử.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[36 — Đáp án C]

\textbf{Giải.} Nhân hai vế với $2$: $2a^2 + 2b^2 + 2 \ge 2ab + 2a + 2b$.

Chuyển hết sang trái và nhóm lại: $(a-b)^2 + (a-1)^2 + (b-1)^2 \ge 0$.

Cách chứng minh là biến đổi tương đương, tức phương pháp tách thành tổng các bình phương.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[37 — Đáp án A]

\textbf{Giải.} $A$: $x^2 - 5x + 6 = 0$ cho $x = 2$ hoặc $x = 3$, tức $A = \{2,3\}$.

$B$ là tập các số chẵn trong $U$: $B = \{2,4,6\}$.

$A \cap B = \{2\}$, nên $\complement_U (A \cap B) = \{1,3,4,5,6\}$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[38 — Đáp án D]

\textbf{Giải.} Xét hai nhánh của hàm.

Với $x < 0$: $x + 2 < 4$ cho $x < 2$, luôn đúng trên $(-\infty,0)$.

Với $x \ge 0$: $x^2 < 4$ cho $0 \le x < 2$.

Hợp hai nhánh: $(-\infty, 2)$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[39 — Đáp án B]

\textbf{Giải.} $a$ âm và $b$ dương nên tích $ab$ âm, tức $ab < 0$.

Phương án A sai vì $\sqrt{a}$ không xác định khi $a < 0$. So sánh $a^2$ với $b^2$ không kết luận được nếu chưa biết độ lớn của chúng.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[40 — Đáp án D]

\textbf{Giải.} $\log_a(MN) = \log_a M + \log_a N$ đúng với $M, N > 0$.

Ba đẳng thức còn lại đều sai: logarit của tích không phải tích các logarit; logarit của tổng không tách được; và $\log_a M^n = n\log_a M$ chứ không phải $(\log_a M) \cdot n$ viết sai vị trí.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[41 — Đáp án A]

\textbf{Giải.} Nguyên hàm: $f(x) = x^3 - x + C$.

Điều kiện $f(1) = 0$ cho $1 - 1 + C = 0$, suy ra $C = 0$.

Vậy $f(x) = x^3 - x$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[42 — Đáp án B]

\textbf{Giải.} Tập nghiệm có đúng một phần tử xảy ra theo hai khả năng.

Nếu phương trình là bậc nhất: $k - 1 = 0$, tức $k = 1$, khi đó $-6x + 6 = 0$ cho $x = 1$.

Nếu là bậc hai có nghiệm kép: $\Delta = (k+5)^2 - 24(k-1) = k^2 - 14k + 49 = (k-7)^2 = 0$, tức $k = 7$.

Tổng các giá trị: $1 + 7 = 8$.


\end{dapan}
\vspace{0.3cm}
\begin{dapan}[43 — Đáp án B]

\textbf{Giải.} $a_7 - a_3 = 4d$, tức $21 - 5 = 4d$, suy ra $d = 4$.


\end{dapan}

\begin{dapan}[44 — Đáp án A]

\begin{dapan}[47 — Đáp án B]

\textbf{Giải.} Bất đẳng thức tam giác cho giá trị tuyệt đối: $\lvert a + b \rvert \le \lvert a \rvert + \lvert b \rvert$.

Chiều ngược lại $\lvert a + b \rvert \ge \lvert a \rvert + \lvert b \rvert$ chỉ xảy ra khi $a$ và $b$ cùng dấu hoặc một trong hai bằng $0$, nên không đúng với mọi số thực. Đó là nhận định sai cần tìm.


\end{dapan}

\begin{dapan}[48 — Đáp án A]

\textbf{Giải.} Từ giả thiết, $a_n - 3 = 4^{n-1}$, nên $a_n = 4^{n-1} + 3$. Kiểm tra: $a_1 = 4$, $a_2 = 6$, $a_3 = 15$, phù hợp hệ thức truy hồi.

Với $n = 1$: $b_1 = \dfrac{9 \cdot 1}{4 \cdot 6} = \dfrac{3}{8}$, $T_1 = \dfrac{3}{8}$. Khi đó $a_2 - 3 = 3$ và $\lambda = 5 \cdot 3 \cdot \left(\dfrac{7}{8} - \dfrac{3}{8}\right) = 15 \cdot \dfrac{1}{2} = \dfrac{15}{2}$, không nguyên.

Với $n = 2$: $b_2 = \dfrac{9 \cdot 3}{6 \cdot 15} = \dfrac{3}{10}$, $T_2 = \dfrac{3}{8} + \dfrac{3}{10} = \dfrac{27}{40}$. Khi đó $a_3 - 3 = 12$ và $\lambda = 5 \cdot 12 \cdot \left(\dfrac{35}{40} - \dfrac{27}{40}\right) = 60 \cdot \dfrac{1}{5} = 12$, là số nguyên.

Suy ra $\lambda n = 12 \cdot 2 = 24$.

Với $n = 3$: $b_3 = \dfrac{9 \cdot 12}{15 \cdot 51} = \dfrac{12}{85}$, $T_3 = \dfrac{27}{40} + \dfrac{12}{85} = \dfrac{111}{136}$. Khi đó $a_4 - 3 = 48$ và $\lambda = 5 \cdot 48 \cdot \left(\dfrac{119}{136} - \dfrac{111}{136}\right) = 240 \cdot \dfrac{1}{17} = \dfrac{240}{17}$, không nguyên.

Vậy giá trị nguyên đầu tiên đạt được ở $n = 2$, và $\lambda n = 24$.
\end{dapan}
