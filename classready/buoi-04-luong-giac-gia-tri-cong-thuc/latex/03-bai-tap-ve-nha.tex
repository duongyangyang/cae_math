\subsection*{NHÓM A — GIÁ TRỊ VÀ CÔNG THỨC LƯỢNG GIÁC}

\begin{baitap}[1]

\noindent A. $\dfrac{3}{2}$ \quad B. $-\dfrac{1}{3}$ \quad C. $\dfrac{1}{2}$ \quad D. $\dfrac{1}{3}$
\end{baitap}

\begin{baitap}[2]

\par\vspace{3pt}
\noindent A. $[0,1]$ \quad B. $[-1,1]$ \quad C. $[-2,2]$ \quad D. $(-\infty, +\infty)$
\end{baitap}

\begin{baitap}[3]
已知 $\sin\alpha = -\dfrac{1}{2}$ ，且 $\alpha$ 为第三象限角，则 $\cos\alpha$ 的值是（ ）
\par\vspace{3pt}
\noindent A. $-\dfrac{\sqrt{3}}{2}$ \quad B. $\dfrac{\sqrt{3}}{2}$ \quad C. $\dfrac{1}{2}$ \quad D. $-\dfrac{1}{2}$
\end{baitap}

\begin{baitap}[4]

\par\vspace{3pt}
\noindent A. 1 \quad B. $\sqrt{2}$ \quad C. 2 \quad D. $\dfrac{\sqrt{2}}{2}$
\end{baitap}

\begin{baitap}[5]
$\cos\!\left(-\dfrac{\pi}{6}\right)$ 的值是（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{1}{2}$ \quad B. $-\dfrac{\sqrt{3}}{2}$ \quad C. $\dfrac{\sqrt{3}}{2}$ \quad D. $-\dfrac{1}{2}$
\end{baitap}

\begin{baitap}[6]
求函数 $f(x) = \sin\!\left(2x + \dfrac{\pi}{4}\right)$ 的导数 $f'(x)$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\cos\!\left(2x + \dfrac{\pi}{4}\right)$ \quad B. $-2\cos\!\left(2x + \dfrac{\pi}{4}\right)$ \quad C. $-\cos\!\left(2x + \dfrac{\pi}{4}\right)$ \quad D. $2\cos\!\left(2x + \dfrac{\pi}{4}\right)$
\end{baitap}

\begin{baitap}[7]
化简 $\dfrac{\sin^2 x}{1 - \cos x}\ (\cos x \ne 1)$ 的结果是（ ）
\par\vspace{3pt}
\noindent A. $\sin x$ \quad B. $1 - \cos x$ \quad C. $1 + \cos x$ \quad D. $\cos x$
\end{baitap}

\begin{baitap}[8]

\par\vspace{3pt}
\noindent A. 增函数 \quad B. 减函数 \quad C. 不单调 \quad D. 既不增也不减
\end{baitap}

\begin{baitap}[9]

\par\vspace{3pt}
\noindent A. y 轴 \quad B. 原点 \quad C. x 轴 \quad D. 直线 $y = x$
\end{baitap}

\begin{baitap}[10]
$\tan 2\alpha$ 等于（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{2\tan\alpha}{1-\tan^2\alpha}$ \quad B. $\dfrac{2\tan\alpha}{1+\tan^2\alpha}$ \quad C. $\dfrac{1-\tan^2\alpha}{2\tan\alpha}$ \quad D. $\dfrac{1+\tan^2\alpha}{2\tan\alpha}$
\end{baitap}

\begin{baitap}[11]
若 $\tan\alpha = 1$ ，且 $\alpha$ 是第三象限角，则 $\sin\alpha$ 等于（ ）

\begin{baitap}[12]

\subsection*{NHÓM B — HÀM BẬC HAI, MŨ, LOGARIT, ĐẠO HÀM}

\begin{baitap}[14]
已知 $f(x) = \log_a x\ (a > 0, a \ne 1)$ ，若 $f(16) = 2$，则 $a$ 的值为（ ）
\par\vspace{3pt}
\noindent A. $\sqrt{2}$ \quad B. 2 \quad C. 4 \quad D. 16
\end{baitap}

\begin{baitap}[15]
已知 $\log_{7}x = 2$ ，则 $x$ 的值是（ ）
\par\vspace{3pt}
\noindent A. 14 \quad B. 49 \quad C. $7^2$ \quad D. 10
\end{baitap}

\begin{baitap}[16]
已知函数 $f(x) = \log_a(x+1)$，若 $f(1)=1$，则 $a =$（ ）
\par\vspace{3pt}
\noindent A. 0 \quad B. 1 \quad C. 2 \quad D. 3
\end{baitap}

\begin{baitap}[17]
假设某物品的价值每年折旧 5\%，如果原价是 1000 元，则 $t$ 年后该物品的价值 $V(t)$ 的函数表达式是（ ）
\par\vspace{3pt}
\noindent A. $V(t) = 1000(1 - 0.05t)$ \quad B. $V(t) = 1000(1 - 0.05)^t$ \quad C. $V(t) = 1000 - 0.05t$ \quad D. $V(t) = 1000(0.05)^t$
\end{baitap}

\begin{baitap}[18]
已知 $x_0$ 是函数 $f(x) = 2^x + x - 7$ 的一个零点，若 $x_1 \in (-1, x_0)$，$x_2 \in (x_0, +\infty)$，则（ ）
\par\vspace{3pt}
\noindent A. $f(x_1) > 0, f(x_2) < 0$ \quad B. $f(x_1) < 0, f(x_2) > 0$ \quad C. $f(x_1) > 0, f(x_2) > 0$ \quad D. $f(x_1) < 0, f(x_2) < 0$
\end{baitap}

\begin{baitap}[19]
函数 $f(x) = (x^2 + 5)(x - 1)$ 的导数 $f'(x)$ 在 $x = 2$ 处的值是（ ）
\par\vspace{3pt}
\noindent A. 26 \quad B. 9 \quad C. 13 \quad D. 10
\end{baitap}

\subsection*{NHÓM C — TÍNH CHẤT HÀM SỐ}

\begin{baitap}[20]

\par\vspace{3pt}
\noindent A. 0 \quad B. $\pm\sqrt{15}$ \quad C. 15 \quad D. -15
\end{baitap}

\begin{baitap}[21]
函数 $f(x) = \dfrac{x^2-1}{3x-1}$ 的定义域是（ ）
\par\vspace{3pt}
\noindent A. $\mathbb{R}$ \quad B. $\mathbb{R}$ \quad C. $\left(\dfrac{1}{3}, +\infty\right)$ \quad D. $\left(-\infty, \dfrac{1}{3}\right) \cup \left(\dfrac{1}{3}, +\infty\right)$
\end{baitap}

\begin{baitap}[22]
若 $y=f(x)\ (x\in\mathbb{R})$ 是奇函数，则下列坐标表示的点一定在 $y=f(x)$ 图象上的是（ ）
\par\vspace{3pt}
\noindent A. $(a,-f(a))$ \quad B. $(-a,-f(a))$ \quad C. $(-a,-f(-a))$ \quad D. $(a,f(-a))$
\end{baitap}

\begin{baitap}[23]
已知 $f\left(\frac{x}{2x-1}\right)=2x^2+3$，则 $f\left(\frac{1}{3}\right)=$（ ）
\par\vspace{3pt}
\noindent A. $-\frac{2}{9}$ \quad B. $-\frac{1}{4}$ \quad C. 5 \quad D. -5
\end{baitap}

\begin{baitap}[25]

\noindent A. $\{0\}$ \quad B. $\{1\}$ \quad C. $\{0,2\}$ \quad D. $\{1,2\}$
