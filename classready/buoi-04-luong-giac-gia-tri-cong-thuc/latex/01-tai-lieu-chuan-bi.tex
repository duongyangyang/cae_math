\section{PHẦN C — BÀI TẬP CHUẨN BỊ}

Làm 10 câu dưới đây \textbf{trước buổi học}. Sáu câu đầu chỉ cần đọc hiểu thuật ngữ là làm được; các câu cuối đòi hỏi biến đổi, nếu chưa làm được thì ghi lại chỗ vướng mắc để hỏi trên lớp.

\subsection{Mức dễ}

\begin{baitap}[1]
下列函数中，是偶函数的是（ ）
\par\vspace{3pt}
\noindent A. $y = x^3$ \quad B. $y = \sin x$ \quad C. $y = \cos x$ \quad D. $y = \log_{2}x$
\end{baitap}
\begin{baitap}[2]
判断函数 $f(x) = \dfrac{\sin x}{x}$ 的奇偶性是（ ）
\par\vspace{3pt}
\noindent A. 奇函数 \quad B. 偶函数 \quad C. 非奇非偶函数 \quad D. 既是奇函数又是偶函数
\end{baitap}
\begin{baitap}[3]
函数 $y = x \cdot \cos x$ 是（ ）
\par\vspace{3pt}
\noindent A. 偶函数 \quad B. 奇函数 \quad C. 既不是奇函数也不是偶函数 \quad D. 既是奇函数又是偶函数
\end{baitap}
\begin{baitap}[4]
下列函数中，在区间 $(0, +\infty)$ 上是增函数的是（ ）
\par\vspace{3pt}
\noindent A. $y = \dfrac{1}{x}$ \quad B. $y = -x^2$ \quad C. $y = \sqrt{x}$ \quad D. $y = \cos x$
\end{baitap}
\begin{baitap}[5]
函数 $f(x) = \sin\!\left(4x - \dfrac{\pi}{3}\right)$ 的最小正周期是（ ）
\par\vspace{3pt}
\noindent A. $\pi$ \quad B. $2\pi$ \quad C. $\dfrac{\pi}{2}$ \quad D. $\dfrac{\pi}{4}$
\end{baitap}
\begin{baitap}[6]
若 $\sin\alpha = a$，$\cos\alpha = b$，则 $\tan\dfrac{\alpha}{2}$ 的值为（ ）
\par\vspace{3pt}
\noindent A. $\dfrac{a}{1+b}$ \quad B. $\dfrac{a}{1-b}$ \quad C. $\sqrt{\dfrac{1-b}{1+b}}$ \quad D. $-\sqrt{\dfrac{1-b}{1+b}}$
\end{baitap}
