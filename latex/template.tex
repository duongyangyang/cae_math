% NỘI DUNG CHUYÊN ĐỀ — thay bằng kiến thức thật

\begin{dinhnghia}[1 — Tập hợp]
  Phát biểu định nghĩa. Ký hiệu toán học đặt trong dấu đô-la, ví dụ \(A \cup B\), \(A \cap B\).
\end{dinhnghia}

\begin{dinhly}[1 — Tính chất]
  Phát biểu định lý / tính chất.
\end{dinhly}

\begin{proof}
  Chứng minh chi tiết được viết \emph{bên dưới} định lý và \emph{ngoài} viền của khung \texttt{dinhly}. Dấu $\blacksquare$ được chèn tự động.
\end{proof}

\textbf{Chú ý.} Lưu ý, lỗi thường gặp, hoặc mẹo giải nhanh — viết thẳng thành đoạn văn, \emph{không} cần khung.
