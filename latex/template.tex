\section{Từ vựng và từ khóa nhận dạng đề}
% NỘI DUNG CHUYÊN ĐỀ — thay bằng nội dung thật

\textbf{Từ vựng.} Thuật ngữ Hán kèm nghĩa tiếng Việt, ví dụ \cjk{交集} (giao), \cjk{并集} (hợp), \cjk{补集} (phần bù).

\textbf{Từ khóa nhận dạng.} Gặp \cjk{恒成立} (đúng với mọi giá trị của biến) → bài toán tìm khoảng tham số.

\section{Hệ thống hóa kiến thức trọng tâm}

\begin{dinhnghia}[1 — Tập hợp]
  Phát biểu định nghĩa. Ký hiệu toán học đặt trong dấu đô-la, ví dụ \(A \cup B\), \(A \cap B\).
\end{dinhnghia}

\begin{dinhly}[1 — Tính chất]
  Phát biểu định lý / tính chất.
\end{dinhly}

\begin{proof}
  Chứng minh chi tiết được viết \emph{bên dưới} định lý và \emph{ngoài} viền của khung \texttt{dinhly}. Dấu $\blacksquare$ được chèn tự động.
\end{proof}

\textbf{Chú ý.} Lỗi thường gặp hoặc kỹ thuật xử lý nhanh — viết thẳng thành đoạn văn, \emph{không} cần khung.

  \textbf{Giải.} Trình bày lời giải từng bước, có căn chỉnh công thức bằng \texttt{align}.
